24struct PlaneLineIter<'a> {
25  tile: &'a Tile,
26  plane: &'a Plane,
27  codec: CodecParams,
28  params: Vec<BandParam<'a>>,
29  iwt_transforms: Vec<WaveletTransform>,
30  //plane_buf: Vec<i32>,
31  next_row: usize,
32}
33
34impl<'a> PlaneLineIter<'a> {
35  /// Create a new PlaneLine iterator for decoding
36  fn new(codec: CodecParams, tile: &'a Tile, plane: &'a Plane, mdat: &'a [u8]) -> Result<Self> {
37    // Some checks for correct input
38    assert!(tile.plane_height > 0);
39    assert!(tile.plane_width > 0);
40
41    // Reference to data section in MDAT
42    // All calculated offsets are relative to the data section.
43    let data = codec.get_data(mdat);
44
45    let plane_mdat_offset =
46      tile.data_offset + tile.qp_data.as_ref().map(|qp| qp.mdat_qp_data_size + qp.mdat_extra_size as u32).unwrap_or(0) as usize + plane.data_offset;
47
48    let mut params = Vec::with_capacity(plane.subbands.len());
49    for (band_id, band) in plane.subbands.iter().enumerate() {
50      let band_mdat_offset = plane_mdat_offset + band.data_offset;
51      debug!("Band {} has MDAT offset: {}", band_id, band_mdat_offset);
52      let band_buf = &data[band_mdat_offset..band_mdat_offset + band.data_size];
53      // Line length is subband + one additional pixel at start and end
54      let line_len = 1 + band.width + 1;
55      let bitpump = BitReader::endian(Cursor::new(band_buf), bitstream_io::BigEndian);
56
57      let param = BandParam {
58        subband_width: band.width,
59        subband_height: band.height,
60        rounded_bits_mask: if plane.support_partial && band_id == 0 { plane.rounded_bits_mask } else { 0 },
61        rounded_bits: 0,
62        cur_line: 0,
63        line_buf: [vec![0; line_len], vec![0; line_len]],
64        line_k: vec![0; line_len],
65        line_pos: 0,
66        line_len,
67        s_param: 0,
68        q_param: band.q_param,
69        supports_partial: plane.support_partial && band_id == 0,
70        rice: RiceDecoder::new(bitpump),
71      };
72      params.push(param);
73    }
74
75    let mut iwt_transforms = Vec::with_capacity(codec.levels);
76
77    if codec.levels > 0 {
78      // create Wavelet transforms
79      for level in 0..codec.levels {
80        let band = 3 * level + 1;
81        let (height, width) = if level >= codec.levels - 1 {
82          (tile.plane_height, tile.plane_width)
83        } else {
84          (plane.subbands[band + 3].height, plane.subbands[band + 4].width)
85        };
86        iwt_transforms.push(WaveletTransform::new(height, width));
87      }
88      codec.idwt_53_filter_init(tile, plane, &mut params, &mut iwt_transforms, codec.levels)?;
89    }
90
91    Ok(Self {
92      params,
93      tile,
94      plane,
95      codec,
96      iwt_transforms,
97      next_row: 0,
98    })
99  }
100
101  /// Decode a single line from plane
102  fn decode_plane_line(&mut self) -> Result<&[i32]> {
103    if self.next_row < self.tile.plane_height {
104      self.next_row += 1;
105      if self.codec.levels > 0 {
106        self
107          .codec
108          .idwt_53_filter_decode(self.tile, self.plane, &mut self.params, &mut self.iwt_transforms, self.codec.levels - 1)?;
109        self
110          .codec
111          .idwt_53_filter_transform(self.tile, self.plane, &mut self.params, &mut self.iwt_transforms, self.codec.levels - 1)?;
112        let line_data = self.iwt_transforms[self.codec.levels - 1].getline();
113        debug_assert_eq!(line_data.len(), self.tile.plane_width);
114        Ok(line_data)
115      } else {
116        debug_assert_eq!(self.plane.subbands.len(), 1);
117        let param = &mut self.params[0];
118        self.codec.decode_line(param)?;
119        let line_data = param.decoded_buf();
120        debug_assert_eq!(line_data.len(), param.subband_width as usize);
121        debug_assert_eq!(line_data.len(), self.tile.plane_width);
122        Ok(line_data)
123      }
124    } else {
125      Err(CrxError::General("All rows processed, can't decode more".to_string()))
126    }
127  }
128}

130/// Iterator over a plane, returning on each call a new decoded line
131impl<'a> Iterator for PlaneLineIter<'a> {
132  type Item = Result<PlaneLine>;
133
134  fn next(&mut self) -> Option<Self::Item> {
135    if self.next_row < self.tile.plane_height {
136      match self.decode_plane_line() {
137        Ok(line) => Some(Ok(line.into())),
138        Err(e) => Some(Err(e)),
139      }
140    } else {
141      None
142    }
143  }
144
145  fn size_hint(&self) -> (usize, Option<usize>) {
146    (0, Some(self.tile.plane_height))
147  }
148}

153/// Wrapper for PlaneLineIter
154/// Decodes a complete plane and returns it as vector of lines
155fn decode_full_plane(codec: &CodecParams, tile: &Tile, plane: &Plane, mdat: &[u8]) -> Result<Vec<PlaneLine>> {

159}

161impl CodecParams {
162  /// Decode MDAT section into a single CFA image
163  ///
164  /// Decoding processes all planes in all tiles and assembles the
165  /// decoded planes into proper tile output position and CFA pattern.
166  pub fn decode(mut self, mdat: &[u8]) -> Result<Vec<u16>> {
167    let instant = Instant::now();
168    debug!("Tile configuration: rows: {}, columns: {}", self.tile_rows, self.tile_cols);
169    // Build nested Tiles/Planes/Bands
170    let mut tiles = parse_header(self.get_header(mdat))?;
171    self.process_tiles(&mut tiles);
172    for tile in tiles.iter_mut() {
173      tile.generate_qstep_table(&self, self.get_data(mdat))?;
174    }
175
176    // cfa output is of final resolution
177    let mut cfa: Vec<u16> = vec![0; self.resolution()];
178
179    // Combine all tiles and planes into parallel iterators
180    // and decode the full planes.
181    let plane_bufs: Result<Vec<Vec<Vec<PlaneLine>>>> = tiles
182      .par_iter()
183      .map(|tile| tile.planes.par_iter().map(move |plane| decode_full_plane(&self, tile, plane, mdat)).collect())
184      .collect();
185
186    // Now we have a list of tiles->planes->plane-lines
187    // and can combine them to the final CFA
188    match plane_bufs {
189      Ok(bufs) => {
190        for (tile_id, tile) in bufs.into_iter().enumerate() {
191          let plane_count = tile.len();
192          debug_assert_eq!(plane_count, 4);
193          // Convert vector of planes to excact count of 4 planes - or fail
194          let planes: [Vec<PlaneLine>; 4] = tile
195            .try_into()
196            .map_err(|_| CrxError::General(format!("Invalid plane count {} (expected 4) for tile {}", plane_count, tile_id)))?;
197          // References to all 4 plane buffers
198          let (p0, p1, p2, p3) = (&planes[0], &planes[1], &planes[2], &planes[3]);
199          // Process each PlaneLine in all 4 buffers
200          for (plane_row, (l0, l1, l2, l3)) in izip!(p0, p1, p2, p3).enumerate() {
201            let (c0, c1, c2, c3) = convert_plane_line(&self, l0, l1, l2, l3)?;
202            integrate_cfa(&self, &tiles, &mut cfa, tile_id, 0, plane_row, &c0)?;
203            integrate_cfa(&self, &tiles, &mut cfa, tile_id, 1, plane_row, &c1)?;
204            integrate_cfa(&self, &tiles, &mut cfa, tile_id, 2, plane_row, &c2)?;
205            integrate_cfa(&self, &tiles, &mut cfa, tile_id, 3, plane_row, &c3)?;
206          }
207        }
208      }
209      Err(e) => {
210        return Err(e);
211      }
212    }
213    debug!("MDAT decoding and CFA build: {} s", instant.elapsed().as_secs_f32());
214    Ok(cfa)
215  }

217  /// Decode top line without a previous K buffer
218  fn decode_top_line_no_ref_prev_line(&self, p: &mut BandParam) -> Result<()> {
219    debug_assert_eq!(p.line_pos, 1);
220    let mut remaining = p.subband_width as u32;
221    // Init coef a and c (real image pixel starts at 1)
222    p.line_buf[0][p.line_pos - 1] = 0; // is [0] because at start line_pos is 1
223    p.line_buf[1][p.line_pos - 1] = 0; // is [0] because at start line_pos is 1
224    while remaining > 1 {
225      //println!("remaining: {}", remaining);
226      // Loop over full width of line (backwards)
227      if p.coeff_a() != 0 {
228        //println!("coeff {} is != 0", p.coeff_a());
229        let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
230        p.line_buf[1][p.line_pos] = error_code_signed(bit_code);
231      } else {
232        //println!("coeff {} = 0", p.coeff_a());
233        if p.rice.bitstream_get_bits(1)? == 1 {
234          let n_syms = self.symbol_run_count(p, remaining)?;
235          //println!("found {} syms", n_syms);
236          remaining = remaining.saturating_sub(n_syms);
237          // copy symbol n_syms times
238          for _ in 0..n_syms {
239            // For the first line, run-length coding uses only the symbol
240            // value 0, so we can fill the line buffer and K buffer with 0.
241            p.line_buf[1][p.line_pos] = 0;
242            p.line_k[p.line_pos - 1] = 0;
243            p.line_pos += 1;
244          }
245
246          if remaining == 0 {
247            break;
248          }
249        } // if bitstream == 1
250
251        let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
252        p.line_buf[1][p.line_pos] = error_code_signed(bit_code + 1); // Caution: + 1
253        //println!("code: {}", p.line_buf[1][p.line_pos]);
254      }
255      p.line_k[p.line_pos - 1] = p.rice.k();
256      p.line_pos += 1;
257      remaining = remaining.saturating_sub(1);
258    }
259    // Remaining pixel?
260    if remaining == 1 {
261      let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
262      p.line_buf[1][p.line_pos] = error_code_signed(bit_code);
263      p.line_k[p.line_pos - 1] = p.rice.k();
264      p.line_pos += 1;
265    }
266    debug_assert!(p.line_pos < p.line_buf[1].len());
267    p.line_buf[1][p.line_pos] = 0;
268    Ok(())
269  }

271  /// Decode nontop line with a previous K buffer
272  fn decode_nontop_line_no_ref_prev_line(&self, p: &mut BandParam) -> Result<()> {
273    //println!("Decode nontop {}", p.cur_line);
274    debug_assert_eq!(p.line_pos, 1);
275    let mut remaining = p.subband_width as u32;
276    while remaining > 1 {
277      // Loop over full width of line (backwards)
278      if (p.coeff_d() | p.coeff_b() | p.coeff_a()) != 0 {
279        let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, 0)?;
280        p.line_buf[1][p.line_pos] = error_code_signed(bit_code);
281        if p.line_k[p.line_pos].saturating_sub(p.rice.k()) <= 1 {
282          if p.rice.k() >= 15 {
283            p.rice.set_k(15);
284          }
285        } else {
286          p.rice.set_k(p.rice.k() + 1);
287        }
288      } else {
289        if p.rice.bitstream_get_bits(1)? == 1 {
290          debug_assert!(remaining != 1);
291          let n_syms = self.symbol_run_count(p, remaining)?;
292
293          remaining = remaining.saturating_sub(n_syms);
294          // copy symbol n_syms times
295          for _ in 0..n_syms {
296            // For the first line, run-length coding uses only the symbol
297            // value 0, so we can fill the line buffer and K buffer with 0.
298            p.line_buf[1][p.line_pos] = 0;
299            p.line_k[p.line_pos - 1] = 0;
300            p.line_pos += 1;
301          }
302        } // if bitstream == 1
303
304        if remaining <= 1 {
305          if remaining == 1 {
306            let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
307            p.line_buf[1][p.line_pos] = error_code_signed(bit_code + 1);
308            p.line_k[p.line_pos - 1] = p.rice.k();
309            p.line_pos += 1;
310            remaining = remaining.saturating_sub(1); // skip remaining check at end of function
311          }
312          break;
313        } else {
314          let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, 0)?;
315          p.line_buf[1][p.line_pos] = error_code_signed(bit_code + 1); // Caution: + 1
316          if p.line_k[p.line_pos].saturating_sub(p.rice.k()) <= 1 {
317            if p.rice.k() >= 15 {
318              p.rice.set_k(15);
319            }
320          } else {
321            p.rice.set_k(p.rice.k() + 1);
322          }
323        }
324      }
325      p.line_k[p.line_pos - 1] = p.rice.k();
326      p.line_pos += 1;
327      remaining = remaining.saturating_sub(1);
328    }
329    // Remaining pixel?
330    if remaining == 1 {
331      let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
332      p.line_buf[1][p.line_pos] = error_code_signed(bit_code);
333      p.line_k[p.line_pos - 1] = p.rice.k();
334      p.line_pos += 1;
335    }
336    debug_assert!(p.line_pos < p.line_buf[1].len());
337    Ok(())
338  }

340  /// Decode top line
341  /// For the first line (top) in a plane, no MED is used because
342  /// there is no previous line for coeffs b, c and d.
343  /// So this decoding is a simplified version from decode_nontop_line().
344  fn decode_top_line(&self, p: &mut BandParam) -> Result<()> {
345    debug_assert_eq!(p.line_pos, 1);
346    let mut remaining = p.subband_width as u32;
347    // Init coeff a (real image pixel starts at 1)
348    p.line_buf[1][p.line_pos - 1] = 0; // is is [0] because at start line_pos is 1
349    while remaining > 1 {
350      // Loop over full width of line (backwards)
351      if p.coeff_a() != 0 {
352        p.line_buf[1][p.line_pos] = p.coeff_a();
353      } else {
354        if p.rice.bitstream_get_bits(1)? == 1 {
355          let n_syms = self.symbol_run_count(p, remaining)?;
356          remaining = remaining.saturating_sub(n_syms);
357          // copy symbol n_syms times
358          for _ in 0..n_syms {
359            p.line_buf[1][p.line_pos] = p.coeff_a();
360            p.line_pos += 1;
361          }
362          if remaining == 0 {
363            break;
364          }
365        } // if bitstream == 1
366        p.line_buf[1][p.line_pos] = 0;
367      }
368      let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
369      p.line_buf[1][p.line_pos] += error_code_signed(bit_code);
370      p.line_pos += 1;
371      remaining = remaining.saturating_sub(1);
372    }
373    // Remaining pixel?
374    if remaining == 1 {
375      let x = p.coeff_a(); // no MED, just use coeff a
376      let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
377      p.line_buf[1][p.line_pos] = x + error_code_signed(bit_code);
378      p.line_pos += 1;
379    }
380    debug_assert!(p.line_pos < p.line_buf[1].len());
381    p.line_buf[1][p.line_pos] = p.coeff_a() + 1;
382    Ok(())
383  }

385  /// Decode a line which is not a top line
386  /// This used run length coding, Median Edge Detection (MED) and
387  /// adaptive Golomb-Rice entropy encoding.
388  /// Golomb-Rice becomes more efficient when using an adaptive K value
389  /// instead of a fixed one.
390  /// The K parameter is used as q = n >> k where n is the sample to encode.
391  fn decode_nontop_line(&self, p: &mut BandParam) -> Result<()> {
392    debug_assert_eq!(p.line_pos, 1);
393    let mut remaining = p.subband_width as u32;
394    // Init coeff a: a = b
395    p.line_buf[1][p.line_pos - 1] = p.coeff_b();
396    // Loop over full width of line (backwards)
397    while remaining > 1 {
398      let mut x = 0;
399      //  c b d
400      //  a x n
401      // Median Edge Detection to predict pixel x. Described in patent US2016/0323602 and T.87
402      if p.coeff_a() == p.coeff_b() && p.coeff_a() == p.coeff_d() {
403        // different than step [0104], where Condition: "a=c and c=b and b=d", c not used
404        if p.rice.bitstream_get_bits(1)? == 1 {
405          let n_syms = self.symbol_run_count(p, remaining)?;
406          remaining = remaining.saturating_sub(n_syms);
407          // copy symbol n_syms times
408          for _ in 0..n_syms {
409            p.line_buf[1][p.line_pos] = p.coeff_a();
410            p.line_pos += 1;
411          }
412        } // if bitstream == 1
413        if remaining > 0 {
414          x = p.coeff_b(); // use new coeff b because we moved line_pos!
415        }
416      } else {
417        // no run length coding, use MED instead
418        x = med(p.coeff_a(), p.coeff_b(), p.coeff_c());
419      }
420      if remaining > 0 {
421        let mut bit_code = p.rice.adaptive_rice_decode(false, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
422        // add converted (+/-) error code to predicted value
423        p.line_buf[1][p.line_pos] = x + error_code_signed(bit_code);
424        // for not end of the line - use one symbol ahead to estimate next K
425        if remaining > 1 {
426          let delta: i32 = (p.coeff_d() - p.coeff_b()) << 1;
427          bit_code = (bit_code + delta.unsigned_abs()) >> 1;
428        }
429        p.rice.update_k_param(bit_code, PREDICT_K_MAX);
430        p.line_pos += 1;
431      }
432      remaining = remaining.saturating_sub(1);
433    } // end while length > 1
434    // Remaining pixel?
435    if remaining == 1 {
436      let x = med(p.coeff_a(), p.coeff_b(), p.coeff_c());
437      let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
438      // add converted (+/-) error code to predicted value
439      p.line_buf[1][p.line_pos] = x + error_code_signed(bit_code);
440      p.line_pos += 1;
441    }
442    debug_assert!(p.line_pos < p.line_buf[1].len());
443    p.line_buf[1][p.line_pos] = p.coeff_a() + 1;
444    Ok(())
445  }

447  /// Decode a symbol x in rounded mode.
448  /// Used only when levels==0 (lossless mode)
449  fn decode_symbol_rounded(&self, p: &mut BandParam, use_med: bool, not_eol: bool) -> Result<()> {
450    let sym = if use_med { med(p.coeff_a(), p.coeff_b(), p.coeff_c()) } else { p.coeff_b() };
451    let bit_code = p.rice.adaptive_rice_decode(false, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
452    let mut code = error_code_signed(bit_code);
453    let x = p.rounded_bits_mask * 2 * code + (code >> 31);
454    p.line_buf[1][p.line_pos] = x + sym;
455
456    if not_eol {
457      if p.coeff_d() > p.coeff_b() {
458        code = (p.coeff_d() - p.coeff_b() + p.rounded_bits_mask - 1) >> p.rounded_bits;
459      } else {
460        code = -((p.coeff_b() - p.coeff_d() + p.rounded_bits_mask) >> p.rounded_bits);
461      }
462      p.rice.update_k_param((bit_code + 2 * code.unsigned_abs()) >> 1, PREDICT_K_MAX);
463    } else {
464      p.rice.update_k_param(bit_code, PREDICT_K_MAX);
465    }
466
467    p.line_pos += 1;
468    Ok(())
469  }

471  /// Decode a rounded line which is not a top line
472  fn decode_top_line_rounded(&self, p: &mut BandParam) -> Result<()> {
473    debug_assert_eq!(p.line_pos, 1);
474    let mut remaining = p.subband_width as u32;
475    // Init coeff a (real image pixel starts at 1)
476    p.line_buf[1][p.line_pos - 1] = 0; // is is [0] because at start line_pos is 1
477    while remaining > 1 {
478      // Loop over full width of line (backwards)
479      if p.coeff_a().abs() > p.rounded_bits_mask {
480        p.line_buf[1][p.line_pos] = p.coeff_a();
481      } else {
482        if p.rice.bitstream_get_bits(1)? == 1 {
483          let n_syms = self.symbol_run_count(p, remaining)?;
484          remaining = remaining.saturating_sub(n_syms);
485          // copy symbol n_syms times
486          for _ in 0..n_syms {
487            p.line_buf[1][p.line_pos] = p.coeff_a();
488            p.line_pos += 1;
489          }
490          if remaining == 0 {
491            break;
492          }
493        } // if bitstream == 1
494        p.line_buf[1][p.line_pos] = 0;
495      }
496      let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
497      let code = error_code_signed(bit_code);
498      p.line_buf[1][p.line_pos] += p.rounded_bits_mask * 2 * code + (code >> 31);
499      p.line_pos += 1;
500      remaining = remaining.saturating_sub(1);
501    }
502    // Remaining pixel?
503    if remaining == 1 {
504      let bit_code = p.rice.adaptive_rice_decode(true, PREDICT_K_ESCAPE, PREDICT_K_ESCBITS, PREDICT_K_MAX)?;
505      let code = error_code_signed(bit_code);
506      p.line_buf[1][p.line_pos] += p.rounded_bits_mask * 2 * code + (code >> 31);
507      p.line_pos += 1;
508    }
509    debug_assert!(p.line_pos < p.line_buf[1].len());
510    p.line_buf[1][p.line_pos] = p.coeff_a() + 1;
511    Ok(())
512  }
513
514  /// Decode a line which is not a top line
515  /// This used run length coding, Median Edge Detection (MED) and
516  /// adaptive Golomb-Rice entropy encoding.
517  /// Golomb-Rice becomes more efficient when using an adaptive K value
518  /// instead of a fixed one.
519  /// The K parameter is used as q = n >> k where n is the sample to encode.
520  #[allow(clippy::comparison_chain)]
521  fn decode_nontop_line_rounded(&self, p: &mut BandParam) -> Result<()> {
522    debug_assert_eq!(p.line_pos, 1);
523    let mut remaining = p.subband_width as u32;
524    let mut value_reached = false;
525    p.line_buf[0][p.line_pos - 1] = p.coeff_b();
526    p.line_buf[1][p.line_pos - 1] = p.coeff_b();
527    // Loop over full width of line (backwards)
528    while remaining > 1 {
529      if (p.coeff_d() - p.coeff_b()).abs() > p.rounded_bits_mask {
530        self.decode_symbol_rounded(p, true, true)?;
531        value_reached = true;
532      } else if value_reached || (p.coeff_c() - p.coeff_a()).abs() > p.rounded_bits_mask {
533        self.decode_symbol_rounded(p, true, true)?;
534        value_reached = false;
535      } else {
536        if p.rice.bitstream_get_bits(1)? == 1 {
537          let n_syms = self.symbol_run_count(p, remaining)?;
538          remaining = remaining.saturating_sub(n_syms);
539          // copy symbol n_syms times
540          for _ in 0..n_syms {
541            p.line_buf[1][p.line_pos] = p.coeff_a();
542            p.line_pos += 1;
543          }
544        } // if bitstream == 1
545        if remaining > 1 {
546          self.decode_symbol_rounded(p, false, true)?;
547          value_reached = (p.coeff_b() - p.coeff_c()).abs() > p.rounded_bits_mask;
548        } else if remaining == 1 {
549          self.decode_symbol_rounded(p, false, false)?;
550        }
551      }
552      remaining = remaining.saturating_sub(1);
553    } // end while length > 1
554    // Remaining pixel?
555    if remaining == 1 {
556      self.decode_symbol_rounded(p, true, false)?;
557    }
558    debug_assert!(p.line_pos < p.line_buf[1].len());
559    p.line_buf[1][p.line_pos] = p.coeff_a() + 1;
560    Ok(())
561  }
562
563  /// Decode a single line from input band
564  /// For decoding, two line buffers are required (except for the first line).
565  /// After each decoding line, the two buffers are swapped, so the previous one
566  /// is always in line_buf[0] (containing coefficents c, b, d) and the current
567  /// line is in line_buf[1] (containing coefficents a, x, n).
568  ///
569  /// The line buffers has an extra sample on both ends. So the buffer layout is:
570  ///
571  /// |E|Samples........................|E|
572  /// |c|bd                           cb|d|
573  /// |a|xn                           ax|n|
574  ///  ^ ^                               ^
575  ///  | |                               |-- Extra sample to provide fake d coefficent
576  ///  | |---- First sample value
577  ///  |------ Extra sample to provide a fake a/c coefficent
578  ///
579  /// After line is decoded, the E samples are ignored when
580  /// copied into the final plane buffer.
581  ///
582  /// For non-LL bands, decoding process differs a little bit
583  /// because some value rounding is added.
584  pub(super) fn decode_line(&self, param: &mut BandParam) -> Result<()> {
585    debug_assert!(param.cur_line < param.subband_height);
586    // We start at first real pixel value
587    param.line_pos = 1;
588    if param.cur_line == 0 {
589      param.s_param = 0;
590      param.rice.set_k(0); // TODO: required?
591      if param.supports_partial {
592        if param.rounded_bits_mask <= 0 {
593          self.decode_top_line(param)?;
594        } else {
595          param.rounded_bits = 1;
596          if (param.rounded_bits_mask & !1) != 0 {
597            while param.rounded_bits_mask >> param.rounded_bits != 0 {
598              param.rounded_bits += 1;
599            }
600          }
601          self.decode_top_line_rounded(param)?;
602        }
603      } else {
604        self.decode_top_line_no_ref_prev_line(param)?;
605      }
606    } else if !param.supports_partial {
607      // Swap line buffers so previous decoded (1) is now above (0)
608      param.line_buf.swap(0, 1);
609      self.decode_nontop_line_no_ref_prev_line(param)?;
610    } else if param.rounded_bits_mask <= 0 {
611      // Swap line buffers so previous decoded (1) is now above (0)
612      param.line_buf.swap(0, 1);
613      self.decode_nontop_line(param)?;
614    } else {
615      // Swap line buffers so previous decoded (1) is now above (0)
616      param.line_buf.swap(0, 1);
617      self.decode_nontop_line_rounded(param)?;
618    }
619    param.cur_line += 1;
620    Ok(())
621  }
622}

624/// Constrain a given value into min/max
625pub(super) fn constrain(value: i32, min: i32, max: i32) -> i32 {
626  std::cmp::min(std::cmp::max(value, min), max)
627  /*
628  let res = if value < min {
629    min
630  } else if value > max {
631    max
632  } else {
633    value
634  };
635  debug_assert!(res <= u16::MAX as i32);
636  res
637   */
638}

640/// The error code contains a sign bit at bit 0.
641/// Example: 10010 1 -> negative value, 10010 0 -> positive value
642/// This routine converts an unsigned bit_code to the correct
643/// signed integer value.
644/// For this, the sign bit is inverted and XOR with
645/// the shifted integer value.
646pub(super) fn error_code_signed(bit_code: u32) -> i32 {
647  -((bit_code & 1) as i32) ^ (bit_code >> 1) as i32
648}

650/// Median Edge Detection
651/// [0053] Obtains a predictive value p of the coefficient by using
652/// MED prediction, thereby performing predictive coding.
653pub(super) fn med(a: i32, b: i32, c: i32) -> i32 {
654  if c >= std::cmp::max(a, b) {
655    std::cmp::min(a, b)
656  } else if c <= std::cmp::min(a, b) {
657    std::cmp::max(a, b)
658  } else {
659    a + b - c // no edge detected
660  }
661}

663/// Convert a decoded line to plane output
664/// Results from decode_line() are signed 32 bit integers.
665/// By using a median and max value, these are converted
666/// to unsigned 16 bit integers.
667#[allow(clippy::type_complexity)]
668fn convert_plane_line(codec: &CodecParams, l0: &[i32], l1: &[i32], l2: &[i32], l3: &[i32]) -> Result<(Vec<u16>, Vec<u16>, Vec<u16>, Vec<u16>)> {
669  let mut p0 = vec![0; l0.len()];
670  let mut p1 = vec![0; l1.len()];
671  let mut p2 = vec![0; l2.len()];
672  let mut p3 = vec![0; l3.len()];
673
674  match codec.enc_type {
675    0 => {
676      let median: i32 = 1 << (codec.median_bits - 1);
677      let max_val: i32 = (1 << codec.median_bits) - 1;
678
679      izip!(l0, l1, l2, l3).enumerate().for_each(|(i, (v0, v1, v2, v3))| {
680        p0[i] = constrain(median + v0, 0, max_val) as u16;
681        p1[i] = constrain(median + v1, 0, max_val) as u16;
682        p2[i] = constrain(median + v2, 0, max_val) as u16;
683        p3[i] = constrain(median + v3, 0, max_val) as u16;
684      });
685    }
686    3 => {
687      let median: i32 = 1 << (codec.median_bits - 1) << 10;
688      let max_val: i32 = (1 << codec.median_bits) - 1;
689
690      izip!(l0, l1, l2, l3).enumerate().for_each(|(i, (v0, v1, v2, v3))| {
691        let mut gr: i32 = median + (v0 << 10) - 168 * v1 - 585 * v3;
692        if gr < 0 {
693          gr = -(((gr.abs() + 512) >> 9) & !1);
694        } else {
695          gr = ((gr.abs() + 512) >> 9) & !1;
696        }
697        p0[i] = constrain((median + (v0 << 10) + 1510 * v3 + 512) >> 10, 0, max_val) as u16;
698        p1[i] = constrain((v2 + gr + 1) >> 1, 0, max_val) as u16;
699        p2[i] = constrain((gr - v2 + 1) >> 1, 0, max_val) as u16;
700        p3[i] = constrain((median + (v0 << 10) + 1927 * v1 + 512) >> 10, 0, max_val) as u16;
701      });
702    }
703    enc_type => {
704      return Err(CrxError::General(format!("Unsupported encoding type {}", enc_type)));
705    }
706  }
707
708  Ok((p0, p1, p2, p3))
709}

711/// Integrate a plane buffer into CFA output image
712///
713/// A plane is a single monochrome image for one of the four CFA colors.
714/// `plane_id` is 0, 1, 2 or 3 for R, G1, G2, B
715fn integrate_cfa(codec: &CodecParams, tiles: &[Tile], cfa_buf: &mut [u16], tile_id: usize, plane_id: usize, plane_row: usize, plane_buf: &[u16]) -> Result<()> {
716  // 2x2 pixel for RGGB
717  const CFA_DIM: usize = 2;
718
719  debug_assert_ne!(plane_buf.len(), 0);
720  debug_assert_ne!(cfa_buf.len(), 0);
721  debug_assert!(codec.tile_cols > 0);
722  debug_assert!(codec.tile_rows > 0);
723
724  if plane_id > 3 {
725    return Err(CrxError::Overflow(format!(
726      "More then 4 planes detected, unable to process plane_id {}",
727      plane_id
728    )));
729  }
730
731  let tile_row_idx = tile_id / codec.tile_cols; // round down
732  let tile_col_idx = tile_id % codec.tile_cols; // round down
733
734  // Offset from top
735  let row_offset = tile_row_idx * codec.tile_width;
736  // Offset from left
737  let col_offset = tile_col_idx * codec.tile_width;
738  let (row_shift, col_shift) = match plane_id {
739    0 => (0, 0),
740    1 => (0, 1),
741    2 => (1, 0),
742    3 => (1, 1),
743    _ => {
744      return Err(CrxError::General("Invalid plane id".to_string()));
745    }
746  };
747  //println!("plane_width: {}, buf_size: {}", tiles[tile_id].plane_width, plane_buf.len());
748  let row_idx = row_offset + (plane_row * CFA_DIM) + row_shift;
749  for plane_col in 0..tiles[tile_id].plane_width {
750    // Row index into CFA for untiled full area
751    let col_idx = col_offset + (plane_col * CFA_DIM) + col_shift;
752
753    // Copy from plane to CFA
754    cfa_buf[(row_idx * codec.image_width) + col_idx] = plane_buf[plane_col];
755  }
756  Ok(())
757}
