8use super::BitPump;
9use super::Result;
10use bitstream_io::BitRead;
11
12/// Adaptive Golomb-Rice decoder
13pub(super) struct RiceDecoder<'mdat> {
14  /// Bitstream from MDAT
15  bitpump: BitPump<'mdat>,
16  k_param: u32,
17}
18
19impl<'mdat> RiceDecoder<'mdat> {
20  /// Create new decoder for given bit pump
21  pub(super) fn new(bitpump: BitPump<'mdat>) -> Self {
22    Self { bitpump, k_param: 0 }
23  }
24
25  /// Get current K parameter
26  #[inline(always)]
27  pub(super) fn k(&self) -> u32 {
28    self.k_param
29  }
30
31  /// Set K parameter
32  #[inline(always)]
33  pub(super) fn set_k(&mut self, k: u32) {
34    self.k_param = k;
35  }
36
37  /// Return the positive number of 0-bits in bitstream.
38  /// All 0-bits are consumed.
39  #[inline(always)]
40  pub(super) fn bitstream_zeros(&mut self) -> Result<u32> {
41    Ok(self.bitpump.read_unary::<1>()?)
42  }
43
44  /// Return the requested bits
45  // All bits are consumed.
46  // The maximum number of bits are 32
47  #[inline(always)]
48  pub(super) fn bitstream_get_bits(&mut self, bits: u32) -> Result<u32> {
49    debug_assert!(bits <= 32);
50    Ok(self.bitpump.read_var(bits)?)
51  }
52
53  /// Golomb-Rice decoding
54  /// https://w3.ual.es/~vruiz/Docencia/Apuntes/Coding/Text/03-symbol_encoding/09-Golomb_coding/index.html
55  /// escape and esc_bits are used to interrupt decoding when
56  /// a value is not encoded using Golomb-Rice but directly encoded
57  /// by esc_bits bits.
58  fn rice_decode(&mut self, escape: u32, esc_bits: u32) -> Result<u32> {
59    // q, quotient = n//m, with m = 2^k (Rice coding)
60    let prefix = self.bitstream_zeros()?;
61    if prefix >= escape {
62      // n
63      Ok(self.bitstream_get_bits(esc_bits)?)
64    } else if self.k_param > 0 {
65      // Golomb-Rice coding : n = q * 2^k + r, with r is next k bits. r is n - (q*2^k)
66      Ok((prefix << self.k_param) | self.bitstream_get_bits(self.k_param)?)
67    } else {
68      // q
69      Ok(prefix)
70    }
71  }
72
73  /// Adaptive Golomb-Rice decoding, by adapting k value
74  /// Sometimes adapting is based on the next coefficent (n) instead
75  /// of current (x) coefficent. So you can disable it with `adapt_k`
76  /// and update k later.
77  pub(super) fn adaptive_rice_decode(&mut self, adapt_k: bool, escape: u32, esc_bits: u32, k_max: u32) -> Result<u32> {
78    let val = self.rice_decode(escape, esc_bits)?;
79    if adapt_k {
80      self.k_param = Self::predict_k_param_max(self.k_param, val, k_max);
81    }
82    Ok(val)
83  }
84
85  /// Update current K parameter
86  pub(super) fn update_k_param(&mut self, bit_code: u32, k_max: u32) {
87    self.k_param = Self::predict_k_param_max(self.k_param, bit_code, k_max);
88  }
89
90  /// Predict K parameter with maximum constraint
91  /// Golomb-Rice becomes more efficient when used with an adaptive
92  /// K parameter. This is done by predicting the next K value for the
93  /// next sample value.
94  fn predict_k_param_max(prev_k: u32, value: u32, k_max: u32) -> u32 {
95    let mut new_k = prev_k;
96    if value >> prev_k > 2 {
97      new_k += 1;
98    }
99    if value >> prev_k > 5 {
100      new_k += 1;
101    }
102    if value < ((1 << prev_k) >> 1) {
103      new_k -= 1;
104    }
105
106    if k_max > 0 { std::cmp::min(new_k, k_max) } else { new_k }
107  }
108}
