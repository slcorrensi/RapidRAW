21impl Demosaic<f32, 3> for XTransDemosaic {
22  /// Demosaics an X-Trans image to full resolution using a gradient-based method.
23  /// This version is corrected to handle the non-uniform X-Trans CFA pattern,
24  /// avoiding artifacts by correctly identifying and averaging available neighbor pixels.
25  fn demosaic(&self, pixels: &PixF32, cfa: &CFA, _colors: &PlaneColor, roi: Rect) -> Color2D<f32, 3> {
26    let mut out = Color2D::<f32, 3>::new(roi.width(), roi.height());
27    let cfa = cfa.shift(roi.p.x, roi.p.y);
28
29    // Pass 1: Copy known sensor values into the correct channels of the output buffer.
30    for y in 0..roi.height() {
31      for x in 0..roi.width() {
32        let color_idx = cfa.color_at(y, x);
33        if color_idx < 3 {
34          // Ensure we only handle R, G, B
35          out.at_mut(y, x)[color_idx] = *pixels.at(roi.p.y + y, roi.p.x + x);
36        }
37      }
38    }
39
40    // Create a padded copy for easier border handling during interpolation
41    let mut padded = out.make_padded(2);
42
43    // Pass 2: Interpolate Green channel at R and B locations using gradient detection.
44    for y in 2..padded.height - 2 {
45      for x in 2..padded.width - 2 {
46        let roi_y = y - 2;
47        let roi_x = x - 2;
48        let color_idx = cfa.color_at(roi_y, roi_x);
49
50        if color_idx == CFA_COLOR_R || color_idx == CFA_COLOR_B {
51          // Gradients of the current color (R or B). In X-Trans, same-colored
52          // neighbors are typically 2 pixels away in cardinal directions.
53          let h_grad = (padded.at(y, x - 2)[color_idx] - padded.at(y, x + 2)[color_idx]).abs();
54          let v_grad = (padded.at(y - 2, x)[color_idx] - padded.at(y + 2, x)[color_idx]).abs();
55
56          let mut g_h_sum = 0.0;
57          let mut g_h_count = 0;
58          let mut g_v_sum = 0.0;
59          let mut g_v_count = 0;
60
61          // Find and sum horizontal Green neighbors
62          if roi_x > 0 && cfa.color_at(roi_y, roi_x - 1) == CFA_COLOR_G {
63            g_h_sum += padded.at(y, x - 1)[CFA_COLOR_G];
64            g_h_count += 1;
65          }
66          if cfa.color_at(roi_y, roi_x + 1) == CFA_COLOR_G {
67            g_h_sum += padded.at(y, x + 1)[CFA_COLOR_G];
68            g_h_count += 1;
69          }
70
71          // Find and sum vertical Green neighbors
72          if roi_y > 0 && cfa.color_at(roi_y - 1, roi_x) == CFA_COLOR_G {
73            g_v_sum += padded.at(y - 1, x)[CFA_COLOR_G];
74            g_v_count += 1;
75          }
76          if cfa.color_at(roi_y + 1, roi_x) == CFA_COLOR_G {
77            g_v_sum += padded.at(y + 1, x)[CFA_COLOR_G];
78            g_v_count += 1;
79          }
80
81          let g_h = if g_h_count > 0 { g_h_sum / g_h_count as f32 } else { 0.0 };
82          let g_v = if g_v_count > 0 { g_v_sum / g_v_count as f32 } else { 0.0 };
83
84          let g = if g_h_count == 0 && g_v_count == 0 {
85            0.0 // Fallback, should not happen for R/B in X-Trans
86          } else if g_h_count == 0 {
87            g_v // Only vertical Gs available
88          } else if g_v_count == 0 {
89            g_h // Only horizontal Gs available
90          } else {
91            // Both horizontal and vertical Gs exist, use gradient to decide.
92            if (h_grad - v_grad).abs() < 0.001 { // Gradients are similar
93              (g_h_sum + g_v_sum) / (g_h_count + g_v_count) as f32
94            } else if h_grad < v_grad { // Horizontal edge
95              g_h
96            } else { // Vertical edge
97              g_v
98            }
99          };
100          padded.at_mut(y, x)[CFA_COLOR_G] = g;
101        }
102      }
103    }
104
105    // Pass 3: Interpolate R/B at G locations
106    for y in 2..padded.height - 2 {
107      for x in 2..padded.width - 2 {
108        let roi_y = y - 2;
109        let roi_x = x - 2;
110        if cfa.color_at(roi_y, roi_x) == CFA_COLOR_G {
111          let mut r_sum = 0.0;
112          let mut r_count = 0;
113          let mut b_sum = 0.0;
114          let mut b_count = 0;
115
116          // Check cardinal neighbors with boundary checks
117          // Left
118          if roi_x > 0 {
119            match cfa.color_at(roi_y, roi_x - 1) {
120              CFA_COLOR_R => { r_sum += padded.at(y, x - 1)[CFA_COLOR_R]; r_count += 1; }
121              CFA_COLOR_B => { b_sum += padded.at(y, x - 1)[CFA_COLOR_B]; b_count += 1; }
122              _ => {}
123            }
124          }
125          // Right
126          match cfa.color_at(roi_y, roi_x + 1) {
127            CFA_COLOR_R => { r_sum += padded.at(y, x + 1)[CFA_COLOR_R]; r_count += 1; }
128            CFA_COLOR_B => { b_sum += padded.at(y, x + 1)[CFA_COLOR_B]; b_count += 1; }
129            _ => {}
130          }
131          // Top
132          if roi_y > 0 {
133            match cfa.color_at(roi_y - 1, roi_x) {
134              CFA_COLOR_R => { r_sum += padded.at(y - 1, x)[CFA_COLOR_R]; r_count += 1; }
135              CFA_COLOR_B => { b_sum += padded.at(y - 1, x)[CFA_COLOR_B]; b_count += 1; }
136              _ => {}
137            }
138          }
139          // Bottom
140          match cfa.color_at(roi_y + 1, roi_x) {
141            CFA_COLOR_R => { r_sum += padded.at(y + 1, x)[CFA_COLOR_R]; r_count += 1; }
142            CFA_COLOR_B => { b_sum += padded.at(y + 1, x)[CFA_COLOR_B]; b_count += 1; }
143            _ => {}
144          }
145
146          if r_count > 0 {
147            padded.at_mut(y, x)[CFA_COLOR_R] = r_sum / r_count as f32;
148          }
149          if b_count > 0 {
150            padded.at_mut(y, x)[CFA_COLOR_B] = b_sum / b_count as f32;
151          }
152        }
153      }
154    }
155
156    // Pass 4: Interpolate R at B and B at R
157    for y in 2..padded.height - 2 {
158      for x in 2..padded.width - 2 {
159        let roi_y = y - 2;
160        let roi_x = x - 2;
161        let color_idx = cfa.color_at(roi_y, roi_x);
162
163        if color_idx == CFA_COLOR_R || color_idx == CFA_COLOR_B {
164          let mut r_sum = 0.0;
165          let mut b_sum = 0.0;
166          let mut g_neighbor_count = 0;
167
168          // Interpolate using diagonal neighbors with boundary checks.
169          // We only use G neighbors, which now have interpolated R and B values from Pass 3.
170          
171          // Top-Left
172          if roi_y > 0 && roi_x > 0 && cfa.color_at(roi_y - 1, roi_x - 1) == CFA_COLOR_G {
173            r_sum += padded.at(y - 1, x - 1)[CFA_COLOR_R];
174            b_sum += padded.at(y - 1, x - 1)[CFA_COLOR_B];
175            g_neighbor_count += 1;
176          }
177          // Top-Right
178          if roi_y > 0 && cfa.color_at(roi_y - 1, roi_x + 1) == CFA_COLOR_G {
179            r_sum += padded.at(y - 1, x + 1)[CFA_COLOR_R];
180            b_sum += padded.at(y - 1, x + 1)[CFA_COLOR_B];
181            g_neighbor_count += 1;
182          }
183          // Bottom-Left
184          if roi_x > 0 && cfa.color_at(roi_y + 1, roi_x - 1) == CFA_COLOR_G {
185            r_sum += padded.at(y + 1, x - 1)[CFA_COLOR_R];
186            b_sum += padded.at(y + 1, x - 1)[CFA_COLOR_B];
187            g_neighbor_count += 1;
188          }
189          // Bottom-Right
190          if cfa.color_at(roi_y + 1, roi_x + 1) == CFA_COLOR_G {
191            r_sum += padded.at(y + 1, x + 1)[CFA_COLOR_R];
192            b_sum += padded.at(y + 1, x + 1)[CFA_COLOR_B];
193            g_neighbor_count += 1;
194          }
195
196          if g_neighbor_count > 0 {
197            if color_idx == CFA_COLOR_B {
198              // Interpolate R at B locations
199              padded.at_mut(y, x)[CFA_COLOR_R] = r_sum / g_neighbor_count as f32;
200            } else { // color_idx == CFA_COLOR_R
201              // Interpolate B at R locations
202              padded.at_mut(y, x)[CFA_COLOR_B] = b_sum / g_neighbor_count as f32;
203            }
204          }
205        }
206      }
207    }
208
209    // Crop the padding off to return the final image
210    padded.crop(Rect::new_with_points(
211      Point::new(2, 2),
212      Point::new(padded.width - 2, padded.height - 2),
213    ))
214  }
215}

226impl Demosaic<f32, 3> for XTransSuperpixelDemosaic {
227  /// Debayer an X-Trans image using a 6x6 superpixel method.
228  /// Each output RGB pixel is the average of all R, G, and B pixels
229  /// within a 6x6 block of the original sensor data.
230  /// The resulting image is 1/36th the size (1/6 width, 1/6 height).
231  fn demosaic(&self, pixels: &PixF32, cfa: &CFA, colors: &PlaneColor, roi: Rect) -> Color2D<f32, 3> {
232    // ROI width/height must be a multiple of 6 for this algorithm.
233    let roi = Rect::new(roi.p, Dim2::new(roi.width() / 6 * 6, roi.height() / 6 * 6));
234    let dim = pixels.dim();
235
236    // The CFA pattern must be shifted according to the ROI's top-left corner.
237    let cfa = cfa.shift(roi.p.x, roi.p.y);
238
239    // This lookup table maps a CFAColor (R,G,B) to its correct output channel index (0,1,2).
240    let plane_map = colors.plane_lookup_table();
241
242    // Get a slice of the image corresponding to the ROI's starting row.
243    let window = &pixels[roi.y() * dim.w..];
244
245    let out_data: Vec<[f32; 3]> = window
246      .par_chunks_exact(dim.w * 6) // Process 6 rows at a time
247      .take(roi.height() / 6) // Process roi.height() / 6 blocks of rows
248      .flat_map(|six_rows_slice| {
249        // six_rows_slice contains 6 full rows of the original image.
250        // We process them in 6-pixel wide chunks.
251        let rows: Vec<_> = (0..6)
252          .map(|i| &six_rows_slice[i * dim.w + roi.x()..i * dim.w + roi.x() + roi.width()])
253          .collect();
254
255        (0..roi.width() / 6)
256          .map(|block_x| {
257            let mut sums = [0.0f32; 3];
258            let mut counts = [0u32; 3];
259
260            for y_offset in 0..6 {
261              for x_offset in 0..6 {
262                // Get the color (R, G, or B) at this position from the shifted CFA.
263                let cfa_color_val = cfa.color_at(y_offset, x_offset);
264                if cfa_color_val < 3 {
265                  // Ensure it's R, G, or B
266                  // Use the plane_map to find the correct output channel for this color.
267                  let plane_index = plane_map[cfa_color_val];
268                  if plane_index < 3 {
269                    let pixel_value = rows[y_offset][block_x * 6 + x_offset];
270                    sums[plane_index] += pixel_value;
271                    counts[plane_index] += 1;
272                  }
273                }
274              }
275            }
276
277            // The sums are now in the correct R, G, B order thanks to plane_map.
278            let r = if counts[0] > 0 { sums[0] / counts[0] as f32 } else { 0.0 };
279            let g = if counts[1] > 0 { sums[1] / counts[1] as f32 } else { 0.0 };
280            let b = if counts[2] > 0 { sums[2] / counts[2] as f32 } else { 0.0 };
281
282            [r, g, b]
283          })
284          .collect::<Vec<_>>()
285      })
286      .collect();
287
288    Color2D::new_with(out_data, roi.width() / 6, roi.height() / 6)
289  }
