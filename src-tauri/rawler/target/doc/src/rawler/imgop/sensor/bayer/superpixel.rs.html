22impl Demosaic<f32, 3> for Superpixel3Channel {
23  /// Debayer image by using superpixel method.
24  /// Each output pixel RGB tuple is generated by 4 pixels from input.
25  /// The result image is 1/4 of size (half width, half height).
26  fn demosaic(&self, pixels: &PixF32, cfa: &CFA, colors: &PlaneColor, roi: Rect) -> Color2D<f32, 3> {
27    if colors.plane_count() != 3 {
28      panic!("Demosaic for 3 channels needs 3 color planes, but {} given", colors.plane_count());
29    }
30    if !cfa.is_rgb() {
31      panic!("Demosaic for 3 channels requires RGB CFA pattern, but CFA {} given", cfa);
32    }
33    // ROI width / height must be align on bayer pattern size, so deleting the rightmost bit will do the job.
34    let roi = Rect::new(roi.p, Dim2::new(roi.width() & !1, roi.height() & !1));
35    let dim = pixels.dim();
36    log::debug!("Superpixel debayer ROI: {:?}", roi);
37
38    let cfa = cfa.shift(roi.p.x, roi.p.y);
39    let pattern = match cfa.name.as_str() {
40      "RGGB" => RgbBayerPattern::RGGB,
41      "BGGR" => RgbBayerPattern::BGGR,
42      "GBRG" => RgbBayerPattern::GBRG,
43      "GRBG" => RgbBayerPattern::GRBG,
44      _ => unreachable!(), // Guarded by is_rgb()
45    };
46
47    // Truncate ROI outer lines
48    let window = &pixels[roi.y() * dim.w..roi.y() * dim.w + roi.height() * dim.w];
49
50    let out = window
51      .par_chunks_exact(dim.w * 2)
52      .map(|s| {
53        let (r1, r2) = s.split_at(dim.w);
54        // Truncate ROI outer columns
55        let (r1, r2) = (&r1[roi.x()..roi.x() + roi.width()], &r2[roi.x()..roi.x() + roi.width()]);
56        r1.chunks_exact(2)
57          .zip(r2.chunks_exact(2))
58          .map(|(a, b)| {
59            let p = [a[0], a[1], b[0], b[1]];
60            match pattern {
61              RgbBayerPattern::RGGB => [p[0], (p[1] + p[2]) / 2.0, p[3]],
62              RgbBayerPattern::BGGR => [p[3], (p[1] + p[2]) / 2.0, p[0]],
63              RgbBayerPattern::GBRG => [p[2], (p[0] + p[3]) / 2.0, p[1]],
64              RgbBayerPattern::GRBG => [p[1], (p[0] + p[3]) / 2.0, p[2]],
65            }
66          })
67          .collect::<Vec<_>>()
68      })
69      .flatten()
70      .collect();
71    Color2D::new_with(out, roi.d.w >> 1, roi.d.h >> 1)
72  }
73}

84impl Demosaic<f32, 4> for Superpixel4Channel {
85  /// Debayer image by using superpixel method.
86  /// Each output pixel RGB tuple is generated by 4 pixels from input.
87  /// The result image is 1/4 of size.
88  fn demosaic(&self, pixels: &PixF32, cfa: &CFA, colors: &PlaneColor, roi: Rect) -> Color2D<f32, 4> {
89    if colors.plane_count() != 4 {
90      panic!("Demosaic for 4 channels needs 4 color planes, but {} given", colors.plane_count());
91    }
92    // ROI width / height must be align on bayer pattern size, so deleting the rightmost bit will do the job.
93    let roi = Rect::new(roi.p, Dim2::new(roi.width() & !1, roi.height() & !1));
94
95    log::debug!("Superpixel debayer ROI: {:?}", roi);
96
97    let cfa = cfa.shift(roi.p.x, roi.p.y);
98    let dim = pixels.dim();
99    // Index into colormap is plane number, value is the index into 2x2 Bayer superpixel.
100    let colormap: [usize; 4] = colors.plane_colors::<4>().map(|c| PlaneColor::cfa_index(&cfa, c));
101
102    // Truncate ROI outer lines
103    let window = &pixels[roi.y() * dim.w..roi.y() * dim.w + roi.height() * dim.w];
104
105    let out = window
106      .par_chunks_exact(dim.w * 2)
107      .map(|s| {
108        let (r1, r2) = s.split_at(dim.w);
109        // Truncate ROI outer columns
110        let (r1, r2) = (&r1[roi.x()..roi.x() + roi.width()], &r2[roi.x()..roi.x() + roi.width()]);
111        r1.chunks_exact(2)
112          .zip(r2.chunks_exact(2))
113          .map(|(a, b)| {
114            let superpixel = [a[0], a[1], b[0], b[1]];
115            // Map superpixel into correct ordering (CFAPlaneColor)
116            [
117              superpixel[colormap[0]],
118              superpixel[colormap[1]],
119              superpixel[colormap[2]],
120              superpixel[colormap[3]],
121            ]
122          })
123          .collect::<Vec<_>>()
124      })
125      .flatten()
126      .collect();
127
128    Color2D::new_with(out, roi.d.w >> 1, roi.d.h >> 1)
129  }
130}

142impl Demosaic<f32, 3> for SuperpixelQuarterRes3Channel {
143  /// Debayer image by using a 4x4 superpixel method.
144  /// Each output pixel RGB tuple is generated by 16 pixels from input.
145  /// The result image is 1/16 of size (1/4 width, 1/4 height).
146  fn demosaic(&self, pixels: &PixF32, cfa: &CFA, colors: &PlaneColor, roi: Rect) -> Color2D<f32, 3> {
147    if !cfa.is_rgb() {
148      panic!("Demosaic for 3 channels requires RGB CFA pattern, but CFA {} given", cfa);
149    }
150    if colors.plane_count() != 3 {
151      panic!("Demosaic for 3 channels needs 3 color planes, but {} given", colors.plane_count());
152    }
153
154    // ROI width / height must be a multiple of 4.
155    let roi = Rect::new(roi.p, Dim2::new(roi.width() & !3, roi.height() & !3));
156    let dim = pixels.dim();
157    log::debug!("Superpixel Quarter-Res debayer ROI: {:?}", roi);
158
159    let cfa = cfa.shift(roi.p.x, roi.p.y);
160
161    // Get a slice of the image corresponding to the ROI.
162    let window = &pixels[roi.y() * dim.w..];
163
164    let out_data: Vec<[f32; 3]> = window
165      .par_chunks_exact(dim.w * 4) // Process 4 rows at a time
166      .take(roi.height() / 4) // Process roi.height() rows in total
167      .flat_map(|four_rows_slice| {
168        // four_rows_slice has length dim.w * 4
169        // It contains 4 full rows of the original image.
170        // We need to get the ROI part of these rows.
171        let row0 = &four_rows_slice[roi.x()..roi.x() + roi.width()];
172        let row1 = &four_rows_slice[dim.w + roi.x()..dim.w + roi.x() + roi.width()];
173        let row2 = &four_rows_slice[dim.w * 2 + roi.x()..dim.w * 2 + roi.x() + roi.width()];
174        let row3 = &four_rows_slice[dim.w * 3 + roi.x()..dim.w * 3 + roi.x() + roi.width()];
175
176        // Now we have the 4 rows for the ROI. We need to process them in 4-pixel chunks.
177        row0
178          .chunks_exact(4)
179          .zip(row1.chunks_exact(4))
180          .zip(row2.chunks_exact(4))
181          .zip(row3.chunks_exact(4))
182          .map(|(((p0, p1), p2), p3)| {
183            // p0, p1, p2, p3 are slices of 4 pixels each.
184            // They form a 4x4 block.
185            let mut sums = [0.0f32; 3];
186            let mut counts = [0; 3];
187
188            let block = [p0, p1, p2, p3];
189
190            for y_offset in 0..4 {
191              for x_offset in 0..4 {
192                // Use relative offset for cfa pattern lookup
193                let color_index = cfa.color_at(y_offset, x_offset);
194                let pixel_value = block[y_offset][x_offset];
195                sums[color_index] += pixel_value;
196                counts[color_index] += 1;
197              }
198            }
199
200            let r = if counts[0] > 0 { sums[0] / counts[0] as f32 } else { 0.0 };
201            let g = if counts[1] > 0 { sums[1] / counts[1] as f32 } else { 0.0 };
202            let b = if counts[2] > 0 { sums[2] / counts[2] as f32 } else { 0.0 };
203
204            [r, g, b]
205          })
206          .collect::<Vec<_>>()
207      })
208      .collect();
209
210    Color2D::new_with(out_data, roi.width() >> 2, roi.height() >> 2)
211  }
