21/// Extend a single pixel component from bayer pattern to RGB
22///
23/// The other channels (missing colors) are set to 0.0.
24#[multiversion(targets("x86_64+avx+avx2", "x86+sse", "aarch64+neon"))]
25fn expand_bayer_rgb(raw: &[f32], dim: Dim2, cfa: &CFA, roi: Rect) -> RgbF32 {
26  // The ROI changes the pattern if not perfectly aligned on the origin pattern
27  let cfa_roi = cfa.shift(roi.x(), roi.y());
28  let mut out = RgbF32::new(roi.width(), roi.height());
29  out.pixels_mut().par_chunks_exact_mut(roi.width()).enumerate().for_each(|(row_out, buf)| {
30    assert_eq!(roi.width() % cfa.width, 0); // Area must be bound to CFA bounds
31    let row_in = roi.p.y + row_out;
32    let start_in = row_in * dim.w + roi.p.x;
33    let line = &raw[start_in..start_in + roi.width()];
34    for (col, (p_out, p_in)) in buf.iter_mut().zip(line.iter()).enumerate() {
35      p_out[cfa_roi.color_at(row_out, col)] = *p_in;
36    }
37  });
38  out
39}
