33impl TryFrom<u16> for Illuminant {
34  type Error = String;
35
36  fn try_from(v: u16) -> Result<Self, Self::Error> {
37    Ok(match v {
38      0 => Self::Unknown,
39      1 => Self::Daylight,
40      2 => Self::Fluorescent,
41      3 => Self::Tungsten,
42      4 => Self::Flash,
43      9 => Self::FineWeather,
44      10 => Self::CloudyWeather,
45      11 => Self::Shade,
46      12 => Self::DaylightFluorescent,
47      13 => Self::DaylightWhiteFluorescent,
48      14 => Self::CoolWhiteFluorescent,
49      15 => Self::WhiteFluorescent,
50      17 => Self::A,
51      18 => Self::B,
52      19 => Self::C,
53      20 => Self::D55,
54      21 => Self::D65,
55      22 => Self::D75,
56      23 => Self::D50,
57      24 => Self::IsoStudioTungsten,
58      _ => {
59        return Err(format!("Unknown illuminant value: {}", v));
60      }
61    })
62  }
63}

65impl From<Illuminant> for u16 {

67    value as u16
68  }

162#[allow(non_snake_case)]
163pub fn xyY_to_XYZ(x: f32, y: f32, Y: f32) -> [f32; 3] {
164  if y.is_normal() && y.is_sign_positive() {
165    [x * Y / y, Y, (1.0 - x - y) * Y / y]
166  } else {
167    panic!("xy_to_XYZ(): 'y' argument must be greater than zero");
168  }
169}
170
171#[allow(non_snake_case)]
172pub fn xy_to_XYZ(x: f32, y: f32) -> [f32; 3] {
173  const Y: f32 = 1.0;
174  xyY_to_XYZ(x, y, Y)
175}

177/// Convert a given xy whitepoint to white balance coefficents,
178/// adapted to
179pub fn xy_whitepoint_to_wb_coeff(x: f32, y: f32, colormatrix: &[[f32; 3]; 3]) -> [f32; 3] {
180  let mut result = [0.0, 0.0, 0.0];
181  if y > 0.0 {
182    let as_shot_white = xy_to_XYZ(x, y);
183    for i in 0..3 {
184      let c = colormatrix[i][0] * as_shot_white[0] + colormatrix[i][1] * as_shot_white[1] + colormatrix[i][2] * as_shot_white[2];
185      if c > 0.0 {
186        result[i] = 1.0 / c;
187      }
188    }
189  }
190  result
