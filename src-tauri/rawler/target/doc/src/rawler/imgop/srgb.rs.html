15/// sRGB gamma
16const SRGB_GAMMA: f32 = 2.4;
17
18/// sRGB crossover point between linear and exponential part
19const SRGB_CROSSOVER_POINT: f32 = 0.00304;
20
21/// sRGB gain for linear part, specified for gamma 2.4
22const LINEAR_GAIN: f32 = 12.92;
23
24/// sRGB gain for exponential part
25const SRGB_GAIN: f32 = 1.055;
26
27/// sRGB offset for exponential part
28const SRGB_OFFSET: f32 = SRGB_GAIN - 1.0; // 0.055
29
30/// Apply sRGB gamma
31pub fn srgb_apply_gamma(v: f32) -> f32 {
32  if v <= SRGB_CROSSOVER_POINT {
33    v * LINEAR_GAIN
34  } else {
35    v.powf(1.0 / SRGB_GAMMA) * SRGB_GAIN - SRGB_OFFSET
36  }
37}
38
39/// Apply sRGB gamma
40pub fn srgb_apply_gamma_n<const N: usize>(v: [f32; N]) -> [f32; N] {
41  v.map(srgb_apply_gamma)
42}
43
44/// Invert sRGB gamma correction
45pub fn srgb_invert_gamma(v: f32) -> f32 {
46  if v <= SRGB_CROSSOVER_POINT * LINEAR_GAIN {
47    v / LINEAR_GAIN
48  } else {
49    ((v + SRGB_OFFSET) / SRGB_GAIN).powf(SRGB_GAMMA)
50  }
51}
52
53/// Apply gamma correction to whole buffer
54pub fn srgb_apply_gamma_inplace(pixels: &mut [f32]) {
55  pixels.par_iter_mut().for_each(|p| *p = srgb_apply_gamma(*p));
56}
57
58/// Invert gamma correction to whole buffer
59pub fn srgb_invert_gamma_inplace(pixels: &mut [f32]) {
60  pixels.par_iter_mut().for_each(|p| *p = srgb_invert_gamma(*p));
61}
