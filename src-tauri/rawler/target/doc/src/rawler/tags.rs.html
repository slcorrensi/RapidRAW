7macro_rules! tiff_tag_enum {
8  ($e:ty) => {
9    impl From<$e> for u16 {
10      fn from(val: $e) -> u16 {
11        val as u16
12      }
13    }
14    impl TryFrom<u16> for $e {
15      type Error = String;
16
17      fn try_from(value: u16) -> std::result::Result<Self, Self::Error> {
18        Self::n(value).ok_or(format!("Unable to convert tag: {}, not defined in enum", value))
19      }
20    }
21    impl crate::tags::TiffTag for $e {}
22  };
23}
24pub(crate) use tiff_tag_enum;
