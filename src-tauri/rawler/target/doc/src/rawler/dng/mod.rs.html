18/// Convert internal crop rectangle to DNG active area
19///
20/// DNG ActiveArea  is:
21///  Top, Left, Bottom, Right
22pub fn rect_to_dng_area(area: &Rect) -> [u16; 4] {
23  [
24    area.p.y as u16,
25    area.p.x as u16,
26    area.p.y as u16 + area.d.h as u16,
27    area.p.x as u16 + area.d.w as u16,
28  ]
29  /*
30  [
31    image.crops[0] as u16, // top
32    image.crops[3] as u16, // left
33    //(image.height-image.crops[0]-image.crops[2]) as u16, // bottom
34    //(image.width-image.crops[1]-image.crops[3]) as u16, // Right
35    (image.height - (image.crops[2])) as u16, // bottom coord
36    (image.width - (image.crops[1])) as u16,  // Right coord
37  ]
38  */
39}
