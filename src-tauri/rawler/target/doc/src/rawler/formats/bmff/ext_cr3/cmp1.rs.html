9#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
10pub struct Cmp1Box {
11  pub header: BoxHeader,
12  pub unknown1: i16,
13  pub header_size: u16,
14  pub version: u16,
15  pub version_sub: u16,
16  pub f_width: u32,
17  pub f_height: u32,
18  pub tile_width: u32,
19  pub tile_height: u32,
20  pub n_bits: u8,
21  pub n_planes: u8,
22  pub cfa_layout: u8,
23  pub enc_type: u8,
24  pub image_levels: u8,
25  pub has_tile_cols: u8,
26  pub has_tile_rows: u8,
27  pub mdat_hdr_size: u32,
28  pub ext_header: u32,
29  pub median_bits: u8,
30  // TODO: missing fields
31}
32
33impl Cmp1Box {
34  pub const TYP: FourCC = FourCC::with(['C', 'M', 'P', '1']);
35}
36
37impl<R: Read + Seek> ReadBox<&mut R> for Cmp1Box {
38  fn read_box(reader: &mut R, header: BoxHeader) -> Result<Self> {
39    let unknown1 = reader.read_i16::<BigEndian>()?;
40    let header_size = reader.read_u16::<BigEndian>()?;
41    let version = reader.read_u16::<BigEndian>()?;
42    let version_sub = reader.read_u16::<BigEndian>()?;
43    let f_width = reader.read_u32::<BigEndian>()?;
44    let f_height = reader.read_u32::<BigEndian>()?;
45    let tile_width = reader.read_u32::<BigEndian>()?;
46    let tile_height = reader.read_u32::<BigEndian>()?;
47    let n_bits = reader.read_u8()?;
48
49    let (n_planes, cfa_layout) = {
50      let tmp = reader.read_u8()?;
51      (tmp >> 4, tmp & 0xF)
52    };
53
54    let (enc_type, image_levels) = {
55      let tmp = reader.read_u8()?;
56      (tmp >> 4, tmp & 0xF)
57    };
58
59    let (has_tile_cols, has_tile_rows) = {
60      let tmp = reader.read_u8()?;
61      (tmp >> 7, tmp & 1)
62    };
63
64    let mdat_hdr_size = reader.read_u32::<BigEndian>()?;
65    let ext_header = reader.read_u32::<BigEndian>()?; // 32
66
67    // Median bit precision for enc_type 3 is not always same as
68    // image bit precision. For CRM movie files, there is an extended header.
69    let mut median_bits = n_bits;
70
71    // CRM Movie file
72    if (ext_header & 0x80000000 != 0) && n_planes == 4 && header.size >= 56 {
73      let _unknow = reader.read_u32::<BigEndian>()?; // 36
74      let _unknow = reader.read_u32::<BigEndian>()?; // 40
75      let _unknow = reader.read_u32::<BigEndian>()?; // 44
76      let _unknow = reader.read_u32::<BigEndian>()?; // 48
77      let _unknow = reader.read_u32::<BigEndian>()?; // 52
78      let use_median_bits = reader.read_u32::<BigEndian>()? & 0x40000000 != 0; // 56
79      if use_median_bits && header.size > 84 {
80        let _unknow = reader.read_u32::<BigEndian>()?; // 60
81        let _unknow = reader.read_u32::<BigEndian>()?; // 64
82        let _unknow = reader.read_u32::<BigEndian>()?; // 68
83        let _unknow = reader.read_u32::<BigEndian>()?; // 72
84        let _unknow = reader.read_u32::<BigEndian>()?; // 76
85        let _unknow = reader.read_u32::<BigEndian>()?; // 80
86        median_bits = reader.read_u8()?; // 84
87      }
88    }
89
90    reader.seek(SeekFrom::Start(header.end_offset()))?;
91
92    Ok(Self {
93      header,
94      unknown1,
95      header_size,
96      version,
97      version_sub,
98      f_width,
99      f_height,
100      tile_width,
101      tile_height,
102      n_bits,
103      n_planes,
104      cfa_layout,
105      enc_type,
106      image_levels,
107      has_tile_cols,
108      has_tile_rows,
109      mdat_hdr_size,
110      ext_header,
111      median_bits,
112    })
113  }
114}
