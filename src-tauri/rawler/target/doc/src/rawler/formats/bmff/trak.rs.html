5use super::{BmffError, BoxHeader, FourCC, ReadBox, Result, mdia::MdiaBox, tkhd::TkhdBox, vendor::VendorBox};
6use log::debug;
7use serde::{Deserialize, Serialize};
8use std::io::{Read, Seek, SeekFrom};
9
10#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
11pub struct TrakBox {
12  pub header: BoxHeader,
13  pub tkhd: TkhdBox,
14
15  //#[serde(skip_serializing_if = "Option::is_none")]
16  //pub edts: Option<EdtsBox>,
17  pub mdia: MdiaBox,
18  pub vendor: Vec<VendorBox>,
19}
20
21impl TrakBox {
22  pub const TYP: FourCC = FourCC::with(['t', 'r', 'a', 'k']);
23}
24
25impl<R: Read + Seek> ReadBox<&mut R> for TrakBox {
26  fn read_box(mut reader: &mut R, header: BoxHeader) -> Result<Self> {
27    let mut tkhd = None;
28    let mut mdia = None;
29
30    let mut vendors = Vec::new();
31
32    let mut current = reader.stream_position()?;
33
34    while current < header.end_offset() {
35      // get box?
36
37      let header = BoxHeader::parse(&mut reader)?;
38
39      //let ftyp = Some(FtypBox::read_box(&mut file, header)?);
40
41      match header.typ {
42        TkhdBox::TYP => {
43          tkhd = Some(TkhdBox::read_box(&mut reader, header)?);
44        }
45        MdiaBox::TYP => {
46          mdia = Some(MdiaBox::read_box(&mut reader, header)?);
47        }
48
49        _ => {
50          debug!("Vendor box found in trak: {:?}", header.typ);
51          let vendor = VendorBox::read_box(&mut reader, header)?;
52          vendors.push(vendor);
53        }
54      }
55
56      current = reader.stream_position()?;
57    }
58
59    reader.seek(SeekFrom::Start(header.end_offset()))?;
60
61    Ok(Self {
62      header,
63      tkhd: tkhd.ok_or_else(|| BmffError::Parse("tkhd box not found, corrupt file?".into()))?,
64      mdia: mdia.ok_or_else(|| BmffError::Parse("mdia box not found, corrupt file?".into()))?,
65      vendor: vendors,
66    })
67  }
68}
