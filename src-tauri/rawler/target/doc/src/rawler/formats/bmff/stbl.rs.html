11pub struct StblBox {
12  pub header: BoxHeader,
13  pub stsd: StsdBox,
14  pub stts: SttsBox,

16  pub stsc: StscBox,

18  pub co64: Option<Co64Box>,
19  pub vendor: Vec<VendorBox>,
20}

22impl StblBox {
23  pub const TYP: FourCC = FourCC::with(['s', 't', 'b', 'l']);
24
25  pub fn get_sample_offset(&self, sample: u32) -> Option<(usize, usize)> {
26    if let Some(co64) = self.co64.as_ref() {
27      assert!(sample > 0, "sample number must be greater than 0");
28      let desc = self.stsc.get_entry_for_sample(sample);
29      // Chunk number for our sample
30      let chunk = desc.first_chunk + (((sample - 1) - (desc.first_sample - 1)) / desc.samples_per_chunk);
31      // Index of sample inside chunk
32      let chunk_sample_idx = (sample - desc.first_sample) % desc.samples_per_chunk;
33      // Offset of chunk
34      let chunk_offset = co64.entries[chunk as usize - 1];
35
36      // Offset of sample in chunk
37      let mut sample_offset = 0;
38      for i in 0..chunk_sample_idx {
39        sample_offset += self.stsz.sample_size(sample - chunk_sample_idx + i);
40      }
41      let sample_size = self.stsz.sample_size(sample);
42
43      Some((chunk_offset as usize + sample_offset as usize, sample_size as usize))
44    } else {
45      None
46    }
47  }
48}

50impl<R: Read + Seek> ReadBox<&mut R> for StblBox {
51  fn read_box(mut reader: &mut R, header: BoxHeader) -> Result<Self> {
52    let mut stsd = None;
53    let mut stts = None;
54    // let mut ctts = None;
55    //let mut stss = None;
56    let mut stsc = None;
57    let mut stsz = None;
58    //let mut stco = None;
59    let mut co64 = None;
60
61    let mut vendors = Vec::new();
62
63    let mut current = reader.stream_position()?;
64
65    while current < header.end_offset() {
66      // get box?
67
68      let header = BoxHeader::parse(&mut reader)?;
69
70      //let ftyp = Some(FtypBox::read_box(&mut file, header)?);
71
72      match header.typ {
73        StsdBox::TYP => {
74          stsd = Some(StsdBox::read_box(&mut reader, header)?);
75        }
76        SttsBox::TYP => {
77          stts = Some(SttsBox::read_box(&mut reader, header)?);
78        }
79        StscBox::TYP => {
80          stsc = Some(StscBox::read_box(&mut reader, header)?);
81        }
82        StszBox::TYP => {
83          stsz = Some(StszBox::read_box(&mut reader, header)?);
84        }
85        Co64Box::TYP => {
86          co64 = Some(Co64Box::read_box(&mut reader, header)?);
87        }
88
89        _ => {
90          debug!("Vendor box found in stbl: {:?}", header.typ);
91          let vendor = VendorBox::read_box(&mut reader, header)?;
92          vendors.push(vendor);
93        }
94      }
95
96      current = reader.stream_position()?;
97    }
98
99    reader.seek(SeekFrom::Start(header.end_offset()))?;
100
101    Ok(Self {
102      header,
103      stsd: stsd.ok_or_else(|| BmffError::Parse("stsd box not found, corrupt file?".into()))?,
104      stts: stts.ok_or_else(|| BmffError::Parse("stts box not found, corrupt file?".into()))?,
105      stsc: stsc.ok_or_else(|| BmffError::Parse("stsc box not found, corrupt file?".into()))?,
106      stsz: stsz.ok_or_else(|| BmffError::Parse("stsz box not found, corrupt file?".into()))?,
107      co64,
108      vendor: vendors,
109    })
110  }
111}
