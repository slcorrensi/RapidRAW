44impl<R: Read + Seek> ReadSegment<&mut R> for App0 {
45  fn read_segment(reader: &mut R, _symbol: u16) -> Result<Self> {
46    //let pos = reader.stream_position()?;
47    let len: u64 = reader.read_u16::<BigEndian>()? as u64;
48
49    const APP1_JFIF_MARKER: [u8; 5] = [b'J', b'F', b'I', b'F', b'\0'];
50
51    if len as usize >= APP1_JFIF_MARKER.len() {
52      let mut jfif_str = [0; 5];
53      reader.read_exact(&mut jfif_str)?;
54      if jfif_str == APP1_JFIF_MARKER {
55        let version = reader.read_u16::<BigEndian>()?;
56        let density_units = reader.read_u8()?;
57        let xdensity = reader.read_u16::<BigEndian>()?;
58        let ydensity = reader.read_u16::<BigEndian>()?;
59        let xthumbnail = reader.read_u8()?;
60        let ythumbnail = reader.read_u8()?;
61        let thumbnail = if xthumbnail * ythumbnail > 0 {
62          let mut data = vec![0; (3 * xthumbnail * ythumbnail) as usize];
63          reader.read_exact(&mut data)?;
64          Some(data)
65        } else {
66          None
67        };
68        Ok(Self {
69          len,
70          version,
71          density_units,
72          xdensity,
73          ydensity,
74          xthumbnail,
75          ythumbnail,
76          thumbnail,
77        })
78      } else {
79        Err(JfifError::FormatMismatch("Failed to find JFIF marker in APP0 segment".into()))
80      }
81    } else {
82      Err(JfifError::FormatMismatch("APP0 segment is too short".into()))
83    }
84  }
85}

100impl<R: Read + Seek> ReadSegment<&mut R> for SOS {
101  fn read_segment(reader: &mut R, _symbol: u16) -> Result<Self> {
102    let pos = reader.stream_position()?;
103    let mut prev = 0;
104
105    loop {
106      let v = reader.read_u8()?;
107      if prev == 0xFF && v == 0xFF {
108        prev = 0;
109        continue;
110      }
111      if prev == 0xFF && (v & !0x7) == 0xD0 {
112        prev = 0;
113        continue;
114      }
115      if prev == 0xFF && v == 0 {
116        prev = 0;
117        continue;
118      }
119      if prev == 0xFF {
120        reader.seek(SeekFrom::Current(-2))?;
121        break;
122      }
123      prev = v;
124    }
125
126    let len = reader.stream_position()? - pos;
127
128    Ok(Self { len })
129  }
130}

135  Xpacket(Vec<u8>),

139impl<R: Read + Seek> ReadSegment<&mut R> for App1 {
140  fn read_segment(reader: &mut R, _symbol: u16) -> Result<Self> {
141    let pos = reader.stream_position()?;
142    let len: u64 = reader.read_u16::<BigEndian>()? as u64;
143
144    const APP1_EXIF_MARKER: [u8; 6] = [b'E', b'x', b'i', b'f', b'\0', b'\0'];
145
146    if len as usize >= APP1_EXIF_MARKER.len() {
147      let mut exif_str = [0; 6];
148      reader.read_exact(&mut exif_str)?;
149      if exif_str == APP1_EXIF_MARKER {
150        let ifd = IFD::new_root(reader, pos as u32 + 2 + 6);
151        reader.seek(SeekFrom::Start(pos + len))?;
152
153        /*
154        for i in ifd.dump::<TiffCommonTag>(10) {
155          println!("{}", i);
156        }
157
158        for i in ifd.get_sub_ifds(TiffCommonTag::ExifIFDPointer).unwrap()[0].dump::<ExifTag>(10) {
159          println!("{}", i);
160        }
161         */
162
163        if let Ok(ifd) = ifd {
164          return Ok(Self {
165            len,
166            payload: Payload::Exif(ifd),
167          });
168        }
169
170        return Err(JfifError::General("Failed to read exif".into()));
171      } else {
172        reader.seek(SeekFrom::Current(-(exif_str.len() as i64)))?;
173      }
174    }
175
176    const APP1_XMP_MARKER: &str = concat!("http://ns.adobe.com/xap/1.0/", '\0');
177    if len as usize >= APP1_XMP_MARKER.len() {
178      let mut xmp_str = [0; APP1_XMP_MARKER.len()];
179      reader.read_exact(&mut xmp_str)?;
180      if xmp_str == APP1_XMP_MARKER.as_bytes() {
181        log::debug!("Found APP1 XMP marker");
182        let mut xpacket = vec![0; len as usize - xmp_str.len() - 2];
183        reader.read_exact(&mut xpacket)?;
184        reader.seek(SeekFrom::Start(pos + len))?;
185        return Ok(Self {
186          len,
187          payload: Payload::Xpacket(xpacket),
188        });
189      } else {
190        reader.seek(SeekFrom::Current(-(xmp_str.len() as i64)))?;
191      }
192    }
193    reader.seek(SeekFrom::Start(pos + len))?;
194    Ok(Self {
195      len,
196      payload: Payload::Unknown,
197    })
198  }
199}

201#[derive(Debug, Clone)]
202pub enum Segment {
203  SOI { offset: u64 },
204  APP0 { offset: u64, app0: App0 },
205  APP1 { offset: u64, app1: App1 },
206  SOS { offset: u64, sos: SOS },
207  EOI,
208  Unknown { offset: u64, marker: u16 },
209}

211#[derive(Debug, Clone)]
212pub struct Jfif {
213  pub segments: Vec<Segment>,
214}
215
216impl Jfif {
217  pub fn parse<R: Read + Seek>(mut reader: R) -> Result<Self> {
218    let mut segments = Vec::new();
219    let mut pos = reader.stream_position()?;
220    let mut sym = Some(reader.read_u16::<BigEndian>()?);
221
222    while let Some(symbol) = sym {
223      log::debug!("Found symbol: {:X}", symbol);
224      let segment = match symbol {
225        0xFFD8 => Segment::SOI { offset: pos },
226        0xFFD9 => Segment::EOI,
227        0xFFE0 => Segment::APP0 {
228          offset: pos,
229          app0: App0::read_segment(&mut reader, symbol)?,
230        },
231        0xFFE1 => match App1::read_segment(&mut reader, symbol) {
232          Ok(app1) => Segment::APP1 { offset: pos, app1 },
233          Err(err) => {
234            log::warn!("Failed to read APP1 (EXIF) segement, maybe corrupt: {:?}", err);
235            continue;
236          }
237        },
238        0xFFDA => Segment::SOS {
239          offset: pos,
240          sos: SOS::read_segment(&mut reader, symbol)?,
241        },
242
243        _ => {
244          log::debug!("Unhandled JFIF segment marker: {:X}", symbol);
245          let len: u64 = reader.read_u16::<BigEndian>()? as u64;
246          reader.seek(SeekFrom::Current(len as i64 - 2))?;
247
248          Segment::Unknown { offset: pos, marker: symbol }
249        }
250      };
251      if segments.is_empty() && sym != Some(0xFFD8) {
252        return Err(JfifError::General("first marker must be SOI".into()));
253      }
254      segments.push(segment);
255      if symbol == 0xFFD9 {
256        break; // EOI reached
257      }
258      pos = reader.stream_position()?;
259      sym = reader.read_u16::<BigEndian>().ok();
260    }
261    if !segments.is_empty() {
262      Ok(Self { segments })
263    } else {
264      Err(JfifError::FormatMismatch("JFIF contains no segements".into()))
265    }
266  }
267
268  pub fn new(file: &RawSource) -> Result<Self> {
269    Self::parse(&mut file.reader())
270  }
271
272  pub fn exif_ifd(&self) -> Option<&IFD> {
273    self.segments.iter().find_map(|seg| match seg {
274      Segment::APP1 {
275        app1: App1 {
276          payload: Payload::Exif(ifd), ..
277        },
278        ..
279      } => Some(ifd),
280      _ => None,
281    })
282  }
283
284  pub fn xpacket(&self) -> Option<&Vec<u8>> {
285    self.segments.iter().find_map(|seg| match seg {
286      Segment::APP1 {
287        app1: App1 {
288          payload: Payload::Xpacket(xpacket),
289          ..
290        },
291        ..
292      } => Some(xpacket),
293      _ => None,
294    })
295  }
296}

298pub fn is_jfif(file: &RawSource) -> bool {
299  match file.subview(0, 4) {
300    Ok(buf) => {
301      let result = buf[0..4] == [0xFF, 0xD8, 0xFF, 0xE0];
302      if !result {
303        //panic!("Failed: {:x} {:x} {:x} {:x}", buf[0], buf[1], buf[2], buf[3]);
304      }
305      result
306    }
307    Err(err) => {
308      log::error!("is_jfif(): {:?}", err);
309      false
310    }
311  }
312}
