27impl<W> TiffWriter<W>
28where
29  W: Write + Seek,
30{
31  pub fn new(writer: W) -> Result<Self> {
32    let mut tmp = Self { writer, ifd_location: 0 };
33    tmp.write_header()?;
34    Ok(tmp)
35  }

37  pub fn new_directory(&self) -> DirectoryWriter {
38    DirectoryWriter::new()
39  }
40
41  fn write_header(&mut self) -> Result<()> {
42    #[cfg(target_endian = "little")]
43    let boi: u8 = 0x49;
44    #[cfg(not(target_endian = "little"))]
45    let boi: u8 = 0x4d;
46
47    self.writer.write_all(&[boi, boi])?;
48    self.writer.write_u16::<NativeEndian>(TIFF_MAGIC)?;
49    self.ifd_location = self.writer.stream_position()?;
50    self.writer.write_u32::<NativeEndian>(0_u32)?;
51
52    Ok(())
53  }
54
55  pub fn write_strips_lzw(&mut self, data: &[u16], cpp: usize, dim: Dim2, mut strip_lines: usize) -> Result<(u32, Vec<(u32, u32)>)> {
56    if strip_lines == 0 {
57      if dim.h > 256 {
58        strip_lines = 256;
59      } else {
60        strip_lines = dim.h; // Use single strip
61      }
62    }
63    let mut tag_data = Vec::with_capacity(dim.h / strip_lines + 1);
64    let strips = data
65      .par_chunks(cpp * dim.w * strip_lines)
66      .map(|strip| {
67        let mut encoder = weezl::encode::Encoder::with_tiff_size_switch(weezl::BitOrder::Msb, 8);
68        encoder.encode(strip.as_bytes())
69      })
70      .collect::<std::result::Result<Vec<Vec<u8>>, LzwError>>()
71      .map_err(|err| TiffError::General(err.to_string()))?;
72
73    for strip in strips {
74      let offset = self.write_data(&strip)?;
75      let bytes = strip.len() as u32;
76      tag_data.push((offset, bytes));
77    }
78    Ok((strip_lines as u32, tag_data))
79  }
80
81  pub fn write_data(&mut self, data: &[u8]) -> Result<u32>
82  where
83    W: Seek + Write,
84  {
85    self.pad_word_boundary()?;
86    let offset = self.position()?;
87    self.writer.write_all(data)?;
88    Ok(offset)
89  }
90
91  pub fn write_data_u16_le(&mut self, data: &[u16]) -> Result<u32>
92  where
93    W: Seek + Write,
94  {
95    self.pad_word_boundary()?;
96    let offset = self.position()?;
97    for v in data {
98      self.writer.write_u16::<LittleEndian>(*v)?;
99    }
100    Ok(offset)
101  }
102
103  pub fn write_data_f32_le(&mut self, data: &[f32]) -> Result<u32>
104  where
105    W: Seek + Write,
106  {
107    self.pad_word_boundary()?;
108    let offset = self.position()?;
109    for v in data {
110      self.writer.write_f32::<LittleEndian>(*v)?;
111    }
112    Ok(offset)
113  }
114
115  pub(crate) fn pad_word_boundary(&mut self) -> Result<()> {
116    if self.position()? % 4 != 0 {
117      let padding = [0, 0, 0];
118      let padd_len = 4 - (self.position()? % 4);
119      self.writer.write_all(&padding[..padd_len as usize])?;
120    }
121    Ok(())
122  }
123
124  pub fn build(mut self, root_ifd: DirectoryWriter) -> Result<()> {
125    let ifd0_offset = root_ifd.build(&mut self)?;
126    self.writer.seek(SeekFrom::Start(self.ifd_location))?;
127    self.writer.write_u32::<NativeEndian>(ifd0_offset)?;
128    Ok(())
129  }
130}

142pub struct DirectoryWriter {
143  // We use BTreeMap to make sure tags are written in correct order
144  entries: BTreeMap<u16, Entry>,
145  next_ifd: u32,

148impl DirectoryWriter {
149  pub fn remove_tag<T: TiffTag>(&mut self, tag: T) {
150    let tag: u16 = tag.into();
151    self.entries.remove(&tag);
152  }

154  pub fn add_tag<T: TiffTag, V: Into<Value>>(&mut self, tag: T, value: V) {
155    let tag: u16 = tag.into();
156    self.entries.insert(
157      tag,
158      Entry {
159        tag,
160        value: value.into(),
161        embedded: None,
162      },
163    );
164  }
165
166  pub fn add_untyped_tag<V: Into<Value>>(&mut self, tag: u16, value: V) {
167    self.entries.insert(
168      tag,
169      Entry {
170        tag,
171        value: value.into(),
172        embedded: None,
173      },
174    );
175  }
176
177  pub fn contains<T: TiffTag>(&self, tag: T) -> bool {
178    self.entries.contains_key(&tag.into())
179  }
180
181  pub fn copy<'a>(&mut self, iter: impl Iterator<Item = (&'a u16, &'a Value)>) {
182    for entry in iter {
183      if !self.entries.contains_key(entry.0) {
184        self.add_untyped_tag(*entry.0, entry.1.clone());
185      }
186    }
187  }
188
189  pub fn copy_with_override<'a>(&mut self, iter: impl Iterator<Item = (&'a u16, &'a Value)>) {
190    for entry in iter {
191      self.add_untyped_tag(*entry.0, entry.1.clone());
192    }
193  }

195  pub fn add_tag_undefined<T: TiffTag>(&mut self, tag: T, data: Vec<u8>) {
196    let tag: u16 = tag.into();
197    //let data = data.as_ref();
198    //let offset = self.write_data(data)?;
199    self.entries.insert(
200      tag,
201      Entry {
202        tag,
203        value: Value::Undefined(data),
204        embedded: None,
205      },
206    );
207  }
208
209  pub fn add_value<T: TiffTag>(&mut self, tag: T, value: Value) {
210    let tag: u16 = tag.into();
211    self.entries.insert(tag, Entry { tag, value, embedded: None });
212  }
213
214  pub fn entry_count(&self) -> u16 {
215    self.entries.len() as u16
216  }
217
218  pub fn new() -> Self {
219    Self {
220      entries: BTreeMap::new(),
221      next_ifd: 0,
222    }
223  }
224
225  pub fn is_empty(&self) -> bool {
226    self.entries.is_empty()
227  }
228
229  pub fn build<W>(mut self, tiff: &mut TiffWriter<W>) -> Result<u32>
230  where
231    W: Seek + Write,
232  {
233    if self.entries.is_empty() {
234      return Err(TiffError::General("IFD is empty, not allowed by TIFF specification".to_string()));
235    }
236    for &mut Entry {
237      ref mut value,
238      ref mut embedded,
239      ref tag,
240    } in self.entries.values_mut()
241    {
242      let data_bytes = 4;
243
244      if value.byte_size() > data_bytes {
245        tiff.pad_word_boundary()?;
246        let offset = tiff.position()?;
247        value.write(&mut tiff.writer)?;
248        embedded.replace(offset as u32);
249      } else {
250        if value.count() == 0 {
251          panic!("TIFF value is empty, tag: {:?}", tag);
252        }
253        embedded.replace(value.as_embedded()?);
254      }
255    }
256
257    tiff.pad_word_boundary()?;
258    let offset = tiff.position()?;
259
260    tiff.writer.write_all(&self.entry_count().to_ne_bytes())?;
261
262    for (tag, entry) in self.entries {
263      tiff.writer.write_u16::<NativeEndian>(tag)?;
264      tiff.writer.write_u16::<NativeEndian>(entry.value_type())?;
265      tiff.writer.write_u32::<NativeEndian>(entry.count())?;
266      tiff.writer.write_u32::<NativeEndian>(entry.embedded.unwrap())?;
267    }
268    tiff.writer.write_u32::<NativeEndian>(self.next_ifd)?; // Next IFD
269
270    Ok(offset)
271  }

391pub(crate) fn transfer_entry<T, V>(ifd: &mut DirectoryWriter, tag: T, entry: &Option<V>) -> Result<()>
392where
393  T: TiffTag,
394  V: Into<Value> + Clone,
395{
396  if let Some(entry) = entry {
397    ifd.add_tag(tag, entry.clone());
398  }
399  Ok(())
400}

403where
404  T: TiffTag,
405{
406  if let Some(entry) = entry {
407    ifd.add_tag_undefined(tag, entry.clone());
408  }
409  Ok(())
