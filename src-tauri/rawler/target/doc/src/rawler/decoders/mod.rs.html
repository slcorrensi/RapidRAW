245  pub fn last_modified(&self) -> Result<Option<SystemTime>> {
246    let mtime = self
247      .exif
248      .modify_date
249      .as_ref()
250      .map(|mtime| NaiveDateTime::parse_from_str(mtime, "%Y:%m:%d %H:%M:%S"))
251      .transpose()
252      .map_err(|err| RawlerError::DecoderFailed(err.to_string()))?;
253    if let Some(mtime) = mtime {
254      // Probe for available timezone information
255      let tz = if let Some(offset) = self.exif.timezone_offset.as_ref().and_then(|x| x.get(1)) {
256        if let Some(tz) = FixedOffset::east_opt(*offset as i32 * 3600) {
257          Some(tz)
258        } else {
259          log::warn!("TZ Offset overflow");
260          None
261        }
262      } else if let Some(offset) = &self.exif.offset_time {
263        match FixedOffset::from_str(offset) {
264          Ok(tz) => Some(tz),
265          Err(err) => {
266            log::warn!("Invalid fixed offset: {}", err);
267            None
268          }
269        }
270      } else {
271        None
272      };
273      // Any timezone? Then correct...
274      if let Some(tz) = tz {
275        let x = tz
276          .from_local_datetime(&mtime)
277          .earliest()
278          .ok_or(RawlerError::DecoderFailed(format!("Failed to convert datetime to local: {:?}", mtime)))?;
279        return Ok(Some(x.into()));
280      } else {
281        match chrono::Local.from_local_datetime(&mtime).earliest() {
282          Some(ts) => return Ok(Some(ts.into())),
283          None => {
284            log::warn!("Failed to convert ts to local time");
285          }
286        }
287      }
288    }
289
290    Ok(None)
291  }

335/// Possible orientations of an image
336///
337/// Values are taken from the IFD tag Orientation (0x0112) in most cases but they can be
338/// obtained from other metadata in the file.
339#[derive(Debug, Copy, Clone, PartialEq, Eq, Serialize, Deserialize, Hash)]
340#[allow(missing_docs)]
341pub enum Orientation {
342  Normal,
343  HorizontalFlip,
344  Rotate180,
345  VerticalFlip,
346  Transpose,
347  Rotate90,
348  Transverse,
349  Rotate270,
350  Unknown,
351}
352
353impl Orientation {
354  /// Convert a u16 from the IFD tag Orientation (0x0112) into its corresponding
355  /// enum value
356  pub fn from_u16(orientation: u16) -> Orientation {
357    match orientation {
358      1 => Orientation::Normal,
359      2 => Orientation::HorizontalFlip,
360      3 => Orientation::Rotate180,
361      4 => Orientation::VerticalFlip,
362      5 => Orientation::Transpose,
363      6 => Orientation::Rotate90,
364      7 => Orientation::Transverse,
365      8 => Orientation::Rotate270,
366      _ => Orientation::Unknown,
367    }
368  }
369
370  /// Extract orienation from a TiffIFD. If the given TiffIFD has an invalid
371  /// value or contains no orientation data `Orientation::Unknown` is returned
372  fn from_tiff(tiff: &IFD) -> Orientation {
373    match tiff.get_entry(TiffCommonTag::Orientation) {
374      Some(entry) => Orientation::from_u16(entry.force_usize(0) as u16),
375      None => Orientation::Unknown,
376    }
377  }
378
379  /// Convert orientation to an image flip operation tuple. The first field is
380  /// if x and y coordinates should be swapped (transposed). The second and
381  /// third field is horizontal and vertical flipping respectively. For a
382  /// correct result, flipping must be done before transposing.
383  pub fn to_flips(&self) -> (bool, bool, bool) {
384    match *self {
385      Orientation::Normal | Orientation::Unknown => (false, false, false),
386      Orientation::VerticalFlip => (false, false, true),
387      Orientation::HorizontalFlip => (false, true, false),
388      Orientation::Rotate180 => (false, true, true),
389      Orientation::Transpose => (true, false, false),
390      Orientation::Rotate90 => (true, false, true),
391      Orientation::Rotate270 => (true, true, false),
392      Orientation::Transverse => (true, true, true),
393    }
394  }
395
396  /// Does the opposite of to_flips()
397  pub fn from_flips(flips: (bool, bool, bool)) -> Self {
398    match flips {
399      (false, false, false) => Orientation::Normal,
400      (false, false, true) => Orientation::VerticalFlip,
401      (false, true, false) => Orientation::HorizontalFlip,
402      (false, true, true) => Orientation::Rotate180,
403      (true, false, false) => Orientation::Transpose,
404      (true, false, true) => Orientation::Rotate90,
405      (true, true, false) => Orientation::Rotate270,
406      (true, true, true) => Orientation::Transverse,
407    }
408  }
409
410  /// Convert orientation to the Tiff Orientation value
411  pub fn to_u16(&self) -> u16 {
412    match *self {
413      Orientation::Unknown => 0,
414      Orientation::Normal => 1,
415      Orientation::HorizontalFlip => 2,
416      Orientation::Rotate180 => 3,
417      Orientation::VerticalFlip => 4,
418      Orientation::Transpose => 5,
419      Orientation::Rotate90 => 6,
420      Orientation::Transverse => 7,
421      Orientation::Rotate270 => 8,
422    }
423  }
424}

473/// The struct that holds all the info about the cameras and is able to decode a file
474#[derive(Debug, Clone, Default)]
475pub struct RawLoader {
476  cameras: HashMap<(String, String, String), Camera>,
477  #[allow(dead_code)] // TODO: remove once naked cams supported again
478  naked: HashMap<usize, Camera>,
479}
480
481impl RawLoader {
482  /// Creates a new raw loader using the camera information included in the library
483  pub fn new() -> RawLoader {
484    let toml = match CAMERAS_TOML.parse::<Value>() {
485      Ok(val) => val,
486      Err(e) => panic!("{}", format!("Error parsing cameras.toml: {:?}", e)),
487    };
488
489    let mut cams = Vec::new();
490    for camera in toml.get("cameras").unwrap().as_array().unwrap() {
491      // Create a list of all the camera modes including the base one
492      let mut cammodes = Vec::new();
493      let ct = camera.as_table().unwrap();
494      cammodes.push(ct);
495      if let Some(val) = ct.get("modes") {
496        for mode in val.as_array().unwrap() {
497          cammodes.push(mode.as_table().unwrap());
498        }
499      }
500
501      // Start with the basic camera
502      let mut cam = Camera::new();
503      cam.update_from_toml(cammodes[0]);
504      // Create a list of alias names including the base one
505      let mut camnames = vec![(cam.model.clone(), cam.clean_model.clone())];
506      if let Some(val) = ct.get("model_aliases") {
507        for alias in val.as_array().unwrap() {
508          camnames.push((alias[0].as_str().unwrap().to_string().clone(), alias[1].as_str().unwrap().to_string().clone()));
509        }
510      }
511
512      // For each combination of alias and mode (including the base ones) create Camera
513      for (model, clean_model) in camnames {
514        for ct in cammodes.clone() {
515          let mut mcam = cam.clone();
516          mcam.update_from_toml(ct);
517          mcam.model = model.clone();
518          mcam.clean_model = clean_model.clone();
519          cams.push(mcam);
520        }
521      }
522    }
523
524    let mut map = HashMap::new();
525    let mut naked = HashMap::new();
526    for cam in cams {
527      map.insert((cam.make.clone(), cam.model.clone(), cam.mode.clone()), cam.clone());
528      if cam.filesize > 0 {
529        naked.insert(cam.filesize, cam);
530      }
531    }
532
533    RawLoader { cameras: map, naked }
534  }

541  /// Returns a decoder for a given buffer
542  pub fn get_decoder<'b>(&'b self, rawfile: &RawSource) -> Result<Box<dyn Decoder + 'b>> {
543    if mrw::is_mrw(rawfile) {
544      let dec = Box::new(mrw::MrwDecoder::new(rawfile, self)?);
545      return Ok(dec as Box<dyn Decoder>);
546    }
547
548    if raf::is_raf(rawfile) {
549      let dec = Box::new(raf::RafDecoder::new(rawfile, self)?);
550      return Ok(dec as Box<dyn Decoder>);
551    }
552
553    if ari::is_ari(rawfile) {
554      let dec = Box::new(ari::AriDecoder::new(rawfile, self)?);
555      return Ok(dec as Box<dyn Decoder>);
556    }
557
558    if qtk::is_qtk(rawfile) {
559      let dec = Box::new(qtk::QtkDecoder::new(rawfile, self)?);
560      return Ok(dec as Box<dyn Decoder>);
561    }
562
563    if ciff::is_ciff(rawfile) {
564      let dec = Box::new(crw::CrwDecoder::new(rawfile, self)?);
565      return Ok(dec as Box<dyn Decoder>);
566    }
567
568    if jfif::is_exif(rawfile) {
569      let exif = jfif::Jfif::new(rawfile)?;
570      if let Some(make) = exif
571        .exif_ifd()
572        .and_then(|ifd| ifd.get_entry(TiffCommonTag::Make))
573        .and_then(|entry| entry.value.as_string().map(|s| s.as_str().trim_end()))
574      {
575        match make {
576          "Konica Minolta Photo Imaging, Inc." => {
577            let dec = Box::new(mrw::MrwDecoder::new_jfif(rawfile, exif, self)?);
578            return Ok(dec as Box<dyn Decoder>);
579          }
580          _ => {
581            log::warn!("Unknown make for EXIF file: {}", make);
582          }
583        }
584      }
585    }
586
587    if x3f::is_x3f(rawfile) {
588      let dec = Box::new(x3f::X3fDecoder::new(rawfile, self)?);
589      return Ok(dec as Box<dyn Decoder>);
590    }
591
592    match Bmff::new(&mut rawfile.reader()) {
593      Ok(bmff) => {
594        if bmff.compatible_brand("crx ") {
595          return Ok(Box::new(cr3::Cr3Decoder::new(rawfile, bmff, self)?));
596        }
597      }
598      Err(e) => {
599        debug!("It's not a BMFF file: {:?}", e);
600      }
601    }
602
603    match GenericTiffReader::new(&mut rawfile.reader(), 0, 0, None, &[]) {
604      Ok(tiff) => {
605        debug!("File is is TIFF file!");
606
607        macro_rules! use_decoder {
608          ($dec:ty, $buf:ident, $tiff:ident, $rawdec:ident) => {
609            Ok(Box::new(<$dec>::new(rawfile, $tiff, $rawdec)?) as Box<dyn Decoder>)
610          };
611        }
612
613        if tiff.has_entry(DngTag::DNGVersion) {
614          return Ok(Box::new(dng::DngDecoder::new(rawfile, tiff, self)?));
615        }
616
617        // The DCS560C is really a CR2 camera so we just special case it here
618        if let Some(model) = tiff.get_entry(TiffCommonTag::Model) {
619          if model.get_string().ok() == Some(&String::from("DCS560C")) {
620            return use_decoder!(cr2::Cr2Decoder, rawfile, tiff, self);
621          }
622        }
623
624        if let Some(make) = tiff
625          .get_entry(TiffCommonTag::Make)
626          .and_then(|entry| entry.value.as_string().map(|s| s.as_str().trim_end()))
627        {
628          match make {
629            "Canon" => return use_decoder!(cr2::Cr2Decoder, rawfile, tiff, self),
630            "PENTAX" => return use_decoder!(pef::PefDecoder, rawfile, tiff, self),
631            "PENTAX Corporation" => return use_decoder!(pef::PefDecoder, rawfile, tiff, self),
632            "RICOH IMAGING COMPANY, LTD." => return use_decoder!(pef::PefDecoder, rawfile, tiff, self),
633            "Phase One" => return use_decoder!(iiq::IiqDecoder, rawfile, tiff, self),
634            "Phase One A/S" => return use_decoder!(iiq::IiqDecoder, rawfile, tiff, self),
635            "Leaf" => return use_decoder!(iiq::IiqDecoder, rawfile, tiff, self),
636            "Hasselblad" => return use_decoder!(tfr::TfrDecoder, rawfile, tiff, self),
637            "SONY" => return use_decoder!(arw::ArwDecoder, rawfile, tiff, self),
638            "Mamiya-OP Co.,Ltd." => return use_decoder!(mef::MefDecoder, rawfile, tiff, self),
639            "OLYMPUS IMAGING CORP." => return use_decoder!(orf::OrfDecoder, rawfile, tiff, self),
640            "OLYMPUS CORPORATION" => return use_decoder!(orf::OrfDecoder, rawfile, tiff, self),
641            "OLYMPUS OPTICAL CO.,LTD" => return use_decoder!(orf::OrfDecoder, rawfile, tiff, self),
642            "OM Digital Solutions" => return use_decoder!(orf::OrfDecoder, rawfile, tiff, self),
643            "SAMSUNG" => return use_decoder!(srw::SrwDecoder, rawfile, tiff, self),
644            "SEIKO EPSON CORP." => return use_decoder!(erf::ErfDecoder, rawfile, tiff, self),
645            "EASTMAN KODAK COMPANY" => return use_decoder!(kdc::KdcDecoder, rawfile, tiff, self),
646            "Eastman Kodak Company" => return use_decoder!(kdc::KdcDecoder, rawfile, tiff, self),
647            "KODAK" => return use_decoder!(dcs::DcsDecoder, rawfile, tiff, self),
648            "Kodak" => return use_decoder!(dcr::DcrDecoder, rawfile, tiff, self),
649            "Panasonic" => return use_decoder!(rw2::Rw2Decoder, rawfile, tiff, self),
650            "LEICA" => return use_decoder!(rw2::Rw2Decoder, rawfile, tiff, self),
651            "LEICA CAMERA AG" => return use_decoder!(rw2::Rw2Decoder, rawfile, tiff, self),
652            //"FUJIFILM" => return use_decoder!(raf::RafDecoder, rawfile, tiff, self),
653            "NIKON" => return use_decoder!(nrw::NrwDecoder, rawfile, tiff, self),
654            "NIKON CORPORATION" => return use_decoder!(nef::NefDecoder, rawfile, tiff, self),
655            x => {
656              return Err(RawlerError::Unsupported {
657                what: format!("Couldn't find a decoder for make \"{}\"", x),
658                make: make.to_string(),
659                model: tiff
660                  .get_entry(TiffCommonTag::Model)
661                  .and_then(|entry| entry.as_string().cloned())
662                  .unwrap_or_default(),
663                mode: String::new(),
664              });
665            }
666          }
667        }
668
669        if tiff.has_entry(TiffCommonTag::Software) {
670          // Last ditch effort to identify Leaf cameras without Make and Model
671          if fetch_tiff_tag!(tiff, TiffCommonTag::Software).as_string() == Some(&"Camera Library".to_string()) {
672            return use_decoder!(mos::MosDecoder, rawfile, tiff, self);
673          }
674        }
675      }
676      Err(e) => {
677        debug!("File is not a tiff file: {:?}", e);
678      }
679    }
680
681    // If all else fails see if we match by filesize to one of those CHDK style files
682    let data = rawfile.as_vec().unwrap();
683    if let Some(cam) = self.naked.get(&data.len()) {
684      return Ok(Box::new(nkd::NakedDecoder::new(cam.clone(), self)?));
685    }
686
687    Err(RawlerError::Unsupported {
688      what: String::from("No decoder found"),
689      model: "".to_string(),
690      make: "".to_string(),
691      mode: "".to_string(),
692    })
693  }

696  fn check_supported_with_everything<'a>(&'a self, make: &str, model: &str, mode: &str) -> Result<Camera> {
697    match self.cameras.get(&(make.to_string(), model.to_string(), mode.to_string())) {
698      Some(cam) => Ok(cam.clone()),
699      None => Err(RawlerError::Unsupported {
700        what: String::from("Unknown camera"),
701        model: model.to_string(),
702        make: make.to_string(),
703        mode: mode.to_string(),
704      }),
705    }

732  }

758pub fn decode_unthreaded<F>(width: usize, height: usize, dummy: bool, closure: &F) -> PixU16
759where
760  F: Fn(&mut [u16], usize) + Sync,
761{
762  let mut out: PixU16 = alloc_image!(width, height, dummy);
763  out.pixels_mut().chunks_exact_mut(width).enumerate().for_each(|(row, line)| {
764    closure(line, row);
765  });
766  out
767}

780pub fn decode_threaded_multiline<F>(width: usize, height: usize, lines: usize, dummy: bool, closure: &F) -> std::result::Result<PixU16, String>
781where
782  F: Fn(&mut [u16], usize) -> std::result::Result<(), String> + Sync,
783{
784  let mut out: PixU16 = alloc_image_ok!(width, height, dummy);
785  out
786    .pixels_mut()
787    .par_chunks_mut(width * lines)
788    .enumerate()
789    .map(|(row, line)| closure(line, row * lines))
790    .collect::<std::result::Result<Vec<()>, _>>()?;
791  Ok(out)
792}

794/// This is used for streams where not chunked at line boundaries.
795pub fn decode_threaded_chunked<F>(width: usize, height: usize, chunksize: usize, dummy: bool, closure: &F) -> PixU16
796where
797  F: Fn(&mut [u16], usize) + Sync,
798{
799  let mut out: PixU16 = alloc_image!(width, height, dummy);
800  out.pixels_mut().par_chunks_mut(chunksize).enumerate().for_each(|(chunk_id, chunk)| {
801    closure(chunk, chunk_id);
802  });
803  out
