60/// Check if file has RAF signature
61pub fn is_raf(file: &RawSource) -> bool {
62  match file.subview(0, 8) {
63    Ok(buf) => buf[0..8] == b"FUJIFILM"[..],
64    Err(_) => false,
65  }
66}

68/// We need to inject a virtual IFD into main IFD.
69/// The RAF data block is not a regular TIFF structure but
70/// a proprietary structure. This tag id should be unlikely
71/// to appear in main IFD.
72const RAF_TAG_VIRTUAL_RAF_DATA: u16 = 0xfaaa;
73
74/// Parse a proprietary RAF data structure and return a virtual IFD.
75/// Unfortunately, Fujifilm forgot to add a type field to these
76/// tags, so we need to match by tag.
77pub fn parse_raf_format(file: &RawSource, offset: u32) -> Result<IFD> {
78  let mut entries = BTreeMap::new();
79  let stream = &mut file.reader();
80  stream.seek(SeekFrom::Start(offset as u64))?;
81  let num = stream.read_u32::<BigEndian>()?; // Directory entries in this IFD
82  if num > 4000 {
83    return Err(format_args!("too many entries in IFD ({})", num).into());
84  }
85  for _ in 0..num {
86    let tag = stream.read_u16::<BigEndian>()?;
87    let len = stream.read_u16::<BigEndian>()? as usize;
88    //eprintln!("RAF tag: 0x{:X}, len: {}", tag, len);
89
90    match RafTags::try_from(tag) {
91      Ok(RafTags::RawImageFullSize)
92      | Ok(RafTags::RawImageCropTopLeft)
93      | Ok(RafTags::RawImageCroppedSize)
94      | Ok(RafTags::RawImageAspectRatio)
95      | Ok(RafTags::WB_GRGBLevels) => {
96        let n = len / size_of::<u16>();
97        let entry = Entry {
98          tag,
99          value: Value::Short((0..n).map(|_| stream.read_u16::<BigEndian>()).collect::<std::io::Result<Vec<_>>>()?),
100          embedded: None,
101        };
102        entries.insert(tag, entry);
103      }
104      Ok(RafTags::FujiLayout) | Ok(RafTags::XTransLayout) => {
105        let n = len / size_of::<u8>();
106        let entry = Entry {
107          tag,
108          value: Value::Byte((0..n).map(|_| stream.read_u8()).collect::<std::io::Result<Vec<_>>>()?),
109          embedded: None,
110        };
111        entries.insert(tag, entry);
112      }
113      // This one is in other byte-order...
114      Ok(RafTags::RAFData) => {
115        let n = len / size_of::<u32>();
116        let entry = Entry {
117          tag,
118          value: Value::Long((0..n).map(|_| stream.read_u32::<LittleEndian>()).collect::<std::io::Result<Vec<_>>>()?),
119          embedded: None,
120        };
121        entries.insert(tag, entry);
122      }
123      // Skip other tags
124      _ => {
125        stream.seek(SeekFrom::Current(len as i64))?;
126      }
127    }
128  }
129  Ok(IFD {
130    entries,
131    endian: Endian::Big,
132    offset: 0,
133    base: offset as u32,
134    corr: 0,
135    next_ifd: 0,
136    sub: Default::default(),
137    chain: Default::default(),
138  })
139}

141/// RAF format contains multiple TIFF and TIFF-like structures.
142/// This creates a IFD with all other IFDs found collected as SubIFDs.
143fn parse_raf(file: &RawSource) -> Result<IFD> {
144  const RAF_TIFF1_PTR_OFFSET: u64 = 84;
145  const RAF_TIFF2_PTR_OFFSET: u64 = 100;
146  const RAF_TAGS_PTR_OFFSET: u64 = 92;
147  //const RAF_BLOCK_PTR_OFFSET2: u64 = 120; TODO: ?!?
148  log::debug!("parse RAF");
149  let stream = &mut file.reader();
150  stream.seek(SeekFrom::Start(RAF_TIFF1_PTR_OFFSET))?;
151  let offset = stream.read_u32::<BigEndian>()?;
152
153  // Main IFD
154  let mut main = IFD::new_root(stream, offset + 12)?;
155
156  //main.dump::<TiffCommonTag>(10).iter().for_each(|line| eprintln!("MAIN: {}", line));
157
158  // There is a second TIFF structure, the pointer is stored at offset 100.
159  // If it is not a valid TIFF structure, the pointer itself is the RAF offset.
160  stream.seek(SeekFrom::Start(RAF_TIFF2_PTR_OFFSET))?;
161  let ioffset = stream.read_u32::<BigEndian>()?;
162
163  match IFD::new_root_with_correction(stream, 0, ioffset, 0, 10, &[FujiIFD::FujiIFD.into()]) {
164    Ok(val) => {
165      log::debug!("Found valid FujiIFD (0xF000)");
166      //val.dump::<FujiIFD>(10).iter().for_each(|line| eprintln!("FujiIFD: {}", line));
167      main.sub.insert(FujiIFD::FujiIFD as u16, vec![val]);
168    }
169    Err(_) => {
170      // We fake an FujiIFD to pass the StripOffsets
171      log::debug!("Unable to find FujiIFD (0xF000), let's fake it");
172      let mut entries = BTreeMap::<u16, Entry>::new();
173      entries.insert(
174        FujiIFD::StripOffsets as u16,
175        Entry {
176          tag: FujiIFD::StripOffsets as u16,
177          value: Value::Long(vec![ioffset]), // The ioffset is absolute to the file start.
178          embedded: Some(RAF_TIFF2_PTR_OFFSET as u32),
179        },
180      );
181      let fake = IFD {
182        offset: 0,
183        base: 0, // For the faked IFD, the offsets are already absolute to the file start.
184        corr: 0,
185        next_ifd: 0,
186        entries,
187        endian: main.endian,
188        sub: Default::default(),
189        chain: Default::default(),
190      };
191      main.sub.insert(FujiIFD::FujiIFD as u16, vec![fake]);
192    }
193  }
194  // And we maybe have a RAF data block, try to parse it.
195  stream.seek(SeekFrom::Start(RAF_TAGS_PTR_OFFSET))?;
196  let raf_offset = stream.read_u32::<BigEndian>()?;
197  match parse_raf_format(file, raf_offset) {
198    Ok(val) => {
199      //val.dump::<RafTags>(10).iter().for_each(|line| eprintln!("RAFTAGS: {}", line));
200      main.sub.insert(RAF_TAG_VIRTUAL_RAF_DATA, vec![val]);
201    }
202    Err(_) => {
203      log::debug!("RAF block pointer is not valid, ignoring");
204    }
205  }
206
207  Ok(main)
208}

210impl<'a> RafDecoder<'a> {
211  pub fn new(file: &RawSource, rawloader: &'a RawLoader) -> Result<RafDecoder<'a>> {
212    let ifd = parse_raf(file)?;
213    let camera = rawloader.check_supported(&ifd)?;
214    let makernotes = if let Some(exif) = ifd.find_first_ifd_with_tag(ExifTag::MakerNotes) {
215      exif.parse_makernote(&mut file.reader(), OffsetMode::Absolute, &[])?
216    } else {
217      None
218    }
219    .ok_or("File has not makernotes")?;
220    //makernotes.dump::<RafMakernotes>(10).iter().for_each(|line| eprintln!("MKND: {}", line));
221    Ok(RafDecoder {
222      ifd,
223      makernotes,
224      rawloader,
225      camera,
226    })
227  }

231  fn raw_image(&self, file: &RawSource, _params: &RawDecodeParams, dummy: bool) -> Result<RawImage> {
232    let raw = self.ifd.find_first_ifd_with_tag(FujiIFD::StripOffsets).ok_or("No StripOffsets found")?;
233    let (width, height) = if raw.has_entry(FujiIFD::RawImageFullWidth) {
234      (
235        fetch_tiff_tag!(raw, FujiIFD::RawImageFullWidth).force_usize(0),
236        fetch_tiff_tag!(raw, FujiIFD::RawImageFullHeight).force_usize(0),
237      )
238    } else {
239      let raf = &self
240        .ifd
241        .sub_ifds()
242        .get(&RAF_TAG_VIRTUAL_RAF_DATA)
243        .and_then(|ifds| ifds.get(0))
244        .ok_or("No RAF data IFD found")?;
245      let sizes = fetch_tiff_tag!(raf, TiffCommonTag::ImageWidth);
246      (sizes.force_usize(1), sizes.force_usize(0))
247    };
248
249    let bps = match raw.get_entry(TiffCommonTag::RafBitsPerSample) {
250      Some(val) => val.force_u32(0) as usize,
251      None => 16,
252    };
253
254    // Rotation is only used for SuperCCD sensors, so we handle X-Trans CFA only here.
255    // Some cameras like X-T20 uses different CFA when compression is enabled, so we
256    // read the correct pattern from metadata.
257    let corrected_cfa = if let Some(cfa) = self.get_xtrans_cfa()? {
258      log::debug!(
259        "Found X-Trans CFA pattern in metadata, use this instead of camera config file. Pattern is: {}",
260        cfa
261      );
262      cfa
263    } else {
264      self.camera.cfa.clone()
265    };
266
267    // Strip offset is relative to IFD base
268    let offset = raw.base as u64 + fetch_tiff_tag!(raw, FujiIFD::StripOffsets).force_u64(0);
269    let src = if raw.has_entry(FujiIFD::StripByteCounts) {
270      let strip_count = fetch_tiff_tag!(raw, FujiIFD::StripByteCounts).force_u64(0);
271      file.subview_padded(offset, strip_count)?
272    } else {
273      // Some models like DBP don't have a byte count, so we read until EOF
274      file.subview_until_eof_padded(offset)?
275    };
276
277    log::debug!("BPS: {}, width: {}, height: {}, offset: {}", bps, width, height, offset);
278
279    let image = if self.camera.find_hint("double_width") {
280      // Some fuji SuperCCD cameras include a second raw image next to the first one
281      // that is identical but darker to the first. The two combined can produce
282      // a higher dynamic range image. Right now we're ignoring it.
283      decode_16le_skiplines(&src, width, height, dummy)
284    } else if self.camera.find_hint("jpeg32") {
285      match bps {
286        12 => decode_12be_msb32(&src, width, height, dummy),
287        14 => decode_14be_msb32(&src, width, height, dummy),
288        _ => return Err(RawlerError::unsupported(&self.camera, format!("RAF: Don't know how to decode bps {}", bps))),
289      }
290    } else if self.camera.clean_model == "DBP for GX680" {
291      assert_eq!(bps, 16);
292      dbp::decode_dbp(&src, width, height, dummy)?
293    } else if src.len() < bps * width * height / 8 {
294      if !dummy {
295        decompress_fuji(&src, width, height, bps, &corrected_cfa)?
296      } else {
297        alloc_image_plain!(width, height, dummy)
298      }
299    } else {
300      match bps {
301        12 => decode_12le(&src, width, height, dummy),
302        14 => decode_14le_unpacked(&src, width, height, dummy),
303        16 => {
304          if self.ifd.endian == Endian::Little {
305            decode_16le(&src, width, height, dummy)
306          } else {
307            decode_16be(&src, width, height, dummy)
308          }
309        }
310        _ => {
311          return Err(RawlerError::unsupported(&self.camera, format!("RAF: Don't know how to decode bps {}", bps)));
312        }
313      }
314    };
315
316    let blacklevel = self.get_blacklevel(&corrected_cfa)?;
317    log::debug!("RAF Blacklevels: {:?}", blacklevel);
318
319    // For now, we put the rotated data into DNG. Much better solution
320    // would be to support staggered layouts, but this is not used much
321    // and complicated to implement, because we need rectangular CFA patterns like 2x4.
322    // The code path for staggered data is already implemented here, but remains unused.
323    let rotate_for_dng = false;
324    let cpp = 1;
325    if self.camera.find_hint("fuji_rotation") || self.camera.find_hint("fuji_rotation_alt") {
326      log::debug!("Apply Fuji image rotation");
327      let rotated = if rotate_for_dng {
328        if self.camera.find_hint("fuji_rotation") {
329          fuji_raw_rotate(&image, dummy) // Only required for fuji_rotation
330        } else {
331          image
332        }
333      } else {
334        self.rotate_image(image.pixels(), &self.camera, width, height, dummy)?
335      };
336
337      let mut camera = self.camera.clone();
338      camera.cfa = corrected_cfa;
339      let photometric = RawPhotometricInterpretation::Cfa(CFAConfig::new_from_camera(&camera));
340      let mut image = RawImage::new(
341        self.camera.clone(),
342        rotated,
343        cpp,
344        normalize_wb(self.get_wb()?),
345        photometric,
346        blacklevel,
347        None,
348        dummy,
349      );
350
351      if rotate_for_dng {
352        image.add_dng_tag(TiffCommonTag::CFARepeatPatternDim, [2, 4]);
353        image.add_dng_tag(DngTag::CFALayout, 2_u16);
354        image.add_dng_tag(TiffCommonTag::CFAPattern, &[0_u8, 1, 2, 1, 2, 1, 0, 1][..]);
355
356        todo!();
357        //image.add_dng_tag(DngTag::BlackLevel, image.blacklevel[0]);
358        //image.add_dng_tag(DngTag::BlackLevelRepeatDim, [1_u16, 1_u16]);
359      }
360
361      // Reset crops because we have rotated the data.
362      image.active_area = None;
363      image.crop_area = None;
364      Ok(image)
365    } else {
366      //ok_image(self.camera.clone(), width, height, cpp, self.get_wb()?, image.into_inner())
367
368      let mut camera = self.camera.clone();
369      camera.cfa = corrected_cfa;
370      let whitelevel = if self.camera.whitelevel.is_none() {
371        match bps {
372          12 | 14 | 16 => {
373            let max_value: u32 = (1_u32 << bps) - 1;
374            Some(WhiteLevel::new(vec![max_value; cpp]))
375          }
376          _ => None,
377        }
378      } else {
379        None
380      };
381      let photometric = RawPhotometricInterpretation::Cfa(CFAConfig::new_from_camera(&camera));
382      let mut image = RawImage::new(camera, image, cpp, normalize_wb(self.get_wb()?), photometric, blacklevel, whitelevel, dummy);
383
384      // Overwrite crop if available in metadata
385      if let Some(crop) = self.get_crop()? {
386        log::debug!("RAW file metadata contains crop info, overriding toml definitions: {:?}", crop);
387        image.crop_area = Some(crop);
388      }
389
390      // Ideally, someone would expect that area is at bayer pattern
391      // boundary. This is not the case, so we don't check this here.
392      // if let Some(_area) = image.active_area.as_ref() {
393      //   assert_eq!(area.d.w % image.cfa.width , 0);
394      //   assert_eq!(area.d.h % image.cfa.height , 0);
395      // }
396      Ok(image)
397    }
398  }

400  fn raw_metadata(&self, _file: &RawSource, _params: &RawDecodeParams) -> Result<RawMetadata> {
401    let mut exif = Exif::new(&self.ifd)?;
402    // Fuji RAF has all EXIF tags we need and there is no LensID or something
403    // we can lookup. So this is an exception, we just pass the information.
404    // TODO: better imeplement LensData::from_exif()?
405    if let Some(ifd) = self.ifd.get_sub_ifd(TiffCommonTag::ExifIFDPointer) {
406      exif.lens_make = ifd.get_entry(ExifTag::LensMake).and_then(|entry| entry.as_string().cloned());
407      exif.lens_model = ifd.get_entry(ExifTag::LensModel).and_then(|entry| entry.as_string().cloned());
408      exif.lens_spec = ifd.get_entry(ExifTag::LensSpecification).and_then(|entry| match &entry.value {
409        Value::Rational(data) => Some([data[0], data[1], data[2], data[3]]),
410        _ => None,
411      });
412    }
413    let mdata = RawMetadata::new(&self.camera, exif);
414    Ok(mdata)
415  }

452  fn get_wb(&self) -> Result<[f32; 4]> {
453    let raw = self.ifd.find_first_ifd_with_tag(FujiIFD::StripOffsets).ok_or("No StripOffsets found")?;
454    match raw.get_entry(FujiIFD::WB_GRBLevels) {
455      Some(levels) => Ok([levels.force_f32(1), levels.force_f32(0), levels.force_f32(0), levels.force_f32(2)]),
456      None => {
457        let raf = &self
458          .ifd
459          .sub_ifds()
460          .get(&RAF_TAG_VIRTUAL_RAF_DATA)
461          .and_then(|ifds| ifds.get(0))
462          .ok_or("No RAF data IFD found")?;
463        let levels = fetch_tiff_tag!(raf, TiffCommonTag::RafOldWB);
464        Ok([levels.force_f32(1), levels.force_f32(0), levels.force_f32(0), levels.force_f32(3)])
465      }
466    }
467  }

469  fn get_blacklevel(&self, cfa: &CFA) -> Result<Option<BlackLevel>> {
470    if let Some(fuji) = self.ifd.get_sub_ifd(FujiIFD::FujiIFD) {
471      if let Some(Entry { value: Value::Long(black), .. }) = fuji.get_entry_recursive(FujiIFD::BlackLevel) {
472        let levels: Vec<u16> = black.iter().copied().map(|v| v as u16).collect();
473        return Ok(Some(BlackLevel::new(&levels, cfa.width, cfa.height, 1)));
474      } else {
475        log::debug!("Unable to find black level data");
476      }
477    }
478    Ok(None)
479  }

481  /// Get crop from metadata
482  /// Nearly all models have this parameter, except of FinePix HS10
483  fn get_crop(&self) -> Result<Option<Rect>> {
484    if let Some(raf) = &self.ifd.sub_ifds().get(&RAF_TAG_VIRTUAL_RAF_DATA).and_then(|ifds| ifds.get(0)) {
485      let crops = raf.get_entry(RafTags::RawImageCropTopLeft);
486      let size = raf.get_entry(RafTags::RawImageCroppedSize);
487      if let (Some(crops), Some(size)) = (crops, size) {
488        return Ok(Some(Rect::new(
489          Point::new(crops.force_usize(1), crops.force_usize(0)),
490          Dim2::new(size.force_usize(1), size.force_usize(0)),
491        )));
492      }
493    }
494    Ok(None)
495  }

497  /// Get the X-Trans CFA pattern
498  /// This is encoded in RAF metadata block in XTransLayout.
499  /// For unknown reason, the values are stored in reverse order and
500  /// also falsely reported by exiftoool.
501  fn get_xtrans_cfa(&self) -> Result<Option<CFA>> {
502    Ok(
503      if let Some(raf) = &self
504        .ifd
505        .sub_ifds()
506        .get(&RAF_TAG_VIRTUAL_RAF_DATA)
507        .and_then(|ifds| ifds.get(0).and_then(|ifd| ifd.get_entry(RafTags::XTransLayout)))
508      {
509        match &raf.value {
510          Value::Byte(data) => {
511            let patname: String = data
512              .iter()
513              .rev()
514              .map(|v| match v {
515                0 => 'R',
516                1 => 'G',
517                2 => 'B',
518                _ => 'X', // Unknown, let CFA::new() fail...
519              })
520              .collect();
521            Some(CFA::new(&patname))
522          }
523          _ => {
524            return Err("Invalid XTransLayout data type".into());
525          }
526        }
527      } else {
528        None
529      },
530    )
531  }

542  fn rotate_image(&self, src: &[u16], camera: &Camera, width: usize, height: usize, dummy: bool) -> Result<PixU16> {
543    if let Some(active_area) = self.camera.active_area {
544      let x = active_area[0];
545      let y = active_area[1];
546      let cropwidth = width - active_area[2] - x;
547      let cropheight = height - active_area[3] - y; // TODO: bug, invalid order of crop index
548
549      if camera.find_hint("fuji_rotation_alt") {
550        let rotatedwidth = cropheight + cropwidth / 2;
551        let rotatedheight = rotatedwidth - 1;
552
553        let mut out = alloc_image_plain!(rotatedwidth, rotatedheight, dummy);
554        if !dummy {
555          for row in 0..cropheight {
556            let inb = &src[(row + y) * width + x..];
557            for col in 0..cropwidth {
558              let out_row = rotatedwidth - (cropheight + 1 - row + (col >> 1));
559              let out_col = ((col + 1) >> 1) + row;
560              out[out_row * rotatedwidth + out_col] = inb[col];
561            }
562          }
563        }
564        Ok(out)
565      } else {
566        let rotatedwidth = cropwidth + cropheight / 2;
567        let rotatedheight = rotatedwidth - 1;
568
569        let mut out = alloc_image_plain!(rotatedwidth, rotatedheight, dummy);
570        if !dummy {
571          for row in 0..cropheight {
572            let inb = &src[(row + y) * width + x..];
573            for col in 0..cropwidth {
574              let out_row = cropwidth - 1 - col + (row >> 1);
575              let out_col = ((row + 1) >> 1) + col;
576              out[out_row * rotatedwidth + out_col] = inb[col];
577            }
578          }
579        }
580        Ok(out)
581      }
582    } else {
583      Err(RawlerError::DecoderFailed("no active_area for fuji_rotate".to_string()))
584    }
585  }
586}

588fn normalize_wb(raw_wb: [f32; 4]) -> [f32; 4] {
589  log::debug!("RAF raw wb: {:?}", raw_wb);
590  // We never have more then RGB colors so far (no RGBE etc.)
591  // So we combine G1 and G2 to get RGB wb.
592  let div = raw_wb[1];
593  let mut norm = raw_wb;
594  norm.iter_mut().for_each(|v| {
595    if v.is_normal() {
596      *v /= div
597    }
598  });
599  [norm[0], (norm[1] + norm[2]) / 2.0, norm[3], f32::NAN]
600}
