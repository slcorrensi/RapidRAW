50impl<'a> Decoder for TfrDecoder<'a> {
51  fn raw_image(&self, file: &RawSource, _params: &RawDecodeParams, dummy: bool) -> Result<RawImage> {
52    let raw = self.tiff.find_first_ifd_with_tag(TiffCommonTag::WhiteLevel).unwrap();
53
54    let whitelevel = raw
55      .get_entry(TiffCommonTag::WhiteLevel)
56      .map(|tag| WhiteLevel::new(vec![tag.force_u16(0) as u32]));
57
58    let blacklevel = raw
59      .get_entry(TiffCommonTag::BlackLevels)
60      .map(|tag| BlackLevel::new(&[tag.force_u16(0)], 1, 1, 1));
61
62    let width = fetch_tiff_tag!(raw, TiffCommonTag::ImageWidth).force_usize(0);
63    let height = fetch_tiff_tag!(raw, TiffCommonTag::ImageLength).force_usize(0);
64    let offset = fetch_tiff_tag!(raw, TiffCommonTag::StripOffsets).force_usize(0);
65
66    let src = file.subview_until_eof(offset as u64).unwrap();
67
68    let image = if self.camera.find_hint("uncompressed") {
69      decode_16le(src, width, height, dummy)
70    } else {
71      self.decode_compressed(src, width, height, dummy)?
72    };
73
74    let crop = Rect::from_tiff(raw).or_else(|| self.camera.crop_area.map(|area| Rect::new_with_borders(Dim2::new(width, height), &area)));
75
76    //crate::devtools::dump_image_u16(&image.data, width, height, "/tmp/tfrdump.pnm");
77
78    let cpp = 1;
79    let photometric = RawPhotometricInterpretation::Cfa(CFAConfig::new_from_camera(&self.camera));
80    let mut img = RawImage::new(self.camera.clone(), image, cpp, self.get_wb()?, photometric, blacklevel, whitelevel, dummy);
81
82    img.crop_area = crop;
83
84    Ok(img)
85  }

87  fn full_image(&self, file: &RawSource, params: &RawDecodeParams) -> Result<Option<DynamicImage>> {
88    if params.image_index != 0 {
89      return Ok(None);
90    }
91    let root_ifd = &self.tiff.root_ifd();
92    let buf = root_ifd
93      .singlestrip_data_rawsource(file)
94      .map_err(|e| RawlerError::DecoderFailed(format!("Failed to get strip data: {}", e)))?;
95    let compression = root_ifd.get_entry(TiffCommonTag::Compression).ok_or("Missing tag")?.force_usize(0);
96    let width = fetch_tiff_tag!(root_ifd, TiffCommonTag::ImageWidth).force_usize(0);
97    let height = fetch_tiff_tag!(root_ifd, TiffCommonTag::ImageLength).force_usize(0);
98    if compression == 1 {
99      Ok(Some(DynamicImage::ImageRgb8(
100        ImageBuffer::<Rgb<u8>, Vec<u8>>::from_raw(width as u32, height as u32, buf.to_vec()).unwrap(),
101      )))
102    } else {
103      let img = image::load_from_memory_with_format(buf, image::ImageFormat::Jpeg).unwrap();
104      Ok(Some(img))
105    }
106  }

116  fn raw_metadata(&self, _file: &RawSource, _params: &RawDecodeParams) -> Result<RawMetadata> {
117    let exif = Exif::new(self.tiff.root_ifd())?;
118    let mut mdata = RawMetadata::new_with_lens(&self.camera, exif, self.get_lens_description()?.cloned());
119    // Read Unique ID
120    if let Some(Entry {
121      value: Value::Byte(unique_id), ..
122    }) = self.tiff.root_ifd().get_entry(DngTag::RawDataUniqueID)
123    {
124      if let Ok(id) = unique_id.as_slice().try_into() {
125        mdata.unique_image_id = Some(u128::from_le_bytes(id));
126      }
127    }
128    Ok(mdata)
129  }

131  fn xpacket(&self, _file: &RawSource, _params: &RawDecodeParams) -> Result<Option<Vec<u8>>> {
132    match self.tiff.root_ifd().get_entry(TiffCommonTag::Xmp) {
133      Some(Entry { value: Value::Byte(buf), .. }) => Ok(Some(buf.clone())),
134      _ => Ok(None),
135    }
136  }

139impl<'a> TfrDecoder<'a> {
140  /// Get lens description by analyzing TIFF tags and makernotes
141  fn get_lens_description(&self) -> Result<Option<&'static LensDescription>> {
142    if let Some(exif) = self.tiff.root_ifd().get_sub_ifd(TiffCommonTag::ExifIFDPointer) {
143      let lens_make = exif.get_entry(ExifTag::LensMake).and_then(|entry| entry.as_string());
144      let lens_model = exif.get_entry(ExifTag::LensModel).and_then(|entry| entry.as_string());
145      let focal_len = match exif.get_entry(ExifTag::FocalLength) {
146        Some(Entry { value: Value::Rational(x), .. }) => x.get(0).cloned(),
147        Some(Entry { value: Value::Short(x), .. }) => x.get(0).copied().map(Rational::from),
148        _ => None,
149      };
150      let resolver = LensResolver::new()
151        .with_camera(&self.camera)
152        .with_lens_make(lens_make)
153        .with_lens_model(lens_model)
154        .with_focal_len(focal_len)
155        .with_mounts(&["x-mount".into()]);
156      return Ok(resolver.resolve());
157    }
158    Ok(None)
159  }
160
161  fn get_wb(&self) -> Result<[f32; 4]> {
162    let levels = fetch_tiff_tag!(self.tiff, TiffCommonTag::AsShotNeutral);
163    assert_eq!(levels.count(), 3);
164    Ok([1.0 / levels.force_f32(0), 1.0 / levels.force_f32(1), 1.0 / levels.force_f32(2), f32::NAN])
165  }
166
167  fn decode_compressed(&self, src: &[u8], width: usize, height: usize, dummy: bool) -> Result<PixU16> {
168    let mut out = alloc_image_ok!(width, height, dummy);
169    let decompressor = LjpegDecompressor::new_full(src, true, false)?;
170    decompressor.decode(out.pixels_mut(), 0, width, width, height, dummy)?;
171    Ok(out)
172  }
