6pub(super) fn nef_decrypt(buf: &mut [u8], start: usize, makernote: &IFD) -> Result<()> {
7  let serial = fetch_tiff_tag!(makernote, NikonMakernote::NefSerial);
8  let data = serial.get_data();
9  let mut serialno = 0_usize;
10  for i in 0..serial.count() as usize {
11    if data[i] == 0 {
12      break;
13    }
14    serialno = serialno * 10
15      + if data[i] >= 48 && data[i] <= 57 {
16        // "0" to "9"
17        (data[i] - 48) as usize
18      } else {
19        (data[i] % 10) as usize
20      };
21  }
22
23  // Get the "decryption" key
24  let keydata = fetch_tiff_tag!(makernote, NikonMakernote::NefKey).force_u32(0).to_le_bytes();
25  let keyno = (keydata[0] ^ keydata[1] ^ keydata[2] ^ keydata[3]) as usize;
26
27  let src = &mut buf[start..];
28
29  let ci = WB_SERIALMAP[serialno & 0xff] as u32;
30  let mut cj = WB_KEYMAP[keyno & 0xff] as u32;
31  let mut ck = 0x60_u32;
32  for i in 0..src.len() {
33    cj += ci * ck;
34    ck += 1;
35    src[i] ^= cj as u8;
36  }
37  Ok(())
38}
