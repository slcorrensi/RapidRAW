84  pub fn make_blacklevel(&self, cpp: usize) -> Option<BlackLevel> {
85    self.blacklevel.as_ref().map(|x| {
86      if x.len() == 1 {
87        BlackLevel::new(&vec![x[0]; cpp], 1, 1, cpp)
88      } else if x.len() == self.cfa.width * self.cfa.height * cpp {
89        BlackLevel::new(x, self.cfa.width, self.cfa.height, cpp)
90      } else {
91        panic!("Invalid blacklevel data")
92      }
93    })
94  }
95
96  pub fn make_whitelevel(&self, cpp: usize) -> Option<WhiteLevel> {
97    self.whitelevel.as_ref().map(|x| {
98      if x.len() == 1 {
99        WhiteLevel(vec![x[0] as u32; cpp])
100      } else if x.len() == cpp {
101        WhiteLevel(x.clone())
102      } else {
103        panic!("Invalid whitelevel data")
104      }
105    })
106  }

108  pub fn update_from_toml(&mut self, ct: &toml::value::Table) {
109    for (name, val) in ct {
110      match name.as_ref() {
111        n @ "make" => {
112          self.make = val.as_str().unwrap_or_else(|| panic!("{} must be a string", n)).to_string();
113        }
114        n @ "model" => {
115          self.model = val.as_str().unwrap_or_else(|| panic!("{} must be a string", n)).to_string();
116        }
117        n @ "mode" => {
118          self.mode = val.as_str().unwrap_or_else(|| panic!("{} must be a string", n)).to_string();
119        }
120        n @ "clean_make" => {
121          self.clean_make = val.as_str().unwrap_or_else(|| panic!("{} must be a string", n)).to_string();
122        }
123        n @ "clean_model" => {
124          self.clean_model = val.as_str().unwrap_or_else(|| panic!("{} must be a string", n)).to_string();
125        }
126        n @ "remark" => {
127          self.remark = Some(val.as_str().unwrap_or_else(|| panic!("{} must be a string", n)).to_string());
128        }
129        n @ "whitepoint" => {
130          let white = val.as_integer().unwrap_or_else(|| panic!("{} must be an integer", n)) as u32;
131          self.whitelevel = Some(vec![white]);
132        }
133        n @ "blackpoint" => {
134          let black = val.as_integer().unwrap_or_else(|| panic!("{} must be an integer", n)) as u32;
135          self.blacklevel = Some(vec![black]);
136        }
137        n @ "blackareah" => {
138          let vals = val.as_array().unwrap_or_else(|| panic!("{} must be an array", n));
139          self.blackareah = Some((vals[0].as_integer().unwrap() as usize, vals[1].as_integer().unwrap() as usize));
140        }
141        n @ "blackareav" => {
142          let vals = val.as_array().unwrap_or_else(|| panic!("{} must be an array", n));
143          self.blackareav = Some((vals[0].as_integer().unwrap() as usize, vals[1].as_integer().unwrap() as usize));
144        }
145        "color_matrix" => {
146          if let Some(color_matrix) = val.as_table() {
147            for (illu_str, matrix) in color_matrix.into_iter() {
148              let illu = Illuminant::new_from_str(illu_str).unwrap();
149              let xyz_to_cam = matrix
150                .as_array()
151                .expect("color matrix must be array")
152                .iter()
153                .map(|a| a.as_float().expect("color matrix values must be float") as f32)
154                .collect();
155              self.color_matrix.insert(illu, xyz_to_cam);
156            }
157          } else {
158            eprintln!("Invalid matrix spec for {}", self.clean_model);
159          }
160          assert!(!self.color_matrix.is_empty());
161        }
162        n @ "active_area" => {
163          let crop_vals = val.as_array().unwrap_or_else(|| panic!("{} must be an array", n));
164          let mut crop = [0, 0, 0, 0];
165          for (i, val) in crop_vals.iter().enumerate() {
166            crop[i] = val.as_integer().unwrap() as usize;
167          }
168          self.active_area = Some(crop);
169        }
170        n @ "crop_area" => {
171          let crop_vals = val.as_array().unwrap_or_else(|| panic!("{} must be an array", n));
172          let mut crop = [0, 0, 0, 0];
173          for (i, val) in crop_vals.iter().enumerate() {
174            crop[i] = val.as_integer().unwrap() as usize;
175          }
176          self.crop_area = Some(crop);
177        }
178        n @ "color_pattern" => {
179          self.cfa = CFA::new(val.as_str().unwrap_or_else(|| panic!("{} must be a string", n)));
180        }
181        n @ "plane_color" => {
182          self.plane_color = PlaneColor::new(val.as_str().unwrap_or_else(|| panic!("{} must be a string", n)));
183        }
184        n @ "bps" => {
185          self.bps = Some(val.as_integer().unwrap_or_else(|| panic!("{} must be an integer", n)) as usize);
186        }
187        n @ "real_bps" => {
188          self.real_bps = val.as_integer().unwrap_or_else(|| panic!("{} must be an integer", n)) as usize;
189        }
190        n @ "filesize" => {
191          self.filesize = val.as_integer().unwrap_or_else(|| panic!("{} must be an integer", n)) as usize;
192        }
193        n @ "raw_width" => {
194          self.raw_width = val.as_integer().unwrap_or_else(|| panic!("{} must be an integer", n)) as usize;
195        }
196        n @ "raw_height" => {
197          self.raw_height = val.as_integer().unwrap_or_else(|| panic!("{} must be an integer", n)) as usize;
198        }
199        n @ "highres_width" => {
200          self.highres_width = val.as_integer().unwrap_or_else(|| panic!("{} must be an integer", n)) as usize;
201        }
202        n @ "default_scale" => {
203          let scale_vals = val.as_array().unwrap_or_else(|| panic!("{} must be an array", n));
204          let scale_h = scale_vals[0].as_array().expect("must be array");
205          let scale_v = scale_vals[1].as_array().expect("must be array");
206          let scale = [
207            [
208              scale_h[0].as_integer().expect("must be integer") as u32,
209              scale_h[1].as_integer().expect("must be integer") as u32,
210            ],
211            [
212              scale_v[0].as_integer().expect("must be integer") as u32,
213              scale_v[1].as_integer().expect("must be integer") as u32,
214            ],
215          ];
216          self.default_scale = DefaultScale(scale);
217        }
218        n @ "best_quality_scale" => {
219          let scale_vals = val.as_array().unwrap_or_else(|| panic!("{} must be an array", n));
220          self.best_quality_scale = BestQualityScale([
221            scale_vals[0].as_integer().expect("must be integer") as u32,
222            scale_vals[1].as_integer().expect("must be integer") as u32,
223          ]);
224        }
225        n @ "hints" => {
226          self.hints = Vec::new();
227          for hint in val.as_array().unwrap_or_else(|| panic!("{} must be an array", n)) {
228            self.hints.push(hint.as_str().expect("hints must be a string").to_string());
229          }
230        }
231        n @ "params" => {
232          for (name, val) in val.as_table().unwrap_or_else(|| panic!("{} must be a table", n)) {
233            self.params.insert(name.clone(), val.clone());
234          }
235        }
236        "model_aliases" => {}
237        "modes" => {} // ignore
238        key => {
239          panic!("Unknown key: {}", key);
240        }
241      }
242    }
243  }
