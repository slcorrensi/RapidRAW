21#[derive(Clone, Debug)]
22struct CF2Params {
23  /// Unknown value, it's labeled as strip_height but don't match any height
24  #[allow(dead_code)]
25  strip_height: u32,
26
27  /// Unknown value, it's labeled as strip_width but don't match any width
28  #[allow(dead_code)]
29  strip_width: u32,
30
31  /// Gamma point and slope value.
32  gamma_point: Vec<u32>,
33  gamma_slope: Vec<u32>,
34
35  /// Max data value
36  gamma_clip_val: u16,
37
38  /// Initial values (base) for huffman coding
39  huf_init_val0: u16,
40
41  /// Initial values (base) for huffman coding
42  huf_init_val1: u16,
43
44  /// Initial values (base) for huffman coding
45  huf_init_val2: u16,
46
47  /// Initial values (base) for huffman coding
48  huf_init_val3: u16,
49
50  /// Stored huffman table, usually 17 entries
51  ///
52  /// This is a pair of (bitcnt, symbol).
53  ///
54  /// Example table:
55  /// 0000000     10   1022     11   2046      8    254      9    510
56  /// 0000020      7    126      4     14      4     12      3      4
57  /// 0000040      3      2      2      0      3      3      3      5

62
63  /// Shift down (0 in all samples...)
64  huf_shift_down: Vec<u16>,
65
66  /// Number of H strips
67  num_of_strips_h: u16,
68
69  /// Number of V strips
70  num_of_strips_v: u16,
71
72  /// Offset to bitstream
73  strip_byte_offsets: Vec<u32>,
74
75  /// Starting column offset in a output line
76  strip_line_offsets: Vec<u32>,
77
78  /// Size in bits of compressed bitstream
79  strip_data_size: Vec<u32>,
80
81  /// Strip widths in pixels
82  strip_widths: Vec<u16>,
83
84  /// Strip heights in pixels
85  strip_heights: Vec<u16>,
86}

88impl CF2Params {
89  fn new(ifd: &IFD) -> Result<Self> {
90    let strip_height = fetch_tiff_tag!(ifd, PanasonicTag::CF2StripHeight).force_u32(0);
91    let gamma_point = {
92      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2Unknown1).get_data(), Endian::Little);
93      let count = bs.get_u16();
94      (0..count).map(|_| bs.get_u32()).collect()
95    };
96    let gamma_slope = {
97      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2Unknown2).get_data(), Endian::Little);
98      let count = bs.get_u16();
99      (0..count).map(|_| bs.get_u32()).collect()
100    };
101    let gamma_clip_val = fetch_tiff_tag!(ifd, PanasonicTag::CF2ClipVal).force_u16(0);
102    let huf_init_val0 = fetch_tiff_tag!(ifd, PanasonicTag::CF2HufInitVal0).force_u16(0);
103    let huf_init_val1 = fetch_tiff_tag!(ifd, PanasonicTag::CF2HufInitVal1).force_u16(0);
104    let huf_init_val2 = fetch_tiff_tag!(ifd, PanasonicTag::CF2HufInitVal2).force_u16(0);
105    let huf_init_val3 = fetch_tiff_tag!(ifd, PanasonicTag::CF2HufInitVal3).force_u16(0);
106    let huf_table = {
107      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2HufTable).get_data(), Endian::Little);
108      let count = bs.get_u16();
109      (0..count).map(|_| (bs.get_u16(), bs.get_u16())).collect()
110    };
111    let huf_shift_down = {
112      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2HufShiftDown).get_data(), Endian::Little);
113      let count = bs.get_u16();
114      (0..count).map(|_| bs.get_u16()).collect()
115    };
116    let num_of_strips_h = fetch_tiff_tag!(ifd, PanasonicTag::CF2NumberOfStripsH).force_u16(0);
117    let num_of_strips_v = fetch_tiff_tag!(ifd, PanasonicTag::CF2NumberOfStripsV).force_u16(0);
118    let strip_byte_offsets = {
119      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2StripByteOffsets).get_data(), Endian::Little);
120      let count = bs.get_u16();
121      (0..count).map(|_| bs.get_u32()).collect()
122    };
123    let strip_line_offsets = {
124      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2StripLineOffsets).get_data(), Endian::Little);
125      let count = bs.get_u16();
126      (0..count).map(|_| bs.get_u32()).collect()
127    };
128    let strip_data_size = {
129      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2StripDataSize).get_data(), Endian::Little);
130      let count = bs.get_u16();
131      (0..count).map(|_| bs.get_u32()).collect()
132    };
133    let strip_widths = {
134      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2StripWidths).get_data(), Endian::Little);
135      let count = bs.get_u16();
136      (0..count).map(|_| bs.get_u16()).collect()
137    };
138    let strip_heights = {
139      let mut bs = ByteStream::new(fetch_tiff_tag!(ifd, PanasonicTag::CF2StripHeights).get_data(), Endian::Little);
140      let count = bs.get_u16();
141      (0..count).map(|_| bs.get_u16()).collect()
142    };
143    let strip_width = fetch_tiff_tag!(ifd, PanasonicTag::CF2StripWidth).force_u32(0);
144
145    Ok(Self {
146      strip_height,
147      gamma_point,
148      gamma_slope,
149      gamma_clip_val,
150      huf_init_val0,
151      huf_init_val1,
152      huf_init_val2,
153      huf_init_val3,
154      huf_table,
155      huf_shift_down,
156      num_of_strips_h,
157      num_of_strips_v,
158      strip_byte_offsets,
159      strip_line_offsets,
160      strip_data_size,
161      strip_widths,
162      strip_heights,
163      strip_width,
164    })
165  }
166}

168#[derive(Clone, Debug, Default)]
169struct HuffmanSymbol {
170  /// Length of the symbol in bits
171  bitcnt: u8,
172  /// Actual Huffman symbol, right-padded with 0 bits
173  symbol: u16,
174  /// Pre-calculated bitmask, actually (-1) << (16-bits)
175  mask: u16,
176}
177
178#[derive(Clone, Debug, Default)]
179struct HuffmanDecoder {
180  #[allow(dead_code)]
181  huff_symbols: [HuffmanSymbol; 17],
182
183  /// Lookup cache for all possible u16 values.
184  /// If a 16 bit input value is invalid (the symbol is undefined), the
185  /// value is None, otherwise it's (bitcnt, ssss)
186  cache: Vec<Option<(u8, u8)>>,
187}
188
189impl HuffmanDecoder {
190  fn new(symlens: impl Iterator<Item = u8>, mut symbols: impl Iterator<Item = u16>) -> Self {
191    let mut huff_symbols: [HuffmanSymbol; 17] = Default::default();
192    for (i, symlen) in symlens.enumerate() {
193      let symbol = symbols.next().expect("symbol iterator is shorter than symlens iterator");
194      let bitmask = 0xFFFFu16 >> (16 - symlen);
195      debug_assert_eq!(symbol, symbol & bitmask);
196      // Left-align symbol and mask
197      huff_symbols[i] = HuffmanSymbol {
198        bitcnt: symlen,
199        symbol: (symbol) << (16 - symlen),
200        mask: 0xFFFF << (16 - symlen),
201      };
202    }
203
204    // Generate lookup cache for all possible 16 bit input values.
205    let cache = (0..=0xFFFFu16)
206      .map(|x| Self::slow_lookup(&huff_symbols, x).map(|ssss| (huff_symbols[ssss as usize].bitcnt, ssss)))
207      .collect_vec();
208    Self { huff_symbols, cache }
209  }
210
211  /// Slow lookup into Huffman symbol table
212  fn slow_lookup(huff_symbols: &[HuffmanSymbol; 17], bits: u16) -> Option<u8> {
213    for i in 0..17 {
214      if (bits & huff_symbols[i].mask) == huff_symbols[i].symbol {
215        return Some(i as u8);
216      }
217    }
218    None
219  }
220
221  /// Extract Huffman symbol from bitstream pump and
222  /// return index into symbol table.
223  fn get_next(&self, pump: &mut dyn BitPump) -> u8 {
224    let next_bits = pump.peek_bits(16);
225    debug_assert_eq!(self.cache.len(), u16::MAX as usize + 1);
226    if let Some((bits, ssss)) = unsafe { *self.cache.get_unchecked(next_bits as usize) } {
227      pump.consume_bits(bits as u32);
228      ssss
229    } else {
230      panic!("Input value {:016b} starts not with a valid huffman symbol", next_bits);
231    }
232  }
233}

235/// Internal decoder state
236struct State {
237  huffdec: HuffmanDecoder,
238  gamma_table: Option<Vec<u16>>,
239  datamax: i32,
240  line_base: CoeffBase,
241  current_base: CoeffBase,
242}
243
244impl State {
245  fn new(params: &CF2Params) -> Self {
246    let initial = [params.huf_init_val0, params.huf_init_val1, params.huf_init_val2, params.huf_init_val3];
247    let gamma_table = make_gammatable(params);
248
249    let line_base = CoeffBase::new([initial[0], initial[1], initial[2], initial[3]]);
250    let current_base = line_base;
251
252    let huffdec = HuffmanDecoder::new(
253      params.huf_table.iter().map(|(bits, _symbol)| *bits as u8),
254      params.huf_table.iter().map(|(_bits, symbol)| *symbol),
255    );
256
257    Self {
258      huffdec,
259      gamma_table,
260      datamax: params.gamma_clip_val as i32,
261      line_base,
262      current_base,
263    }
264  }
265}

268struct CoeffBase {
269  coeff: [u16; 4],
270}

272impl CoeffBase {
273  pub fn new(coeff: [u16; 4]) -> Self {
274    Self { coeff }
275  }

277  pub fn update(&mut self, line: &[u16]) {
278    self.coeff = [line[0], line[1], line[2], line[3]];

282fn calc_gamma(params: &CF2Params, idx: u32) -> u16 {
283  let gamma_base = 0;
284  let gamma_points = &params.gamma_point; // [65536, 65536, 65536, 65536, 65536, 65536]
285  let gamma_slopes = &params.gamma_slope; // 0
286  let clipping = params.gamma_clip_val;
287
288  let mut x = {
289    let mut tmp: u32 = idx | 0xFFFF0000;
290    if (idx & 0x10000) == 0 {
291      tmp = idx & 0x1FFFF;
292    }
293    u32::min(gamma_base + tmp, 0xFFFF)
294  };
295
296  let mut idx = 0;
297  if (x & 0x80000000) != 0 {
298    x = 0;
299  }
300
301  if x >= (0xFFFF & gamma_slopes[1]) {
302    idx = 1;
303    if x >= (0xFFFF & gamma_slopes[2]) {
304      idx = 2;
305      if x >= (0xFFFF & gamma_slopes[3]) {
306        idx = 3;
307        if x >= (0xFFFF & gamma_slopes[4]) {
308          idx = (((x as u64 | 0x5_00000000u64) - (0xFFFF & gamma_slopes[5]) as u64) >> 32) as usize;
309        }
310      }
311    }
312  }
313
314  let point = gamma_points[idx];
315  let slope = gamma_slopes[idx];
316  let mut tmp: u32 = x - (slope & 0xFFFF);
317  let result: u16;
318
319  if (point & 0x1F) == 31 {
320    result = if idx == 5 { 0xFFFF } else { (gamma_slopes[idx + 1] >> 16) & 0xFFFF } as u16;
321    return u16::min(result, clipping);
322  }
323  if (point & 0x10) == 0 {
324    if (point & 0x1F) == 15 {
325      let result = ((slope >> 16) & 0xFFFF) as u16;
326      return u16::min(result, clipping);
327    } else if (point & 0x1F) != 0 {
328      tmp = (tmp + (1 << ((point & 0x1F) - 1))) >> (point & 0x1F);
329    }
330  } else {
331    tmp <<= point & 0xF;
332  }
333
334  result = (tmp + ((slope >> 16) & 0xFFFF)) as u16;
335  u16::min(result, clipping)
336}
337
338fn make_gammatable(params: &CF2Params) -> Option<Vec<u16>> {
339  const LINEAR_POINTS: [u32; 6] = [65536; 6];
340  const LINEAR_SLOPES: [u32; 6] = [0; 6];
341
342  if params.gamma_point == LINEAR_POINTS && params.gamma_slope == LINEAR_SLOPES {
343    None
344  } else {
345    let mut table = vec![0; 0x10000];
346    let mut _linear = true;
347    for idx in 0..0x10000 {
348      table[idx] = calc_gamma(params, idx as u32);
349      _linear = _linear && table[idx] == idx as u16;
350    }
351    Some(table)
352  }
353}

355/// Decode Panasonic V8 bitstreams
356pub(crate) fn decode_panasonic_v8(rawfile: &RawSource, width: usize, height: usize, _bps: u32, ifd: &IFD, dummy: bool) -> Result<PixU16> {
357  let out = alloc_image_ok!(width, height, dummy);
358
359  let params = CF2Params::new(ifd)?;
360  log::debug!("pana8: params: {:?}", params);
361
362  assert_eq!(params.strip_widths.iter().map(|x| *x as i32).sum::<i32>() as usize, width);
363  assert_eq!(params.num_of_strips_v, 1, "multiple v strips not supported");
364
365  // Shared output buffer, we need to write from multiple rayon threads to output image.
366  let shared_pix = SharedPix2D::new(out);
367
368  let mut bitstreams = Vec::with_capacity(params.num_of_strips_h as usize);
369  for strip_id in 0..params.num_of_strips_h as usize {
370    bitstreams.push(rawfile.subview(params.strip_byte_offsets[strip_id] as u64, (params.strip_data_size[strip_id] as u64 + 7) / 8)?);
371  }
372
373  // Parallel decode multiplce strips
374  (0..params.num_of_strips_h as usize).into_par_iter().for_each(|strip_id| {
375    let buf = &bitstreams[strip_id];
376    decode_strip(buf, &params, strip_id, unsafe { shared_pix.inner_mut() });
377  });
378  Ok(shared_pix.into_inner())
379}

381/// Decode a single strip
382fn decode_strip(buf: &[u8], params: &CF2Params, strip_id: usize, out: &mut PixU16) {
383  let mut pump = BitPumpReverseBitsMSB::new(buf);
384  let width = params.strip_widths[strip_id] as usize;
385  let height = params.strip_heights[strip_id] as usize;
386  let halfheight = height >> 1;
387  let halfwidth = width >> 1;
388  let doublewidth = halfwidth * 4;
389  let left_margin = params.strip_line_offsets[strip_id] as usize;
390  let mut linebuf = vec![0_u16; doublewidth];
391
392  // for (i, item) in params.huf_table.iter().enumerate() {
393  //   let fmt = format!("{:#032b}", item.1);
394  //   log::debug!("Pana8 huf_table {i}: {} (bits: {})", fmt.split_at((32 - item.0) as usize).1, item.0);
395  //   //log::debug!("Pana8: huf_table {:02}: {}, {}", i, item.0, item.1);
396  // }
397
398  let mut state = State::new(params);
399
400  // Data is encoded in RGRGRG..GBGBGB in a single line (like LJPEG92 4-7 predictors)
401  for curr_row in 0..halfheight {
402    state.current_base = state.line_base;
403    for col in 0..doublewidth {
404      // Calculate index
405      let ssss = state.huffdec.get_next(&mut pump);
406
407      // Shiftdown seems to be the count of bits shifted to right during encoding.
408      // It's 0 for all existing samples so far, highly interested in samples that has
409      let shift_down: u8 = (params.huf_shift_down[ssss as usize] & 0x1F) as u8;
410      assert_eq!(shift_down, 0, "CF2HufShiftDown samples required");
411
412      // Calculate total required bits to read from bitstream.
413      let req_bits: u32 = ssss.saturating_sub(shift_down as u8) as u32;
414      let delta1: i32 = if req_bits == 0 {
415        0
416      } else {
417        debug_assert_ne!(req_bits, 0);
418        let rawbits: u32 = pump.get_bits(req_bits as u32); // Get additional bits
419        let sign = rawbits >> (req_bits - 1); // Get leading sign bit
420        let val = (rawbits << (params.huf_shift_down[ssss as usize] & 0xFF)) as i32;
421
422        if sign == 1 {
423          val
424        } else if ssss > 0 {
425          if shift_down != 0 { val + (-1 << ssss) } else { val + (-1 << ssss) + 1 }
426        } else {
427          0
428        }
429      };
430
431      let delta2 = if shift_down != 0 { 1 << (shift_down - 1) } else { 0 };
432      let delta = delta1 + delta2;
433
434      // For each col iteration, we write to ONE pixel of of 4-pixel group.
435      let destpixel = &mut linebuf[col & !0x3..];
436
437      if col & 3 == 2 {
438        let val = state.current_base.coeff[1] as i32 + delta;
439        destpixel[1] = clamp(val, 0, state.datamax) as u16;
440      } else if col & 3 == 1 {
441        let val = state.current_base.coeff[2] as i32 + delta;
442        destpixel[2] = clamp(val, 0, state.datamax) as u16;
443      } else if col & 3 != 0 {
444        let val = state.current_base.coeff[3] as i32 + delta;
445        destpixel[3] = clamp(val, 0, state.datamax) as u16;
446      } else {
447        let val = state.current_base.coeff[0] as i32 + delta;
448        destpixel[0] = clamp(val, 0, state.datamax) as u16;
449      }
450
451      if col & 3 == 3 {
452        state.current_base.update(destpixel);
453      }
454      if col == 3 {
455        // base for next line (col == 3 -> first 4 pixels are complete in current row)
456        state.line_base.update(&linebuf);
457      }
458    }
459
460    // Copy line buffer into output image.
461    // Line buffer contains two rows packed into one row with double width.
462    assert_eq!(linebuf.len(), 2 * width);
463    for col in (0..width).step_by(2) {
464      let dest_row = curr_row * 2;
465      if let Some(gamma_table) = &state.gamma_table {
466        debug_assert!(!gamma_table.is_empty());
467        *out.at_mut(dest_row + 0, left_margin + col + 0) = gamma_table[linebuf[2 * col + 0] as usize];
468        *out.at_mut(dest_row + 0, left_margin + col + 1) = gamma_table[linebuf[2 * col + 1] as usize];
469        *out.at_mut(dest_row + 1, left_margin + col + 0) = gamma_table[linebuf[2 * col + 2] as usize];
470        *out.at_mut(dest_row + 1, left_margin + col + 1) = gamma_table[linebuf[2 * col + 3] as usize];
471      } else {
472        *out.at_mut(dest_row + 0, left_margin + col + 0) = linebuf[2 * col + 0];
473        *out.at_mut(dest_row + 0, left_margin + col + 1) = linebuf[2 * col + 1];
474        *out.at_mut(dest_row + 1, left_margin + col + 0) = linebuf[2 * col + 2];
475        *out.at_mut(dest_row + 1, left_margin + col + 1) = linebuf[2 * col + 3];
476      }
477    }
478  }
