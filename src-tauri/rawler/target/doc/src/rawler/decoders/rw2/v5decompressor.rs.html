7/// This works for 12 and 14 bit depth images
8pub(crate) fn decode_panasonic_v5(buf: &[u8], width: usize, height: usize, bps: u32, dummy: bool) -> PixU16 {
9  // Raw data is divided into blocks of same size
10  const V5_BLOCK_SIZE: usize = 0x4000;
11  // Each block is splitted and swapped, we need to swap back
12  const V5_SECTION_SPLIT_OFFSET: usize = 0x1FF8;
13  // Each block contains multiple packets
14  const V5_BYTES_PER_PACKET: usize = 16;
15  // Count of packets in a block
16  const V5_PACKETS_PER_BLOCK: usize = V5_BLOCK_SIZE / V5_BYTES_PER_PACKET;
17  // Depending on bit depth, a packet forms different amount of pixels
18  let pixels_per_packet = match bps {
19    12 => 10,
20    14 => 9,
21    _ => unreachable!(),
22  };
23  // Pixel count per full block
24  let pixels_per_block = V5_PACKETS_PER_BLOCK * pixels_per_packet;
25
26  log::debug!("RW2 V5 decoder: pixels per block: {}, bps: {}", pixels_per_block, bps);
27
28  // We decode chunked at pixels_per_block boundary
29  // Each block delivers the same amount of pixels.
30  decode_threaded_chunked(
31    width,
32    height,
33    pixels_per_block,
34    dummy,
35    &(|chunk, block_id| {
36      // Block offset
37      let src = &buf[block_id * V5_BLOCK_SIZE..block_id * V5_BLOCK_SIZE + V5_BLOCK_SIZE];
38      // Now swap the two parts of the block
39      let mut swapped = Vec::with_capacity(V5_BLOCK_SIZE);
40      swapped.extend_from_slice(&src[V5_SECTION_SPLIT_OFFSET..]);
41      swapped.extend_from_slice(&src[..V5_SECTION_SPLIT_OFFSET]);
42      // Transform the packets into final pixels
43      for (out, bytes) in chunk.chunks_exact_mut(pixels_per_packet).zip(swapped.chunks_exact(V5_BYTES_PER_PACKET)) {
44        // The packet is a bitstream in LSB order.
45        let mut pump = BitPumpLSB::new(bytes);
46        out.iter_mut().for_each(|p| *p = pump.get_bits(bps) as u16);
47      }
48    }),
49  )
50}
