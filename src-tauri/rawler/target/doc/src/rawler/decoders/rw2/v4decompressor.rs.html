3pub(crate) fn decode_panasonic_v4(buf: &[u8], width: usize, height: usize, split: bool, dummy: bool) -> PixU16 {
4  decode_threaded_multiline(
5    width,
6    height,
7    5,
8    dummy,
9    &(|out: &mut [u16], row| {
10      let skip = ((width * row * 9) + (width / 14 * 2 * row)) / 8;
11      let blocks = skip / 0x4000;
12      let src = &buf[blocks * 0x4000..];
13      let mut pump = BitPumpPanasonic::new(src, split);
14      for _ in 0..(skip % 0x4000) {
15        pump.get_bits(8);
16      }
17
18      let mut sh: i32 = 0;
19      for out in out.chunks_exact_mut(14) {
20        let mut pred: [i32; 2] = [0, 0];
21        let mut nonz: [i32; 2] = [0, 0];
22
23        for i in 0..14 {
24          if (i % 3) == 2 {
25            sh = 4 >> (3 - pump.get_bits(2));
26          }
27          if nonz[i & 1] != 0 {
28            let j = pump.get_bits(8) as i32;
29            if j != 0 {
30              pred[i & 1] -= 0x80 << sh;
31              if pred[i & 1] < 0 || sh == 4 {
32                pred[i & 1] &= !(-1 << sh);
33              }
34              pred[i & 1] += j << sh;
35            }
36          } else {
37            nonz[i & 1] = pump.get_bits(8) as i32;
38            if nonz[i & 1] != 0 || i > 11 {
39              pred[i & 1] = (nonz[i & 1] << 4) | (pump.get_bits(4) as i32);
40            }
41          }
42          out[i] = pred[i & 1] as u16;
43        }
44      }
45      Ok(())
46    }),
47  )
48  .expect("Failed to decode") // Decoder should never fail
49}

51pub struct BitPumpPanasonic<'a> {
52  buffer: &'a [u8],
53  pos: usize,
54  nbits: u32,
55  split: bool,
56}
57
58impl<'a> BitPumpPanasonic<'a> {
59  pub fn new(src: &'a [u8], split: bool) -> BitPumpPanasonic<'a> {
60    BitPumpPanasonic {
61      buffer: src,
62      pos: 0,
63      nbits: 0,
64      split,
65    }
66  }
67}

69impl<'a> BitPump for BitPumpPanasonic<'a> {
70  fn peek_bits(&mut self, num: u32) -> u32 {
71    if num > self.nbits {
72      self.nbits += 0x4000 * 8;
73      self.pos += 0x4000;
74    }
75    let mut byte = ((self.nbits - num) >> 3) ^ 0x3ff0;
76    if self.split {
77      byte = (byte + 0x4000 - 0x2008) % 0x4000;
78    }
79    let bits = LEu16(self.buffer, byte as usize + self.pos - 0x4000) as u32;
80    (bits >> ((self.nbits - num) & 7)) & (0x0ffffffffu32 >> (32 - num))
81  }
82
83  fn consume_bits(&mut self, num: u32) {
84    self.nbits -= num;
85  }
