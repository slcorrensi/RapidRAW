75impl Strip {
76  const fn line_height() -> usize {
77    6
78  }
79
80  // how many vertical lines does this block encode?
81  fn height(&self) -> u16 {
82    self.header.total_lines
83  }
84
85  // how many horizontal pixels does this block encode?
86  fn width(&self) -> usize {
87    // if this is not the last block, we are good.
88    if (self.n + 1) != (self.header.blocks_in_row as usize) {
89      return self.header.block_size as usize;
90    }
91    // ok, this is the last block...
92    debug_assert!(self.header.block_size as usize * self.header.blocks_in_row as usize >= self.header.raw_width as usize);
93    self.header.raw_width as usize - self.offset_x()
94  }
95
96  // where vertically does this block start?
97  fn offset_y(&self, line: usize) -> usize {
98    debug_assert!(line < (self.height() as usize));
99    Self::line_height() as usize * line
100  }
101
102  // where horizontally does this block start?
103  fn offset_x(&self) -> usize {
104    self.header.block_size as usize * self.n
105  }
106}

108fn div_round_up<T>(a: T, b: T) -> T
109where
110  T: std::ops::Add<Output = T>,
111  T: std::ops::Sub<Output = T>,
112  T: std::ops::Div<Output = T>,
113  T: From<u8>,
114  T: Copy,
115{
116  (a + b - T::from(1_u8)) / b
117}
118
119#[derive(Debug, Clone, Eq, PartialEq)]
120struct Header {
121  signature: u16,
122  lossless: u8,
123  raw_type: u8,
124  raw_bits: u8,
125  raw_height: u16,
126  raw_rounded_width: u16,
127  raw_width: u16,
128  block_size: u16,
129  blocks_in_row: u8,
130  total_lines: u16,
131}
132
133impl Header {
134  fn is_lossless(&self) -> bool {
135    self.lossless == 1
136  }
137
138  fn is_valid(&self) -> bool {
139    !(self.signature != 0x4953
140      //|| self.lossless != 1
141      || self.raw_height > 0x3000
142      || (self.raw_height as usize) < Strip::line_height()
143      || (self.raw_height as usize) % Strip::line_height() != 0
144      || self.raw_width > 0x3000
145      || self.raw_width < 0x300
146      || self.raw_width % 24 != 0
147      || self.raw_rounded_width > 0x3000
148      || self.block_size != 0x300
149      || self.raw_rounded_width < self.block_size
150      || self.raw_rounded_width % self.block_size != 0
151      || self.raw_rounded_width - self.raw_width >= self.block_size
152      || self.blocks_in_row > 0x10
153      || self.blocks_in_row == 0
154      || self.blocks_in_row as u16 != self.raw_rounded_width / self.block_size
155      || self.blocks_in_row as u16 != div_round_up(self.raw_width, self.block_size as u16)
156      || self.total_lines > 0x800
157      || self.total_lines == 0
158      || (self.total_lines as usize) != (self.raw_height as usize) / Strip::line_height()
159      || (self.raw_bits != 12 && self.raw_bits != 14 && self.raw_bits != 16)
160      || (self.raw_type != 16 && self.raw_type != 0))
161  }
162}

164impl Strip {
165  fn decompress_strip(&self, src: &PaddedBuf, header: &Header, params: &Params, q_bases: Option<&[u8]>, out: &mut PixU16) {
166    let mut info_block = CompressedBlock::new(header, params);
167    log::debug!("Fuji strip offset: {}, len: {}", self.offset, self.size);
168
169    let mut pump = if self.offset + self.size == src.len() {
170      BitPumpMSB::new(&src[self.offset..]) // use extra bytes from PaddedBuf
171    } else {
172      let extra_bytes = 16;
173      BitPumpMSB::new(&src[self.offset..self.offset + self.size + extra_bytes])
174    };
175
176    let mtable = [
177      (XT_LINE_R0, XT_LINE_R3),
178      (XT_LINE_R1, XT_LINE_R4),
179      (XT_LINE_G0, XT_LINE_G6),
180      (XT_LINE_G1, XT_LINE_G7),
181      (XT_LINE_B0, XT_LINE_B3),
182      (XT_LINE_B1, XT_LINE_B4),
183    ];
184
185    let ztable = [(XT_LINE_R2, 3), (XT_LINE_G2, 6), (XT_LINE_B2, 3)];
186
187    let mut params = params.clone();
188
189    for cur_line in 0..self.height() as usize {
190      debug_assert_eq!(header.is_lossless(), q_bases.is_none());
191      // init grads and main qtable
192      if !header.is_lossless() {
193        let q_base = q_bases.as_ref().unwrap()[cur_line] as i32;
194        if cur_line == 0 || q_base != params.qtables[0].q_base {
195          let max_value = (1 << header.raw_bits) - 1; // todo: put into header as function?
196          let main_qtable = Params::new_main_qtable(header, max_value, q_base);
197          params.qtables[0] = main_qtable;
198
199          // update grads
200          // total_values depends on q_base for QTable
201          let max_diff = 2.max((params.qtables[0].total_values + 0x20) >> 6);
202
203          for j in 0..3 {
204            debug_assert_eq!(info_block.grad_even[j].lossless_grads.len(), 41);
205            debug_assert_eq!(info_block.grad_odd[j].lossless_grads.len(), 41);
206            for i in 0..41 {
207              info_block.grad_even[j].lossless_grads[i].0 = max_diff;
208              info_block.grad_even[j].lossless_grads[i].1 = 1;
209              info_block.grad_odd[j].lossless_grads[i].0 = max_diff;
210              info_block.grad_odd[j].lossless_grads[i].1 = 1;
211            }
212          }
213        }
214      }
215
216      if header.raw_type == 16 {
217        info_block.fuji_xtrans_decode_block(&mut pump, &params);
218      } else {
219        info_block.fuji_bayer_decode_block(&mut pump, &params);
220      }
221
222      // copy data from line buffers and advance
223      for i in mtable.iter() {
224        debug_assert!(i.0 < i.1);
225        let (dest, src) = info_block.linebuf.split_at_mut(i.0 + 1);
226        dest[i.0].copy_from_slice(&src[i.1 - (i.0 + 1)]);
227        //info_block.linebuf[i.0] = info_block.linebuf[i.1].clone();
228      }
229
230      if header.raw_type == 16 {
231        info_block.copy_line_to_xtrans(self, cur_line, out);
232      } else {
233        info_block.copy_line_to_bayer(self, cur_line, out);
234      }
235
236      for i in ztable.iter() {
237        // Rest all lines
238        for line in i.0..i.0 + i.1 {
239          for p in info_block.linebuf[line].iter_mut() {
240            *p = 0;
241          }
242        }
243        // Initialize extra pixels
244        info_block.linebuf[i.0][0] = info_block.linebuf[i.0 - 1][1];
245        info_block.linebuf[i.0][params.line_width + 1] = info_block.linebuf[i.0 - 1][params.line_width];
246      }
247    }
248  }
249}

251/// We need PaddedBuf here, because the buffer is divided
252/// into multiple strips and each strip is feed into a BitPump.
253/// Each pump need as litte bit more overhead at the end.
254/// For the final strip, we need the extra bytes from PaddedBuf
255/// to prevent out-of-range errors in BitPump.
256pub(super) fn decompress_fuji(buf: &PaddedBuf, width: usize, height: usize, _bps: usize, corrected_cfa: &CFA) -> Result<PixU16> {
257  let mut stream = ByteStream::new(buf, Endian::Big);
258  let header = Header {
259    signature: stream.get_u16(),
260    lossless: stream.get_u8(),
261    raw_type: stream.get_u8(),
262    raw_bits: stream.get_u8(),
263    raw_height: stream.get_u16(),
264    raw_rounded_width: stream.get_u16(),
265    raw_width: stream.get_u16(),
266    block_size: stream.get_u16(),
267    blocks_in_row: stream.get_u8(),
268    total_lines: stream.get_u16(),
269  };
270  log::debug!("Header: {:?}", header);
271  if !header.is_valid() {
272    return Err("Fuji header is not valid".into());
273  }
274  assert_eq!(
275    Dim2::new(width, height),
276    Dim2::new(header.raw_width.into(), header.raw_height.into()),
277    "RAF header specifies different dimensions!"
278  );
279
280  let params = Params::new(&header);
281  log::debug!("Params: {:?}", params);
282
283  let mut cfa: [[CFAColor; 6]; 6] = Default::default();
284  for i in 0..6 {
285    for j in 0..6 {
286      let color = corrected_cfa.cfa_color_at(i, j);
287      match color {
288        CFAColor::RED | CFAColor::GREEN | CFAColor::BLUE => cfa[i][j] = color,
289        _ => panic!("Got unexpected color: {:?}", color),
290      }
291    }
292  }
293
294  let block_sizes: Vec<usize> = (0..header.blocks_in_row).map(|_| stream.get_u32() as usize).collect();
295  let raw_offset = header.blocks_in_row as usize * size_of::<u32>();
296  let raw_offset_padded = (raw_offset + 0xF) & !0xF;
297  stream.consume_bytes(raw_offset_padded - raw_offset);
298
299  // Global Q bases for all strips
300  let q_bases: Option<Vec<u8>> = if !header.is_lossless() {
301    let total_q_bases = block_sizes.len() * ((header.total_lines as usize + 0xF) & !0xF);
302    Some(stream.get_bytes(total_q_bases))
303  } else {
304    None
305  };
306
307  //eprintln!("q_bases: {:?}", q_bases);
308  //eprintln!("First block: {}", stream.get_pos());
309
310  // calculating raw block offsets
311  let strips: Vec<Strip> = block_sizes
312    .iter()
313    .enumerate()
314    .map(|(n, &block_size)| {
315      let strip = Strip {
316        offset: stream.get_pos(),
317        size: block_size,
318        n,
319        header: header.clone(),
320        cfa,
321      };
322      stream.consume_bytes(block_size);
323      strip
324    })
325    .collect();
326
327  assert!(stream.remaining_bytes() <= 16);
328
329  let out = SharedPix2D::new(PixU16::new(width, height));
330
331  // Process each strip
332  strips.par_iter().for_each(|strip| {
333    let line_step = (header.total_lines as usize + 0xF) & !0xF;
334    // Each strip has it's own q_bases
335    let q_bases_strip = q_bases.as_ref().map(|buf| &buf[strip.n * line_step..]);
336    // DANGEROUS: We need multiple mut refs here. This should be
337    // safe as be only write pixels to pre-allocated memory.
338    let outbuf = unsafe { out.inner_mut() };
339    strip.decompress_strip(buf, &header, &params, q_bases_strip, outbuf);
340  });
341
342  Ok(out.into_inner())
343}

393/// A compressed block
394struct CompressedBlock {
395  // tables of gradients
396  grad_even: [GradientList; 3],
397  grad_odd: [GradientList; 3],
398  linebuf: Vec<Vec<u16>>,
399}
400
401impl CompressedBlock {
402  /// Create and initialize new compression block.
403  fn new(header: &Header, params: &Params) -> Self {
404    let linebuf = vec![vec![0; params.line_width + 2]; XT_LINE_TOTAL];
405
406    let mut grad_even: [GradientList; 3] = Default::default();
407    let mut grad_odd: [GradientList; 3] = Default::default();
408
409    if header.is_lossless() {
410      let max_diff = 2.max((params.qtables[0].total_values + 0x20) >> 6) as i32;
411      for j in 0..3 {
412        debug_assert_eq!(grad_even[j].lossless_grads.len(), 41);
413        debug_assert_eq!(grad_odd[j].lossless_grads.len(), 41);
414        for i in 0..41 {
415          grad_even[j].lossless_grads[i].0 = max_diff;
416          grad_even[j].lossless_grads[i].1 = 1;
417          grad_odd[j].lossless_grads[i].0 = max_diff;
418          grad_odd[j].lossless_grads[i].1 = 1;
419        }
420      }
421    } else {
422      // init static grads for lossy only - main ones are done per line
423      for k in 0..3 {
424        let max_diff = 2.max((params.qtables[k + 1].total_values + 0x20) >> 6) as i32;
425
426        for j in 0..3 {
427          for i in 0..5 {
428            grad_even[j].lossy_grads[k][i].0 = max_diff;
429            grad_even[j].lossy_grads[k][i].1 = 1;
430            grad_odd[j].lossy_grads[k][i].0 = max_diff;
431            grad_odd[j].lossy_grads[k][i].1 = 1;
432          }
433        }
434      }
435    }
436
437    Self { grad_even, grad_odd, linebuf }
438  }
439
440  /// Copy line from decoding buffer to output
441  fn copy_line<F>(&self, strip: &Strip, cur_line: usize, index_f: F, out: &mut PixU16)
442  where
443    F: Fn(usize) -> usize,
444  {
445    let mut line_buf_b = [0; 3];
446    let mut line_buf_g = [0; 6];
447    let mut line_buf_r = [0; 3];
448
449    for i in 0..3 {
450      line_buf_r[i] = XT_LINE_R2 + i;
451      line_buf_b[i] = XT_LINE_B2 + i;
452    }
453    for i in 0..6 {
454      line_buf_g[i] = XT_LINE_G2 + i;
455    }
456    for row_count in 0..Strip::line_height() {
457      for pixel_count in 0..strip.width() {
458        let line_idx = match strip.cfa[row_count][pixel_count % 6] {
459          CFAColor::RED => line_buf_r[row_count >> 1],
460          CFAColor::GREEN => line_buf_g[row_count],
461          CFAColor::BLUE => line_buf_b[row_count >> 1],
462          _ => unreachable!(),
463        };
464        let p = self.linebuf[line_idx][1 + index_f(pixel_count)];
465        *out.at_mut(strip.offset_y(cur_line) + row_count, strip.offset_x() + pixel_count) = p;
466      }
467    }
468  }

470  /// Copy line by Bayer pattern
471  fn copy_line_to_bayer(&self, strip: &Strip, cur_line: usize, out: &mut PixU16) {
472    let index = |pixel_count: usize| -> usize { pixel_count >> 1 };
473    self.copy_line(strip, cur_line, index, out);
474  }
475
476  /// Copy line by X-Trans pattern
477  fn copy_line_to_xtrans(&self, strip: &Strip, cur_line: usize, out: &mut PixU16) {
478    let index = |pixel_count: usize| -> usize { (((pixel_count * 2 / 3) & 0x7FFFFFFE) | ((pixel_count % 3) & 1)) + ((pixel_count % 3) >> 1) };
479    self.copy_line(strip, cur_line, index, out);
480  }

482  /// Decode Bayer pattern (RGGB and the like) from block
483  fn fuji_bayer_decode_block(&mut self, pump: &mut BitPumpMSB, params: &Params) {
484    let line_width = params.line_width;
485    let mut colors = Colors::new();
486
487    let pass_red_green = |colors: &mut Colors, pump: &mut BitPumpMSB, block: &mut CompressedBlock, c0: usize, c1: usize, grad: usize| {
488      while colors.g().even < line_width || colors.g().odd < line_width {
489        if colors.g().even < line_width {
490          fuji_decode_sample_even(pump, params, &mut block.linebuf, c0, &mut colors.r().even, &mut block.grad_even[grad]);
491          fuji_decode_sample_even(pump, params, &mut block.linebuf, c1, &mut colors.g().even, &mut block.grad_even[grad]);
492        }
493        if colors.g().even > 8 {
494          fuji_decode_sample_odd(pump, params, &mut block.linebuf, c0, &mut colors.r().odd, &mut block.grad_odd[grad]);
495          fuji_decode_sample_odd(pump, params, &mut block.linebuf, c1, &mut colors.g().odd, &mut block.grad_odd[grad]);
496        }
497      }
498      block.fuji_extend_red(line_width);
499      block.fuji_extend_green(line_width);
500    };
501
502    let pass_green_blue = |colors: &mut Colors, pump: &mut BitPumpMSB, block: &mut CompressedBlock, c0: usize, c1: usize, grad: usize| {
503      while colors.g().even < line_width || colors.g().odd < line_width {
504        if colors.g().even < line_width {
505          fuji_decode_sample_even(pump, params, &mut block.linebuf, c0, &mut colors.g().even, &mut block.grad_even[grad]);
506          fuji_decode_sample_even(pump, params, &mut block.linebuf, c1, &mut colors.b().even, &mut block.grad_even[grad]);
507        }
508        if colors.g().even > 8 {
509          fuji_decode_sample_odd(pump, params, &mut block.linebuf, c0, &mut colors.g().odd, &mut block.grad_odd[grad]);
510          fuji_decode_sample_odd(pump, params, &mut block.linebuf, c1, &mut colors.b().odd, &mut block.grad_odd[grad]);
511        }
512      }
513      block.fuji_extend_green(line_width);
514      block.fuji_extend_blue(line_width);
515    };
516
517    pass_red_green(&mut colors, pump, self, XT_LINE_R2, XT_LINE_G2, 0);
518    colors.g().reset();
519
520    pass_green_blue(&mut colors, pump, self, XT_LINE_G3, XT_LINE_B2, 1);
521    colors.r().reset();
522    colors.g().reset();
523
524    pass_red_green(&mut colors, pump, self, XT_LINE_R3, XT_LINE_G4, 2);
525    colors.g().reset();
526    colors.b().reset();
527
528    pass_green_blue(&mut colors, pump, self, XT_LINE_G5, XT_LINE_B3, 0);
529    colors.r().reset();
530    colors.g().reset();
531
532    pass_red_green(&mut colors, pump, self, XT_LINE_R4, XT_LINE_G6, 1);
533    colors.g().reset();
534    colors.b().reset();
535
536    pass_green_blue(&mut colors, pump, self, XT_LINE_G7, XT_LINE_B4, 2);
537  }

539  /// A single X-Trans decoding pass for the given control colors C0 and C1
540  fn fuji_xtrans_pass<F, const C0: usize, const C1: usize>(
541    &mut self,
542    params: &Params,
543    colors: &mut Colors,
544    pump: &mut BitPumpMSB,
545    c0: usize,
546    c1: usize,
547    grad: usize,
548    even_func: F,
549  ) where
550    F: Fn(&mut CompressedBlock, &mut BitPumpMSB, usize, usize, usize, &mut ColorPos, &mut ColorPos),
551  {
552    let line_width = params.line_width;
553    while colors.g().even < line_width || colors.g().odd < line_width {
554      if colors.g().even < line_width {
555        let mut c0_pos = *colors.at(C0);
556        let mut c1_pos = *colors.at(C1);
557        even_func(self, pump, c0, c1, grad, &mut c0_pos, &mut c1_pos);
558        *colors.at(C0) = c0_pos; // Write back
559        *colors.at(C1) = c1_pos;
560      }
561      if colors.g().even > 8 {
562        fuji_decode_sample_odd(pump, params, &mut self.linebuf, c0, &mut colors.at(C0).odd, &mut self.grad_odd[grad]);
563        fuji_decode_sample_odd(pump, params, &mut self.linebuf, c1, &mut colors.at(C1).odd, &mut self.grad_odd[grad]);
564      }
565    }
566  }

568  /// Decode X-Trans pattern from block
569  fn fuji_xtrans_decode_block(&mut self, pump: &mut BitPumpMSB, params: &Params) {
570    let mut colors = Colors::new();
571    let line_width = params.line_width;
572
573    // Pass 1
574    self.fuji_xtrans_pass::<_, { Colors::R }, { Colors::G }>(
575      params,
576      &mut colors,
577      pump,
578      XT_LINE_R2,
579      XT_LINE_G2,
580      0,
581      |block, pump, c0, c1, grad, c0_pos, c1_pos| {
582        fuji_decode_interpolation_even(block, c0, &mut c0_pos.even);
583        fuji_decode_sample_even(pump, params, &mut block.linebuf, c1, &mut c1_pos.even, &mut block.grad_even[grad]);
584      },
585    );
586    self.fuji_extend_red(line_width);
587    self.fuji_extend_green(line_width);
588    colors.g().reset();
589
590    // Pass 2
591    self.fuji_xtrans_pass::<_, { Colors::G }, { Colors::B }>(
592      params,
593      &mut colors,
594      pump,
595      XT_LINE_G3,
596      XT_LINE_B2,
597      1,
598      |block, pump, c0, c1, grad, c0_pos, c1_pos| {
599        fuji_decode_sample_even(pump, params, &mut block.linebuf, c0, &mut c0_pos.even, &mut block.grad_even[grad]);
600        fuji_decode_interpolation_even(block, c1, &mut c1_pos.even);
601      },
602    );
603    self.fuji_extend_green(line_width);
604    self.fuji_extend_blue(line_width);
605    colors.r().reset();
606    colors.g().reset();
607
608    // Pass 3
609    self.fuji_xtrans_pass::<_, { Colors::R }, { Colors::G }>(
610      params,
611      &mut colors,
612      pump,
613      XT_LINE_R3,
614      XT_LINE_G4,
615      2,
616      |block, pump, c0, c1, grad, c0_pos, c1_pos| {
617        if c0_pos.even & 3 != 0 {
618          fuji_decode_sample_even(pump, params, &mut block.linebuf, c0, &mut c0_pos.even, &mut block.grad_even[grad]);
619        } else {
620          fuji_decode_interpolation_even(block, c0, &mut c0_pos.even);
621        }
622        fuji_decode_interpolation_even(block, c1, &mut c1_pos.even);
623      },
624    );
625    self.fuji_extend_red(line_width);
626    self.fuji_extend_green(line_width);
627    colors.g().reset();
628    colors.b().reset();
629
630    // Pass 4
631    self.fuji_xtrans_pass::<_, { Colors::G }, { Colors::B }>(
632      params,
633      &mut colors,
634      pump,
635      XT_LINE_G5,
636      XT_LINE_B3,
637      0,
638      |block, pump, c0, c1, grad, c0_pos, c1_pos| {
639        fuji_decode_sample_even(pump, params, &mut block.linebuf, c0, &mut c0_pos.even, &mut block.grad_even[grad]);
640
641        if c1_pos.even & 3 == 2 {
642          fuji_decode_interpolation_even(block, c1, &mut c1_pos.even);
643        } else {
644          fuji_decode_sample_even(pump, params, &mut block.linebuf, c1, &mut c1_pos.even, &mut block.grad_even[grad]);
645        }
646      },
647    );
648    self.fuji_extend_green(line_width);
649    self.fuji_extend_blue(line_width);
650    colors.r().reset();
651    colors.g().reset();
652
653    // Pass 5
654    self.fuji_xtrans_pass::<_, { Colors::R }, { Colors::G }>(
655      params,
656      &mut colors,
657      pump,
658      XT_LINE_R4,
659      XT_LINE_G6,
660      1,
661      |block, pump, c0, c1, grad, c0_pos, c1_pos| {
662        if c0_pos.even & 3 == 2 {
663          fuji_decode_interpolation_even(block, c0, &mut c0_pos.even);
664        } else {
665          fuji_decode_sample_even(pump, params, &mut block.linebuf, c0, &mut c0_pos.even, &mut block.grad_even[grad]);
666        }
667
668        fuji_decode_sample_even(pump, params, &mut block.linebuf, c1, &mut c1_pos.even, &mut block.grad_even[grad]);
669      },
670    );
671    self.fuji_extend_red(line_width);
672    self.fuji_extend_green(line_width);
673    colors.g().reset();
674    colors.b().reset();
675
676    // Pass 6
677    self.fuji_xtrans_pass::<_, { Colors::G }, { Colors::B }>(
678      params,
679      &mut colors,
680      pump,
681      XT_LINE_G7,
682      XT_LINE_B4,
683      2,
684      |block, pump, c0, c1, grad, c0_pos, c1_pos| {
685        fuji_decode_interpolation_even(block, c0, &mut c0_pos.even);
686
687        if c1_pos.even & 3 != 0 {
688          fuji_decode_sample_even(pump, params, &mut block.linebuf, c1, &mut c1_pos.even, &mut block.grad_even[grad]);
689        } else {
690          fuji_decode_interpolation_even(block, c1, &mut c1_pos.even);
691        }
692      },
693    );
694    self.fuji_extend_green(line_width);
695    self.fuji_extend_blue(line_width);
696  }
697
698  fn fuji_extend_generic(&mut self, line_width: usize, start: usize, end: usize) {
699    debug_assert!(start > 0);
700    for i in start..=end {
701      self.linebuf[i][0] = self.linebuf[i - 1][1];
702      self.linebuf[i][line_width + 1] = self.linebuf[i - 1][line_width];
703    }
704  }
705
706  fn fuji_extend_red(&mut self, line_width: usize) {
707    self.fuji_extend_generic(line_width, XT_LINE_R2, XT_LINE_R4)
708  }
709
710  fn fuji_extend_green(&mut self, line_width: usize) {
711    self.fuji_extend_generic(line_width, XT_LINE_G2, XT_LINE_G7)
712  }
713
714  fn fuji_extend_blue(&mut self, line_width: usize) {
715    self.fuji_extend_generic(line_width, XT_LINE_B2, XT_LINE_B4)
716  }

728impl QTable {
729  /// Lookup gradient in q_table. The absolute value of this
730  /// is used as an index into the gradients vector.
731  fn lookup_gradient(&self, params: &Params, v1: i32, v2: i32) -> i32 {
732    self.q_gradient_multi * self.q_table[(params.max_value + (v1)) as usize] + self.q_table[(params.max_value + (v2)) as usize]
733  }
734
735  /// Build a quantization table based on 5 quantization points.
736  fn build_table(header: &Header, qp: &[i32; 5]) -> Vec<i32> {
737    let mut qtable = vec![0; 2 * (1 << (header.raw_bits as usize))];
738    let mut cur_val = -(qp[4] as i32);
739
740    for qt in qtable.iter_mut() {
741      if cur_val > qp[4] as i32 {
742        break;
743      }
744      *qt = match cur_val {
745        x if x <= -qp[3] => -4,
746        x if x <= -qp[2] => -3,
747        x if x <= -qp[1] => -2,
748        x if x < -qp[0] => -1,
749        x if x <= qp[0] => 0,
750        x if x < qp[1] => 1,
751        x if x < qp[2] => 2,
752        x if x < qp[3] => 3,
753        _ => 4,
754      };
755      cur_val += 1;
756    }
757    qtable
758  }
759}

761impl Params {
762  /// Construct new main quantization table.
763  fn new_main_qtable(header: &Header, max_value: i32, q_base: i32) -> QTable {
764    let mut qp = [0; 5];
765    qp[0] = q_base;
766    qp[1] = 3 * q_base + 0x12;
767    qp[2] = 5 * q_base + 0x43;
768    qp[3] = 7 * q_base + 0x114;
769    qp[4] = max_value;
770
771    let max_val = max_value + 1;
772    if qp[1] >= max_val || qp[1] < q_base + 1 {
773      qp[1] = q_base + 1;
774    }
775    if qp[2] < qp[1] || qp[2] >= max_val {
776      qp[2] = qp[1];
777    }
778    if qp[3] < qp[2] || qp[3] >= max_val {
779      qp[3] = qp[2];
780    }
781
782    let q_table = QTable::build_table(header, &qp);
783    let total_values = (qp[4] + 2 * q_base) / (2 * q_base + 1) + 1;
784    let raw_bits: usize = log2ceil(total_values as usize);
785
786    QTable {
787      q_base,
788      q_table,
789      q_gradient_multi: 9,
790      max_grad: 0,
791      raw_bits,
792      total_values,
793    }
794  }
795
796  /// Create new parameter
797  fn new(header: &Header) -> Self {
798    if (header.block_size % 3 != 0 && header.raw_type == 16) || (header.block_size & 1 != 0 && header.raw_type == 0) {
799      panic!("Invalid FUJI header");
800    }
801    let min_value = 0x40;
802    let max_value = ((1 << header.raw_bits) - 1) as i32;
803    let max_bits: usize = 4 * log2ceil(max_value as usize + 1);
804    let line_width = if header.raw_type == 16 {
805      (header.block_size as usize * 2) / 3
806    } else {
807      header.block_size as usize >> 1
808    };
809
810    // Build quantization tables.
811    // For lossless, only one table is required.
812    // For lossy, the main table is created on each iteration
813    // while 3 static extra tables are required.
814    let qtables = if header.is_lossless() {
815      // Only a single table is needed for lossless mode
816      let q_base = 0;
817      let main_qtable = Self::new_main_qtable(header, max_value, q_base);
818      vec![main_qtable]
819    } else {
820      let mut qtables = vec![QTable::default(); 4];
821
822      // The main table is left uninitialized here as
823      // the table is setup for each iteration.
824      qtables[0].q_base = -1;
825
826      let mut qp = [0_i32; 5];
827      qp[4] = max_value; // identical for all tables
828
829      // table 0
830      qtables[1].q_base = 0;
831      qtables[1].max_grad = 5;
832      qtables[1].q_gradient_multi = 3;
833      qtables[1].total_values = qp[4] + 1;
834      qtables[1].raw_bits = log2ceil(qtables[1].total_values as usize);
835      qp[0] = qtables[1].q_base;
836      qp[1] = if qp[4] >= 0x12 { 0x12 } else { qp[0] + 1 };
837      qp[2] = if qp[4] >= 0x43 { 0x43 } else { qp[1] };
838      qp[3] = if qp[4] >= 0x114 { 0x114 } else { qp[2] };
839      qtables[1].q_table = QTable::build_table(header, &qp);
840
841      // table 1
842      qtables[2].q_base = 1;
843      qtables[2].max_grad = 6;
844      qtables[2].q_gradient_multi = 3;
845      qtables[2].total_values = (qp[4] + 2) / 3 + 1;
846      qtables[2].raw_bits = log2ceil(qtables[2].total_values as usize);
847      qp[0] = qtables[2].q_base;
848      qp[1] = if qp[4] >= 0x15 { 0x15 } else { qp[0] + 1 };
849      qp[2] = if qp[4] >= 0x48 { 0x48 } else { qp[1] };
850      qp[3] = if qp[4] >= 0x11B { 0x11B } else { qp[2] };
851      qtables[2].q_table = QTable::build_table(header, &qp);
852
853      // table 2
854      qtables[3].q_base = 2;
855      qtables[3].max_grad = 7;
856      qtables[3].q_gradient_multi = 3;
857      qtables[3].total_values = (qp[4] + 4) / 5 + 1;
858      qtables[3].raw_bits = log2ceil(qtables[3].total_values as usize);
859      qp[0] = qtables[3].q_base;
860      qp[1] = if qp[4] >= 0x18 { 0x18 } else { qp[0] + 1 };
861      qp[2] = if qp[4] >= 0x4D { 0x4D } else { qp[1] };
862      qp[3] = if qp[4] >= 0x122 { 0x122 } else { qp[2] };
863      qtables[3].q_table = QTable::build_table(header, &qp);
864
865      qtables
866    };
867
868    Self {
869      qtables,
870      max_bits,
871      min_value,
872      max_value,
873      line_width,
874    }
875  }
876}

878/// Count and consume all zero bits
879/// Additionally, consume the first 1 bit.
880#[inline(always)]
881fn fuji_zerobits(pump: &mut BitPumpMSB) -> u32 {
882  let count = pump.consume_zerobits();
883  debug_assert_eq!(pump.peek_bits(1), 1);
884  pump.consume_bits(1); // consume the next bit which is 0b1
885  count
886}
887
888/// Calculate bit difference between two values
889fn bit_diff(v1: i32, v2: i32) -> u32 {
890  if v2 >= v1 {
891    0
892  } else {
893    let mut dec_bits = 0;
894    while dec_bits <= 14 {
895      dec_bits += 1;
896      if (v2 << dec_bits) >= v1 {
897        return dec_bits;
898      }
899    }
900    dec_bits
901  }
902}

904/// Read a single code from bitstream and ajust gradient.
905/// We use bmi1 feature here because it provides LZCNT for
906/// leading zero count which is used here a lot.
907#[multiversion(targets("x86_64+avx+avx2+fma+bmi1+bmi2", "x86_64+avx+avx2", "x86+sse", "aarch64+neon"))]
908fn read_code(pump: &mut BitPumpMSB, params: &Params, gradient: &mut Gradient, q_table: &QTable) -> i32 {
909  let sample = fuji_zerobits(pump);
910  let mut code: i32 = if (sample as usize) < params.max_bits - q_table.raw_bits - 1 {
911    let dec_bits = bit_diff(gradient.0, gradient.1);
912    let extra_bits = if dec_bits == 0 { 0 } else { pump.get_bits(dec_bits) as i32 };
913    (sample << dec_bits) as i32 + extra_bits
914  } else {
915    1 + pump.get_bits(q_table.raw_bits as u32) as i32
916  };
917  // Validate code
918  if code < 0 || code >= q_table.total_values as i32 {
919    panic!("Invalid code: {}", code);
920  }
921  // Adjust code
922  if (code & 1) != 0 {
923    code = -1 - code / 2;
924  } else {
925    code /= 2;
926  }
927  // Update gradient
928  gradient.0 += code.abs();
929  if gradient.1 == params.min_value {
930    gradient.0 >>= 1;
931    gradient.1 >>= 1;
932  }
933  gradient.1 += 1;
934  code
935}

937/// Decode samples for even positions
938fn fuji_decode_sample_even(pump: &mut BitPumpMSB, params: &Params, linebuf: &mut [Vec<u16>], line: usize, pos: &mut usize, grads: &mut GradientList) {
939  // Line -2 |   | f |   |
940  // Line -1 | c | b | d |
941  // Line  0 | a | x | g |
942  let rb = linebuf[line - 1][1 + *pos + 0] as i32;
943  let rc = linebuf[line - 1][1 + *pos - 1] as i32;
944  let rd = linebuf[line - 1][1 + *pos + 1] as i32;
945  let rf = linebuf[line - 2][1 + *pos + 0] as i32;
946  // Calculate horiz/vert. gradients around current sample x
947  let diff_rc_rb = (rc - rb).abs();
948  let diff_rf_rb = (rf - rb).abs();
949  let diff_rd_rb = (rd - rb).abs();
950  // Quantization table and Gradients to use
951  let mut qtable = &params.qtables[0];
952  let mut gradients = &mut grads.lossless_grads;
953  for i in 1..4 {
954    if params.qtables[0].q_base < i as i32 {
955      break;
956    }
957    if diff_rf_rb + diff_rc_rb <= params.qtables[i].max_grad as i32 {
958      qtable = &params.qtables[i];
959      gradients = &mut grads.lossy_grads[i - 1];
960      break;
961    }
962  }
963  // Determine gradient
964  let grad = qtable.lookup_gradient(params, rb - rf, rc - rb);
965
966  let mut interp_val = if diff_rc_rb > diff_rf_rb && diff_rc_rb > diff_rd_rb {
967    rf + rd + 2 * rb
968  } else if diff_rd_rb > diff_rc_rb && diff_rd_rb > diff_rf_rb {
969    rf + rc + 2 * rb
970  } else {
971    rd + rc + 2 * rb
972  };
973
974  let code = read_code(pump, params, &mut gradients[grad.unsigned_abs() as usize], qtable);
975
976  // Adjustments specific to even positions
977  if grad < 0 {
978    interp_val = (interp_val >> 2) - code * (2 * qtable.q_base as i32 + 1);
979  } else {
980    interp_val = (interp_val >> 2) + code * (2 * qtable.q_base as i32 + 1);
981  };
982
983  // Generic adjustments
984  if interp_val < -(qtable.q_base as i32) {
985    interp_val += (qtable.total_values * (2 * qtable.q_base + 1)) as i32;
986  } else if interp_val > qtable.q_base as i32 + params.max_value {
987    interp_val -= (qtable.total_values * (2 * qtable.q_base + 1)) as i32;
988  }
989
990  if interp_val >= 0 {
991    linebuf[line][1 + *pos] = interp_val.min(params.max_value) as u16
992  } else {
993    linebuf[line][1 + *pos] = 0;
994  }
995
996  *pos += 2;
997}

999/// Decode samples for odd positions
1000fn fuji_decode_sample_odd(pump: &mut BitPumpMSB, params: &Params, linebuf: &mut [Vec<u16>], line: usize, pos: &mut usize, grads: &mut GradientList) {
1001  // Line -2 |   | f |   |
1002  // Line -1 | c | b | d |
1003  // Line  0 | a | x | g |
1004  let ra = linebuf[line + 0][1 + *pos - 1] as i32;
1005  let rb = linebuf[line - 1][1 + *pos + 0] as i32;
1006  let rc = linebuf[line - 1][1 + *pos - 1] as i32;
1007  let rd = linebuf[line - 1][1 + *pos + 1] as i32;
1008  let rg = linebuf[line + 0][1 + *pos + 1] as i32;
1009  // Calculate horiz/vert. gradients around current sample x
1010  let diff_rc_ra = (rc - ra).abs();
1011  let diff_rb_rc = (rb - rc).abs();
1012  // Quantization table and Gradients to use
1013  let mut qtable = &params.qtables[0];
1014  let mut gradients = &mut grads.lossless_grads;
1015  for i in 1..4 {
1016    if params.qtables[0].q_base < i as i32 {
1017      break;
1018    }
1019    if diff_rb_rc + diff_rc_ra <= params.qtables[i].max_grad as i32 {
1020      qtable = &params.qtables[i];
1021      gradients = &mut grads.lossy_grads[i - 1];
1022      break;
1023    }
1024  }
1025  // Determine gradient
1026  let grad = qtable.lookup_gradient(params, rb - rc, rc - ra);
1027
1028  let mut interp_val = if (rb > rc && rb > rd) || (rb < rc && rb < rd) {
1029    (rg + ra + 2 * rb) >> 2
1030  } else {
1031    (ra + rg) >> 1
1032  };
1033
1034  let code = read_code(pump, params, &mut gradients[grad.unsigned_abs() as usize], qtable);
1035
1036  // Adjustments specific to odd positions
1037  if grad < 0 {
1038    interp_val -= code * (2 * qtable.q_base as i32 + 1);
1039  } else {
1040    interp_val += code * (2 * qtable.q_base as i32 + 1);
1041  }
1042
1043  // Generic adjustments
1044  if interp_val < -(qtable.q_base as i32) {
1045    interp_val += (qtable.total_values * (2 * qtable.q_base + 1)) as i32;
1046  } else if interp_val > qtable.q_base as i32 + params.max_value {
1047    interp_val -= (qtable.total_values * (2 * qtable.q_base + 1)) as i32;
1048  }
1049
1050  if interp_val >= 0 {
1051    linebuf[line][1 + *pos] = interp_val.min(params.max_value) as u16
1052  } else {
1053    linebuf[line][1 + *pos] = 0;
1054  }
1055
1056  *pos += 2;
1057}

1059/// Interpolate x value from surrounding pixels
1060fn fuji_decode_interpolation_even(block: &mut CompressedBlock, line: usize, pos: &mut usize) {
1061  // Line -2 |   | f |   |
1062  // Line -1 | c | b | d |
1063  // Line  0 | a | x | g |
1064  let rb = block.linebuf[line - 1][1 + *pos + 0] as i32;
1065  let rc = block.linebuf[line - 1][1 + *pos - 1] as i32;
1066  let rd = block.linebuf[line - 1][1 + *pos + 1] as i32;
1067  let rf = block.linebuf[line - 2][1 + *pos + 0] as i32;
1068
1069  let x = &mut block.linebuf[line][1 + *pos];
1070
1071  let diff_rc_rb = (rc - rb).abs();
1072  let diff_rf_rb = (rf - rb).abs();
1073  let diff_rd_rb = (rd - rb).abs();
1074
1075  if diff_rc_rb > diff_rf_rb && diff_rc_rb > diff_rd_rb {
1076    *x = ((rf + rd + 2 * rb) >> 2) as u16;
1077  } else if diff_rd_rb > diff_rc_rb && diff_rd_rb > diff_rf_rb {
1078    *x = ((rf + rc + 2 * rb) >> 2) as u16;
1079  } else {
1080    *x = ((rd + rc + 2 * rb) >> 2) as u16;
1081  }
1082
1083  *pos += 2;
1084}
