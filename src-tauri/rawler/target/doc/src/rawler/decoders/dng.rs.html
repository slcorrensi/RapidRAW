48  fn raw_image(&self, file: &RawSource, _params: &RawDecodeParams, dummy: bool) -> Result<RawImage> {
49    let raw = self.get_raw_ifd()?;
50    let width = fetch_tiff_tag!(raw, TiffCommonTag::ImageWidth).force_usize(0);
51    let height = fetch_tiff_tag!(raw, TiffCommonTag::ImageLength).force_usize(0);
52    let cpp = fetch_tiff_tag!(raw, TiffCommonTag::SamplesPerPixel).force_usize(0);
53    let bits = fetch_tiff_tag!(raw, TiffCommonTag::BitsPerSample).force_u32(0);
54
55    let mut image = match fetch_tiff_tag!(raw, TiffCommonTag::Compression).force_u32(0) {
56      1 => self.decode_uncompressed(file, raw, width * cpp, height, dummy)?,
57      7 => self.decode_compressed(file, raw, width * cpp, height, cpp, dummy)?,
58      c => return Err(RawlerError::DecoderFailed(format!("Don't know how to read DNGs with compression {}", c))),
59    };
60
61    if let Some(lintable) = raw.get_entry(TiffCommonTag::Linearization) {
62      if bits != 8 && fetch_tiff_tag!(raw, TiffCommonTag::Compression).force_u32(0) != 1 {
63        // 8 bit uncompressed data is already read delinearized
64        apply_linearization(&mut image, &lintable.value, bits);
65      }
66    }
67
68    let orientation = Orientation::from_tiff(self.tiff.root_ifd());
69
70    let mut cam = self.make_camera(raw, width, height)?;
71    // If we know the camera, re-use the clean names
72    if let Ok(known_cam) = self.rawloader.check_supported(self.tiff.root_ifd()) {
73      cam.clean_make = known_cam.clean_make;
74      cam.clean_model = known_cam.clean_model;
75    }
76
77    let blacklevel = self.get_blacklevels(raw)?;
78    let whitelevel = self.get_whitelevels(raw)?.or(Some(WhiteLevel::new_bits(bits, cpp)));
79
80    let photometric = match fetch_tiff_tag!(raw, TiffCommonTag::PhotometricInt).force_u32(0) {
81      1 => RawPhotometricInterpretation::BlackIsZero,
82      32803 => RawPhotometricInterpretation::Cfa(CFAConfig::new_from_camera(&cam)),
83      34892 => RawPhotometricInterpretation::LinearRaw,
84      _ => todo!(),
85    };
86
87    let mut image = RawImage::new(cam, image, cpp, self.get_wb()?, photometric, blacklevel, whitelevel, dummy);
88    image.orientation = orientation;
89
90    Ok(image)
91  }

112  fn thumbnail_image(&self, file: &RawSource, _params: &RawDecodeParams) -> Result<Option<DynamicImage>> {
113    if let Some(thumb_ifd) = Some(self.tiff.root_ifd()).filter(|ifd| ifd.get_entry(TiffCommonTag::NewSubFileType).map(|entry| entry.force_u16(0)) == Some(1)) {
114      let buf = thumb_ifd
115        .strip_data(&mut file.reader())
116        .map_err(|e| RawlerError::DecoderFailed(format!("Failed to get strip data: {}", e)))?
117        .into_iter()
118        .flatten()
119        .collect();
120      let compression = thumb_ifd.get_entry(TiffCommonTag::Compression).ok_or("Missing tag")?.force_usize(0);
121      let width = fetch_tiff_tag!(thumb_ifd, TiffCommonTag::ImageWidth).force_usize(0);
122      let height = fetch_tiff_tag!(thumb_ifd, TiffCommonTag::ImageLength).force_usize(0);
123      if compression == 1 {
124        return Ok(Some(DynamicImage::ImageRgb8(
125          ImageBuffer::<Rgb<u8>, Vec<u8>>::from_raw(width as u32, height as u32, buf).unwrap(), // TODO: remove unwraps
126        )));
127      } else {
128        let img = image::load_from_memory_with_format(&buf, image::ImageFormat::Jpeg).unwrap();
129        return Ok(Some(img));
130      }
131    }
132    Ok(None)
133  }

135  fn full_image(&self, file: &RawSource, params: &RawDecodeParams) -> Result<Option<DynamicImage>> {
136    if params.image_index != 0 {
137      return Ok(None);
138    }
139    if let Some(sub_ifds) = self.tiff.root_ifd().get_sub_ifd_all(TiffCommonTag::SubIFDs) {
140      let first_ifd = sub_ifds
141        .iter()
142        .find(|ifd| ifd.get_entry(TiffCommonTag::NewSubFileType).map(|entry| entry.force_u32(0)) == Some(1));
143      if let Some(preview_ifd) = first_ifd {
144        let buf = preview_ifd
145          .strip_data(&mut file.reader())
146          .map_err(|e| RawlerError::DecoderFailed(format!("Failed to get strip data: {}", e)))?
147          .into_iter()
148          .flatten()
149          .collect();
150        let compression = preview_ifd.get_entry(TiffCommonTag::Compression).ok_or("Missing tag")?.force_usize(0);
151        let width = fetch_tiff_tag!(preview_ifd, TiffCommonTag::ImageWidth).force_usize(0);
152        let height = fetch_tiff_tag!(preview_ifd, TiffCommonTag::ImageLength).force_usize(0);
153        if compression == 1 {
154          return Ok(Some(DynamicImage::ImageRgb8(
155            ImageBuffer::<Rgb<u8>, Vec<u8>>::from_raw(width as u32, height as u32, buf).unwrap(), // TODO: remove unwraps
156          )));
157        } else {
158          let img = image::load_from_memory_with_format(&buf, image::ImageFormat::Jpeg).unwrap();
159          return Ok(Some(img));
160        }
161      }
162    }
163    Ok(None)
164  }

252impl<'a> DngDecoder<'a> {
253  fn get_raw_ifd(&self) -> Result<&IFD> {
254    let ifds = self
255      .tiff
256      .find_ifds_with_tag(TiffCommonTag::Compression)
257      .into_iter()
258      .filter(|ifd| {
259        let compression = (**ifd).get_entry(TiffCommonTag::Compression).unwrap().force_u32(0);
260        let subsampled = match (**ifd).get_entry(TiffCommonTag::NewSubFileType) {
261          Some(e) => e.force_u32(0) & 1 != 0,
262          None => false,
263        };
264        !subsampled && (compression == 7 || compression == 1 || compression == 0x884c)
265      })
266      .collect::<Vec<&IFD>>();
267    
268    if let Some(ifd) = ifds.first() {
269        Ok(ifd)
270    } else {
271        Err(RawlerError::DecoderFailed(
272            "Could not find a suitable RAW Image File Directory (IFD) in the DNG file. The file might have an unsupported format.".to_string()
273        ))
274    }
275  }

277  fn make_camera(&self, raw: &IFD, width: usize, height: usize) -> Result<Camera> {
278    let mode = String::from("dng");
279
280    let make = self
281      .tiff
282      .root_ifd()
283      .get_entry(TiffCommonTag::Make)
284      .and_then(|x| x.as_string())
285      .cloned()
286      .unwrap_or_default();
287    let model = self
288      .tiff
289      .root_ifd()
290      .get_entry(TiffCommonTag::Model)
291      .and_then(|x| x.as_string())
292      .cloned()
293      .unwrap_or_default();
294
295    let active_area = self.get_active_area(raw, width, height);
296    let crop_area = if let Some(crops) = self.get_crop(raw) {
297      if let Some(active_area) = &active_area {
298        let mut full = crops;
299        full.p.x += active_area[0]; // left
300        full.p.y += active_area[1]; // Top
301        Some(full.as_ltrb_offsets(width, height))
302      } else {
303        Some(crops.as_ltrb_offsets(width, height))
304      }
305    } else {
306      None
307    };
308
309    let linear = fetch_tiff_tag!(raw, TiffCommonTag::PhotometricInt).force_usize(0) == 34892;
310    let cfa = if linear { CFA::default() } else { self.get_cfa(raw)? };
311    let color_matrix = self.get_color_matrix()?;
312    let real_bps = if raw.has_entry(TiffCommonTag::Linearization) {
313      // If DNG contains linearization table, output is always 16 bits
314      16
315    } else {
316      raw.get_entry(TiffCommonTag::BitsPerSample).map(|v| v.force_usize(0)).unwrap_or(16)
317    };
318
319    Ok(Camera {
320      clean_make: make.clone(),
321      clean_model: model.clone(),
322      make,
323      model,
324      mode,
325      whitelevel: None,
326      blacklevel: None,
327      blackareah: None,
328      blackareav: None,
329      xyz_to_cam: Default::default(),
330      color_matrix,
331      cfa,
332      active_area,
333      crop_area,
334      real_bps,
335      ..Default::default()
336    })
337  }

339  fn get_wb(&self) -> Result<[f32; 4]> {
340    if let Some(levels) = self.tiff.get_entry(TiffCommonTag::AsShotNeutral) {
341      Ok([1.0 / levels.force_f32(0), 1.0 / levels.force_f32(1), 1.0 / levels.force_f32(2), f32::NAN])
342    } else {
343      Ok([f32::NAN, f32::NAN, f32::NAN, f32::NAN])
344    }

347  fn get_blacklevels(&self, raw: &IFD) -> Result<Option<BlackLevel>> {
348    let cpp = raw.get_entry(TiffCommonTag::SamplesPerPixel).map(|entry| entry.force_usize(0)).unwrap_or(1);
349    if let Some(entry) = raw.get_entry(TiffCommonTag::BlackLevels) {
350      let levels = match &entry.value {
351        Value::Short(black) => black.iter().copied().map(Rational::from).collect(),
352        Value::Long(black) => black.iter().copied().map(Rational::from).collect(),
353        Value::Rational(black) => black.clone(),
354        _ => return Err(format!("Unsupported BlackLevel type: {}", entry.value_type_name()).into()),
355      };
356      let mut repeat = (1, 1);
357      if let Some(Entry {
358        value: Value::Short(value), ..
359      }) = raw.get_entry(DngTag::BlackLevelRepeatDim)
360      {
361        if value.len() == 2 {
362          repeat = (value[0] as usize, value[1] as usize);
363        } else {
364          // Pentax K-3 Mark III Monochrome is known to has invalid tag
365          log::warn!("File has BlackLevelRepeatDim tag but with invalid length: {}", value.len());
366        }
367      }
368      Ok(Some(BlackLevel::new(&levels, repeat.1, repeat.0, cpp)))
369    } else {
370      Ok(None)
371    }
372  }

374  fn get_whitelevels(&self, raw: &IFD) -> Result<Option<WhiteLevel>> {
375    let cpp = fetch_tiff_tag!(raw, TiffCommonTag::SamplesPerPixel).force_usize(0);
376    if let Some(levels) = raw.get_entry(TiffCommonTag::WhiteLevel) {
377      let mut whitelevels = WhiteLevel((0..levels.count()).map(|i| levels.force_u32(i as usize)).collect());
378      // Fixes a bug where only a single whitelevel value is given.
379      if whitelevels.0.len() == 1 && cpp > 1 {
380        whitelevels.0 = vec![whitelevels.0[0]; cpp];
381      }
382      return Ok(Some(whitelevels));
383    }
384    Ok(None)
385  }

402  fn get_crop(&self, raw: &IFD) -> Option<Rect> {
403    if let Some(crops) = raw.get_entry(DngTag::DefaultCropOrigin) {
404      let p = Point::new(crops.force_usize(0), crops.force_usize(1));
405      if let Some(size) = raw.get_entry(DngTag::DefaultCropSize) {
406        let d = Dim2::new(size.force_usize(0), size.force_usize(1));
407        return Some(Rect::new(p, d));
408      }
409    }
410    None

428  fn get_color_matrix(&self) -> Result<HashMap<Illuminant, FlatColorMatrix>> {
429    let mut result = HashMap::new();
430
431    let mut read_matrix = |cal: DngTag, mat: DngTag| -> Result<()> {
432      if let Some(c) = self.tiff.get_entry(mat) {
433        let illuminant: Illuminant = fetch_tiff_tag!(self.tiff, cal).force_u16(0).try_into()?;
434        let mut matrix = FlatColorMatrix::new();
435        for i in 0..c.count() as usize {
436          matrix.push(c.force_f32(i));
437        }
438        assert!(matrix.len() <= 12 && !matrix.is_empty());
439        result.insert(illuminant, matrix);
440      }
441      Ok(())
442    };
443
444    read_matrix(DngTag::CalibrationIlluminant1, DngTag::ColorMatrix1)?;
445    read_matrix(DngTag::CalibrationIlluminant2, DngTag::ColorMatrix2)?;
446    // TODO: add 3
447
448    Ok(result)
449  }

451  pub fn decode_uncompressed(&self, file: &RawSource, raw: &IFD, width: usize, height: usize, dummy: bool) -> Result<PixU16> {
452    let strips: Vec<&[u8]> = raw.strip_data_rawsource(file)?;
453    let strips_continous: Vec<u8>;
454    let src = if strips.len() == 1 {
455      strips[0]
456    } else {
457      strips_continous = strips.into_iter().flatten().copied().collect();
458      &strips_continous
459    };
460    match (raw.endian, fetch_tiff_tag!(raw, TiffCommonTag::BitsPerSample).force_u32(0)) {
461      // 16 bits, encoding depends on TIFF endianess
462      (Endian::Big, 16) => Ok(decode_16be(src, width, height, dummy)),
463      (Endian::Little, 16) => Ok(decode_16le(src, width, height, dummy)),
464      // 12 Bits, DNG spec says it must be always encoded as big-endian
465      (_, 12) => Ok(decode_12be(src, width, height, dummy)),
466      // 10 Bits, DNG spec says it must be always encoded as big-endian
467      (_, 10) => Ok(decode_10be(src, width, height, dummy)),
468      // 8 bits with linearization table
469      (_, 8) if raw.has_entry(TiffCommonTag::Linearization) => {
470        let linearization = fetch_tiff_tag!(self.tiff, TiffCommonTag::Linearization);
471        let curve = {
472          let mut points = vec![0_u16; 256];
473          for i in 0..256 {
474            points[i] = linearization.force_u32(i) as u16;
475          }
476          LookupTable::new(&points)
477        };
478        Ok(decode_8bit_wtable(src, &curve, width, height, dummy))
479      }
480      // 8 bits
481      (_, 8) => Ok(decode_8bit(src, width, height, dummy)),
482      // Generic MSB decoder for exotic packed bit sizes
483      (_, bps) if bps > 0 && bps < 16 => Ok(decode_generic_msb(src, width, height, bps, dummy)),
484      // Unhandled bits
485      (_, bps) => Err(format_args!("DNG: Don't know how to handle DNG with {} bps", bps).into()),
486    }
487  }
488
489  pub fn decode_compressed(&self, file: &RawSource, raw: &IFD, width: usize, height: usize, cpp: usize, dummy: bool) -> Result<PixU16> {
490    if let Some(offsets) = raw.get_entry(TiffCommonTag::StripOffsets) {
491      // We're in a normal offset situation
492      if offsets.count() != 1 {
493        return Err("DNG: files with more than one slice not supported yet".into());
494      }
495      let offset = offsets.force_u64(0);
496      let size = fetch_tiff_tag!(raw, TiffCommonTag::StripByteCounts).force_u64(0);
497      let src = file.subview(offset, size).unwrap();
498      let mut out = alloc_image_ok!(width, height, dummy);
499      let decompressor = LjpegDecompressor::new(src)?;
500      decompressor.decode(out.pixels_mut(), 0, width, width, height, dummy)?;
501      Ok(out)
502    } else if let Some(offsets) = raw.get_entry(TiffCommonTag::TileOffsets) {
503      // They've gone with tiling
504      let twidth = fetch_tiff_tag!(raw, TiffCommonTag::TileWidth).force_usize(0) * cpp;
505      let tlength = fetch_tiff_tag!(raw, TiffCommonTag::TileLength).force_usize(0);
506      let coltiles = (width - 1) / twidth + 1;
507      let rowtiles = (height - 1) / tlength + 1;
508      if coltiles * rowtiles != offsets.count() as usize {
509        return Err(format_args!("DNG: trying to decode {} tiles from {} offsets", coltiles * rowtiles, offsets.count()).into());
510      }
511      let buffer = file.as_vec().unwrap();
512      decode_threaded_multiline(
513        width,
514        height,
515        tlength,
516        dummy,
517        &(|strip: &mut [u16], row| {
518          let row = row / tlength;
519          for col in 0..coltiles {
520            // Calculate output width & length for current tile (right or bottom tile may have smaller dimension)
521            let owidth = cmp::min(width, (col + 1) * twidth) - col * twidth;
522            let olength = cmp::min(height, (row + 1) * tlength) - row * tlength;
523            let offset = offsets.force_usize(row * coltiles + col);
524            let src = &buffer[offset..];
525            let decompressor = LjpegDecompressor::new(src)?;
526
527            // If SOF width & height matches tile dimension, we can decode directly to strip
528            if decompressor.width() == twidth && decompressor.height() == tlength {
529              decompressor.decode(strip, col * twidth, width, owidth, olength, dummy)?;
530            }
531            // If SOF has other dimension but still same pixel count, we can decode
532            // by using a temporary tile buffer. This is the case if RGGB data is encoded
533            // by two input lines into one output line.
534            // Encoded data is aligned like this:
535            //   RGRGRGRGRGRGRGRGRGGBGBGBGBGBGBGBGBGB
536            //   RGRGRGRGRGRGRGRGRGGBGBGBGBGBGBGBGBGB
537            else if decompressor.width() * decompressor.height() == twidth * tlength {
538              // cps is already included in all values, so just compare them.
539              let mut tile = alloc_image_plain!(decompressor.width(), decompressor.height(), dummy);
540              // The stripwidth is of same width as the temporary tile.
541              let stripwidth = decompressor.width();
542              decompressor.decode(tile.pixels_mut(), 0, stripwidth, decompressor.width(), decompressor.height(), dummy)?;
543              // Copy lines from temporary tile to strip buffer, but limit it if destination strip
544              // is smaller than temporary tile (right/bottom tiles).
545              for (i, pix) in tile.pixels().chunks_exact(twidth).enumerate().take(olength) {
546                let start = (i * width) + (col * twidth);
547                strip[start..start + owidth].copy_from_slice(&pix[0..owidth]);
548              }
549            } else {
550              return Err(format!(
551                "ljpeg92 decoding failed: tile dimensions {}x{} not matching SOF dimensions {}x{}, dng cps: {}, sof cps: {}",
552                twidth,
553                tlength,
554                decompressor.width(),
555                decompressor.height(),
556                cpp,
557                decompressor.components(),
558              ));
559            }
560          }
561          Ok(())
562        }),
563      )
564      .map_err(RawlerError::DecoderFailed)
565    } else {
566      Err(RawlerError::DecoderFailed("DNG: didn't find tiles or strips".to_string()))
567    }
568  }
569}
570
571fn apply_linearization(image: &mut PixU16, tbl: &Value, bits: u32) {
572  match tbl {
573    Value::Short(points) => {
574      let table = LookupTable::new_with_bits(points, bits);
575      image.pixel_rows_mut().for_each(|row| {
576        let mut random = LEu32(row.as_bytes(), 0);
577        row.iter_mut().for_each(|p| {
578          *p = table.dither(*p, &mut random);
579        })
580      });
581    }
582    _ => {
583      panic!("Unsupported linear table");
584    }
585  }
