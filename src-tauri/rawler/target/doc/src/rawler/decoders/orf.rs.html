52pub fn parse_makernote<R: Read + Seek>(reader: &mut R, exif_ifd: &IFD) -> Result<Option<IFD>> {
53  if let Some(exif) = exif_ifd.get_entry(ExifTag::MakerNotes) {
54    let offset = exif.offset().unwrap() as u32;
55    log::debug!("Makernote offset: {}", offset);
56    match &exif.value {
57      Value::Undefined(data) => {
58        let mut off = 0;
59        // Olympus starts the makernote with their own name, sometimes truncated
60        if data[0..5] == b"OLYMP"[..] {
61          off += 8;
62          if data[0..7] == b"OLYMPUS"[..] {
63            off += 4;
64          }
65        }
66        // OM Digital Solutions put their name in front of the TIFF structure, too
67        if data[0..9] == b"OM SYSTEM"[..] {
68          off += 16;
69          assert_eq!(data[12..14], b"II"[..]);
70        }
71        let endian = exif_ifd.endian;
72        //assert!(data[off..off + 2] == b"II"[..] || data[off..off + 2] == b"MM"[..], "ORF: must contain endian marker in makernote IFD");
73        //let endian = if data[off..off + 2] == b"II"[..] { Endian::Little } else { Endian::Big };
74        //off += 4;
75
76        let mut mainifd = IFD::new(reader, offset + off as u32, exif_ifd.base, exif_ifd.corr, endian, &[0x3000])?;
77
78        // Parse the Olympus Equipment section if it exists
79        if let Some(entry) = mainifd.get_entry_raw_with_len(OrfMakernotes::EquipmentIFD, reader, 4)? {
80          // The entry is of type UNDEFINED and count = 1. This tag contains a single 32 bit
81          // offset to the IFD.
82          let ioff = entry.get_force_u32(0);
83          log::debug!("Found EquipmentIFD at offset: {}", ioff);
84          // The IFD start at offset+ioff, but all offsets inside the IFD a relative to the main makernote IFD offset.
85          // So we use the main IFD as base offset, but start parsing IFD at ioff.
86          let ifd = IFD::new(reader, ioff, offset, 0, endian, &[])?;
87          mainifd.sub.insert(OrfMakernotes::EquipmentIFD.into(), vec![ifd]);
88        }
89
90        // For Olympus or OM-System models
91        if off == 12 || off == 16 {
92          // Parse the Olympus ImgProc section if it exists
93          let ioff = if let Some(entry) = mainifd.get_entry_raw_with_len(OrfMakernotes::ImageProcessingIFD, reader, 4)? {
94            // The entry is of type UNDEFINED and count = 1. This tag contains a single 32 bit
95            // offset to the IFD.
96            entry.get_force_u32(0)
97          } else {
98            0
99          };
100          if ioff != 0 {
101            log::debug!("Found ImageIFD at offset: {}", ioff);
102            // The IFD start at offset+ioff, but all offsets inside the IFD a relative to the main makernote IFD offset.
103            // So we use the main IFD as base offset, but start parsing IFD at ioff.
104            let iprocifd = IFD::new(reader, ioff, offset, 0, endian, &[])?;
105            mainifd.sub.insert(OrfMakernotes::ImageProcessingIFD.into(), vec![iprocifd]);
106          } else {
107            log::debug!("ORF ImageIFD not found");
108          }
109        }
110        Ok(Some(mainifd))
111      }
112      _ => Err(RawlerError::DecoderFailed("EXIF makernote has unknown type".to_string())),
113    }
114  } else {
115    Ok(None)
116  }
117}

119impl<'a> OrfDecoder<'a> {
120  pub fn new(file: &RawSource, tiff: GenericTiffReader, rawloader: &'a RawLoader) -> Result<OrfDecoder<'a>> {
121    let camera = rawloader.check_supported(tiff.root_ifd())?;
122
123    let makernote = if let Some(exif) = tiff.find_first_ifd_with_tag(ExifTag::MakerNotes) {
124      parse_makernote(&mut file.reader(), exif)?
125    } else {
126      log::warn!("ORF makernote not found");
127      None
128    }
129    .ok_or("File has not makernotes")?;
130
131    //makernote.dump::<ExifTag>(0).iter().for_each(|line| eprintln!("DUMP: {}", line));
132
133    Ok(OrfDecoder {
134      tiff,
135      rawloader,
136      camera,
137      makernote,
138    })
139  }
140}

143  fn raw_image(&self, file: &RawSource, _params: &RawDecodeParams, dummy: bool) -> Result<RawImage> {
144    let raw = self.tiff.find_first_ifd_with_tag(TiffCommonTag::StripOffsets).unwrap();
145    let width = fetch_tiff_tag!(raw, TiffCommonTag::ImageWidth).force_usize(0);
146    let height = fetch_tiff_tag!(raw, TiffCommonTag::ImageLength).force_usize(0);
147    let offset = fetch_tiff_tag!(raw, TiffCommonTag::StripOffsets).force_usize(0);
148    let counts = fetch_tiff_tag!(raw, TiffCommonTag::StripByteCounts);
149
150    let mut size: usize = 0;
151    for i in 0..counts.count() {
152      size += counts.force_u32(i as usize) as usize;
153    }
154
155    let camera = if width >= self.camera.highres_width {
156      self.rawloader.check_supported_with_mode(self.tiff.root_ifd(), "highres")?
157    } else {
158      self.camera.clone()
159    };
160
161    let src = file.subview_padded(offset as u64, size as u64)?; // TODO add size and check all samples
162
163    log::debug!(
164      "ORF raw image size: {}, dim: {}x{}, total mp: {}, strip counts: {}",
165      size,
166      width,
167      height,
168      width * height,
169      counts.count()
170    );
171
172    // These conditions are sorted in descending order.
173    // All ORF files comes with no hints about the used compression.
174    // But we need to differentiate between 12be-interlaced and
175    // 12be-msb32 because they are in the same size range.
176    let image = if size >= width * height * 2 {
177      if self.tiff.little_endian() {
178        log::debug!("ORF: decode_12le_unpacked_left_aligned");
179        decode_12le_unpacked_left_aligned(&src, width, height, dummy)
180      } else {
181        log::debug!("ORF: decode_12be_unpacked_left_aligned");
182        decode_12be_unpacked_left_aligned(&src, width, height, dummy)
183      }
184    } else if size >= width * height / 10 * 16 {
185      log::debug!("ORF: decode_12le_wcontrol");
186      decode_12le_wcontrol(&src, width, height, dummy)
187    } else if size >= width * height * 12 / 8 {
188      if self.camera.find_hint("interlaced") {
189        log::debug!("ORF: decode_12be_interlaced");
190        decode_12be_interlaced(&src, width, height, dummy)
191      } else {
192        log::debug!("ORF: decode_12be_msb32");
193        //decode_12be_interlaced(&src, width, height, dummy)
194        decode_12be_msb32(&src, width, height, dummy)
195      }
196    } else {
197      log::debug!("ORF: fallback to decode_compressed");
198      OrfDecoder::decode_compressed(&src, width, height, dummy)
199    };
200
201    let cpp = 1;
202
203    let blacklevel = self.get_blacklevel()?;
204    let whitelevel = None;
205    let photometric = RawPhotometricInterpretation::Cfa(CFAConfig::new_from_camera(&self.camera));
206    let mut img = RawImage::new(camera, image, cpp, normalize_wb(self.get_wb()?), photometric, blacklevel, whitelevel, dummy);
207    if let Some(crop) = self.get_crop()? {
208      img.crop_area = Some(crop);
209    }
210
211    Ok(img)
212  }

229impl<'a> OrfDecoder<'a> {
230  /* This is probably the slowest decoder of them all.
231   * I cannot see any way to effectively speed up the prediction
232   * phase, which is by far the slowest part of this algorithm.
233   * Also there is no way to multithread this code, since prediction
234   * is based on the output of all previous pixel (bar the first four)
235   */
236
237  pub fn decode_compressed(buf: &PaddedBuf, width: usize, height: usize, dummy: bool) -> PixU16 {
238    let mut out = alloc_image!(width, height, dummy);
239
240    /* Build a table to quickly look up "high" value */
241    let mut bittable: [u8; 4096] = [0; 4096];
242    for i in 0..4096 {
243      let mut b = 12;
244      for high in 0..12 {
245        if ((i >> (11 - high)) & 1) != 0 {
246          b = high;
247          break;
248        }
249      }
250      bittable[i] = b;
251    }
252
253    let mut left: [i32; 2] = [0; 2];
254    let mut nw: [i32; 2] = [0; 2];
255    let mut pump = BitPumpMSB::new(&buf[7..]);
256
257    for row in 0..height {
258      let mut acarry: [[i32; 3]; 2] = [[0; 3]; 2];
259
260      for c in 0..width / 2 {
261        let col: usize = c * 2;
262        for s in 0..2 {
263          // Run twice for odd and even pixels
264          let i = if acarry[s][2] < 3 { 2 } else { 0 };
265          let mut nbits = 2 + i;
266          while ((acarry[s][0] >> (nbits + i)) & 0xffff) > 0 {
267            nbits += 1
268          }
269          nbits = cmp::min(nbits, 16);
270          let b = pump.peek_ibits(15);
271
272          let sign: i32 = -(b >> 14);
273          let low: i32 = (b >> 12) & 3;
274          let mut high: i32 = bittable[(b & 4095) as usize] as i32;
275
276          // Skip bytes used above or read bits
277          if high == 12 {
278            pump.consume_bits(15);
279            high = pump.get_ibits(16 - nbits) >> 1;
280          } else {
281            pump.consume_bits((high + 4) as u32);
282          }
283
284          acarry[s][0] = ((high << nbits) | pump.get_ibits(nbits)) as i32;
285          let diff = (acarry[s][0] ^ sign) + acarry[s][1];
286          acarry[s][1] = (diff * 3 + acarry[s][1]) >> 5;
287          acarry[s][2] = if acarry[s][0] > 16 { 0 } else { acarry[s][2] + 1 };
288
289          if row < 2 || col < 2 {
290            // We're in a border, special care is needed
291            let pred = if row < 2 && col < 2 {
292              // We're in the top left corner
293              0
294            } else if row < 2 {
295              // We're going along the top border
296              left[s]
297            } else {
298              // col < 2, we're at the start of a line
299              nw[s] = out[(row - 2) * width + (col + s)] as i32;
300              nw[s]
301            };
302            left[s] = pred + ((diff << 2) | low);
303            out[row * width + (col + s)] = left[s] as u16;
304          } else {
305            let up: i32 = out[(row - 2) * width + (col + s)] as i32;
306            let left_minus_nw: i32 = left[s] - nw[s];
307            let up_minus_nw: i32 = up - nw[s];
308            // Check if sign is different, and one is not zero
309            let pred = if left_minus_nw * up_minus_nw < 0 {
310              if left_minus_nw.abs() > 32 || up_minus_nw.abs() > 32 {
311                left[s] + up_minus_nw
312              } else {
313                (left[s] + up) >> 1
314              }
315            } else if left_minus_nw.abs() > up_minus_nw.abs() {
316              left[s]
317            } else {
318              up
319            };
320
321            left[s] = pred + ((diff << 2) | low);
322            nw[s] = up;
323            out[row * width + (col + s)] = left[s] as u16;
324          }
325        }
326      }
327    }
328    out
329  }

331  fn get_blacklevel(&self) -> Result<Option<BlackLevel>> {
332    let ifd = self.makernote.find_ifds_with_tag(OrfImageProcessing::OrfBlackLevels);
333    if ifd.is_empty() {
334      log::info!("ORF: Couldn't find ImgProc IFD, unable to read blacklevel");
335      return Ok(None);
336    }
337
338    let blacks = fetch_tiff_tag!(ifd[0], OrfImageProcessing::OrfBlackLevels);
339    let levels = [blacks.force_u16(0), blacks.force_u16(1), blacks.force_u16(2), blacks.force_u16(3)];
340    Ok(Some(BlackLevel::new(&levels, self.camera.cfa.width, self.camera.cfa.height, 1)))
341  }
342
343  fn get_crop(&self) -> Result<Option<Rect>> {
344    let ifd = self.makernote.find_ifds_with_tag(OrfImageProcessing::CropLeft);
345    if ifd.is_empty() {
346      return Ok(None);
347    }
348    let crop_left = fetch_tiff_tag!(ifd[0], OrfImageProcessing::CropLeft).force_usize(0);
349    let crop_top = fetch_tiff_tag!(ifd[0], OrfImageProcessing::CropTop).force_usize(0);
350    let crop_width = fetch_tiff_tag!(ifd[0], OrfImageProcessing::CropWidth).force_usize(0);
351    let crop_height = fetch_tiff_tag!(ifd[0], OrfImageProcessing::CropHeight).force_usize(0);
352    Ok(Some(Rect::new(Point::new(crop_left, crop_top), Dim2::new(crop_width, crop_height))))
353  }

355  /// Get lens description by analyzing TIFF tags and makernotes
356  fn get_lens_description(&self) -> Result<Option<&'static LensDescription>> {
357    if let Some(ifd) = self.makernote.get_sub_ifd(OrfMakernotes::EquipmentIFD) {
358      match ifd.get_entry(OrfEquipmentTags::LensType) {
359        Some(Entry {
360          value: Value::Byte(settings), ..
361        }) => {
362          log::debug!("Lens type tag: {:?}", settings);
363          let make_id = settings[0];
364          let model_id = settings[2];
365          let submodel_id = settings[3];
366          let composite_id = format!("{:02X} {:02X} {:02X}", make_id, model_id, submodel_id);
367          log::debug!("ORF lens composite ID: {}", composite_id);
368          let resolver = LensResolver::new()
369            .with_olympus_id(Some(composite_id))
370            .with_camera(&self.camera)
371            .with_focal_len(self.get_focal_len()?)
372            .with_mounts(&[MFT_MOUNT.into()]);
373          return Ok(resolver.resolve());
374        }
375        _ => {
376          log::warn!("Camera settings in makernote not found, no lens data available");
377        }
378      }
379    }
380    log::warn!("No lens data found");
381    Ok(None)
382  }

384  fn get_focal_len(&self) -> Result<Option<Rational>> {
385    if let Some(exif) = self.tiff.find_first_ifd_with_tag(ExifTag::MakerNotes) {
386      if let Some(Entry {
387        value: Value::Short(focal), ..
388      }) = exif.get_entry(ExifTag::FocalLength)
389      {
390        return Ok(focal.get(1).map(|v| Rational::new(*v as u32, 1)));
391      }
392    }
393    Ok(None)
394  }

396  fn get_wb(&self) -> Result<[f32; 4]> {
397    let redmul = self.makernote.get_entry(OrfMakernotes::OlympusRedMul);
398    let bluemul = self.makernote.get_entry(OrfMakernotes::OlympusBlueMul);
399    match (redmul, bluemul) {
400      (Some(redmul), Some(bluemul)) => Ok([redmul.force_u32(0) as f32, 256.0, 256.0, bluemul.force_u32(0) as f32]),
401      _ => {
402        let ifd = self.makernote.find_ifds_with_tag(OrfImageProcessing::OrfBlackLevels);
403        if ifd.is_empty() {
404          return Err(RawlerError::DecoderFailed("ORF: Couldn't find ImgProc IFD".to_string()));
405        }
406        let wbs = fetch_tiff_tag!(ifd[0], OrfImageProcessing::WB_RBLevels);
407        Ok([wbs.force_f32(0), 256.0, 256.0, wbs.force_f32(1)])
408      }
409    }
410  }

413fn normalize_wb(raw_wb: [f32; 4]) -> [f32; 4] {
414  log::debug!("ORF raw wb: {:?}", raw_wb);
415  let div = raw_wb[1];
416  let mut norm = raw_wb;
417  norm.iter_mut().for_each(|v| {
418    if v.is_normal() {
419      *v /= div
420    }
421  });
422  [norm[0], (norm[1] + norm[2]) / 2.0, norm[3], f32::NAN]
423}
