36impl<'a> Decoder for DcsDecoder<'a> {
37  fn raw_image(&self, file: &RawSource, _params: &RawDecodeParams, dummy: bool) -> Result<RawImage> {
38    let raw = self.tiff.find_ifd_with_new_subfile_type(0).unwrap();
39    let width = fetch_tiff_tag!(raw, TiffCommonTag::ImageWidth).force_usize(0);
40    let height = fetch_tiff_tag!(raw, TiffCommonTag::ImageLength).force_usize(0);
41    let offset = fetch_tiff_tag!(raw, TiffCommonTag::StripOffsets).force_usize(0);
42    let src = file.subview_until_eof_padded(offset as u64)?; // TODO add size and check all samples
43
44    let linearization = fetch_tiff_tag!(self.tiff, TiffCommonTag::GrayResponse);
45    let table = {
46      let mut t: [u16; 256] = [0; 256];
47      for i in 0..256 {
48        t[i] = linearization.force_u32(i) as u16;
49      }
50      LookupTable::new(&t)
51    };
52
53    let image = decode_8bit_wtable(&src, &table, width, height, dummy);
54    let cpp = 1;
55    ok_cfa_image(self.camera.clone(), cpp, [f32::NAN, f32::NAN, f32::NAN, f32::NAN], image, dummy)
56  }
