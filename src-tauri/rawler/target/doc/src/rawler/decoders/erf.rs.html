38impl<'a> ErfDecoder<'a> {
39  pub fn new(file: &RawSource, tiff: GenericTiffReader, rawloader: &'a RawLoader) -> Result<ErfDecoder<'a>> {
40    let camera = rawloader.check_supported(tiff.root_ifd())?;
41
42    let makernote = if let Some(exif) = tiff.find_first_ifd_with_tag(ExifTag::MakerNotes) {
43      exif.parse_makernote(&mut file.reader(), OffsetMode::Absolute, &[])?
44    } else {
45      warn!("ERF makernote not found");
46      None
47    }
48    .ok_or("File has not makernotes")?;
49
50    Ok(ErfDecoder {
51      tiff,
52      rawloader,
53      camera,
54      makernote,
55    })
56  }
57}

59impl<'a> Decoder for ErfDecoder<'a> {
60  fn raw_image(&self, file: &RawSource, _params: &RawDecodeParams, dummy: bool) -> Result<RawImage> {
61    let raw = self.tiff.find_first_ifd_with_tag(TiffCommonTag::CFAPattern).unwrap();
62    let width = fetch_tiff_tag!(raw, TiffCommonTag::ImageWidth).force_usize(0);
63    let height = fetch_tiff_tag!(raw, TiffCommonTag::ImageLength).force_usize(0);
64    let offset = fetch_tiff_tag!(raw, TiffCommonTag::StripOffsets).force_usize(0);
65    let src = file.subview_until_eof(offset as u64).unwrap();
66
67    let image = decode_12be_wcontrol(src, width, height, dummy);
68    let cpp = 1;
69
70    let blacklevel = self.get_blacklevel(cpp);
71    let photometric = RawPhotometricInterpretation::Cfa(CFAConfig::new_from_camera(&self.camera));
72    let img = RawImage::new(self.camera.clone(), image, cpp, self.get_wb()?, photometric, blacklevel, None, dummy);
73    Ok(img)
74  }

91impl<'a> ErfDecoder<'a> {
92  fn get_wb(&self) -> Result<[f32; 4]> {
93    let levels = fetch_tiff_tag!(self.makernote, TiffCommonTag::EpsonWB);
94    if levels.count() != 256 {
95      Err(RawlerError::DecoderFailed("ERF: Levels count is off".to_string()))
96    } else {
97      let r = BEu16(levels.get_data(), 48) as f32;
98      let b = BEu16(levels.get_data(), 50) as f32;
99      Ok([r * 508.0 * 1.078 / 65536.0, 1.0, b * 382.0 * 1.173 / 65536.0, f32::NAN])
100    }
101  }
102
103  fn get_blacklevel(&self, cpp: usize) -> Option<BlackLevel> {
104    if let Some(levels) = self.makernote.get_entry(0x0401) {
105      let levels = [levels.force_u16(0), levels.force_u16(1), levels.force_u16(2), levels.force_u16(3)];
106      return Some(BlackLevel::new(&levels, self.camera.cfa.width, self.camera.cfa.height, cpp));
107    }
108    None
109  }
