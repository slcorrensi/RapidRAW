84  fn raw_image(&self, file: &RawSource, _params: &RawDecodeParams, dummy: bool) -> Result<RawImage> {
85    /*
86    for (i, ifd) in self.tiff.chains().iter().enumerate() {
87      eprintln!("IFD {}", i);
88      for line in ifd_dump::<crate::tags::LegacyTiffRootTag>(ifd, 10) {
89        eprintln!("{}", line);
90      }
91    }
92     */
93
94    let camera = &self.camera;
95    let (raw, offset) = {
96      if let Some(raw) = self.tiff.find_first_ifd(TiffCommonTag::Cr2Id) {
97        (raw, fetch_tiff_tag!(raw, TiffCommonTag::StripOffsets).force_usize(0))
98      } else if let Some(raw) = self.tiff.find_first_ifd(TiffCommonTag::CFAPattern) {
99        (raw, fetch_tiff_tag!(raw, TiffCommonTag::StripOffsets).force_usize(0))
100      } else if let Some(off) = self.makernote.as_ref().and_then(|md| md.get_entry(TiffCommonTag::Cr2OldOffset)) {
101        // Old Canon TIF files contains the offset in makernote tags
102        (self.tiff.root_ifd(), off.value.force_usize(0))
103      } else {
104        return Err(RawlerError::DecoderFailed("CR2: Couldn't find raw info".to_string()));
105      }
106    };
107
108    // We don't have an excact length, so read until end.
109    let src = file.subview_until_eof(offset as u64)?;
110
111    let (cpp, image) = {
112      let decompressor = LjpegDecompressor::new(src)?;
113      let ljpegwidth = decompressor.width();
114      let mut width = ljpegwidth;
115      let mut height = decompressor.height();
116      let cpp = if decompressor.super_h() == 2 { 3 } else { 1 };
117      debug!("CR2 ljpeg components: {}", decompressor.components());
118      debug!("CR2 final cpp: {}", cpp);
119      debug!("CR2 dimension: {},{}", width / cpp, height);
120      let mut ljpegout = alloc_image_plain!(width, height, dummy);
121      if !dummy {
122        decompressor.decode(ljpegout.pixels_mut(), 0, width, width, height, dummy)?;
123      }
124
125      //crate::devtools::dump_image_u16(&ljpegout, width, height, "/tmp/cr2_before_striped.pnm");
126
127      // Linearize the output (applies only to D2000 as far as I can tell)
128      if !dummy && camera.find_hint("linearization") {
129        let table = {
130          let linearization = fetch_tiff_tag!(raw, TiffCommonTag::GrayResponse);
131          let mut t = [0_u16; 4096];
132          for i in 0..t.len() {
133            t[i] = linearization.force_u16(i);
134          }
135          LookupTable::new(&t)
136        };
137
138        let mut random = ljpegout[0] as u32;
139        for p in ljpegout.pixels_mut().iter_mut() {
140          *p = table.dither(*p, &mut random);
141        }
142      }
143
144      if cpp == 3 {
145        if raw.has_entry(TiffCommonTag::ImageWidth) {
146          width = fetch_tiff_tag!(raw, TiffCommonTag::ImageWidth).force_usize(0) * cpp;
147          height = fetch_tiff_tag!(raw, TiffCommonTag::ImageLength).force_usize(0);
148        } else if width / cpp < height {
149          let temp = width / cpp;
150          width = height * cpp;
151          height = temp;
152        }
153      } else if camera.find_hint("double_line") {
154        width /= 2;
155        height *= 2;
156      }
157
158      debug!("CR2 dimension2: {},{}", width / cpp, height);
159
160      // Take each of the vertical fields and put them into the right location
161      // FIXME: Doing this at the decode would reduce about 5% in runtime but I haven't
162      //        been able to do it without hairy code
163      if let Some(canoncol) = raw.get_entry(TiffCommonTag::Cr2StripeWidths) {
164        debug!("Found Cr2StripeWidths tag: {:?}", canoncol.value);
165        if canoncol.value.force_usize(0) == 0 {
166          if cpp == 3 {
167            self.convert_to_rgb(file, camera, &decompressor, width, height, ljpegout.pixels_mut(), dummy)?;
168            //width /= 3;
169          }
170          ljpegout.update_dimension(Dim2::new(width, height));
171          (cpp, ljpegout)
172          /*
173          if camera.find_hint("double_line") {
174            (width, height, cpp, PixU16::new_with(ljpegout.into_inner(), width, height))
175          } else {
176            (width, height, cpp, ljpegout)
177          }
178           */
179        } else {
180          let mut out = alloc_image_plain!(width, height, dummy);
181          if !dummy {
182            let mut fieldwidths = Vec::new();
183            debug_assert!(canoncol.value.force_usize(0) > 0);
184            debug_assert!(canoncol.value.force_usize(1) > 0);
185            debug_assert!(canoncol.value.force_usize(2) > 0);
186            for _ in 0..canoncol.value.force_usize(0) {
187              fieldwidths.push(canoncol.value.force_usize(1));
188            }
189            fieldwidths.push(canoncol.value.force_usize(2));
190
191            if decompressor.super_v() == 2 {
192              debug!("CR2 v=2 decoder used, h={}", decompressor.super_h());
193              // We've decoded 2 lines at a time so we also need to copy two strips at a time
194              let nfields = fieldwidths.len();
195              let fieldwidth = fieldwidths[0];
196              let mut fieldstart = 0;
197              let mut inpos = 0;
198              for _ in 0..nfields {
199                for row in (0..height).step_by(2) {
200                  for col in (0..fieldwidth).step_by(3) {
201                    let outpos = row * width + fieldstart + col;
202                    out[outpos..outpos + 3].copy_from_slice(&ljpegout[inpos..inpos + 3]);
203                    let outpos = (row + 1) * width + fieldstart + col;
204                    let inpos2 = inpos + ljpegwidth;
205                    out[outpos..outpos + 3].copy_from_slice(&ljpegout[inpos2..inpos2 + 3]);
206                    inpos += 3;
207                    if inpos % ljpegwidth == 0 {
208                      // we've used a full input line and we're reading 2 by 2 so skip one
209                      inpos += ljpegwidth;
210                    }
211                  }
212                }
213                fieldstart += fieldwidth;
214              }
215            } else {
216              let sh = decompressor.super_h();
217              debug!("CR2 v=1 decoder used, super_h: {}", sh);
218              let mut fieldstart = 0;
219              let mut fieldpos = 0;
220              for fieldwidth in fieldwidths {
221                // fix the inconsistent slice width in sRaw mode, ask Canon.
222                let fieldwidth = fieldwidth / sh * cpp;
223                // The output for full height of a vertical stripe is
224                // composed by the lines of all input stripes N:
225                // outb(line0) = slice[0](line[0])
226                // outb(line1) = slice[1](line[0])
227                // outb(line2) = slice[N-1](line[0])
228                for row in 0..height {
229                  let outpos = row * width + fieldstart;
230                  let inpos = fieldpos + row * fieldwidth;
231                  let outb = &mut out[outpos..outpos + fieldwidth];
232                  let inb = &ljpegout[inpos..inpos + fieldwidth];
233                  outb.copy_from_slice(inb);
234                }
235                fieldstart += fieldwidth;
236                fieldpos += fieldwidth * height;
237              }
238            }
239          }
240          if cpp == 3 {
241            self.convert_to_rgb(file, camera, &decompressor, width, height, out.pixels_mut(), dummy)?;
242            //width /= 3;
243          }
244          (cpp, out)
245          //(width, height, cpp, out)
246        }
247      } else {
248        ljpegout.update_dimension(Dim2::new(width, height));
249        (cpp, ljpegout)
250        // (width, height, cpp, PixU16::new_with(ljpegout.into_inner(), width, height))
251      }
252    };
253
254    let wb = self.get_wb(file, camera)?;
255    debug!("CR2 WB: {:?}", wb);
256    //assert_eq!(image.width, width * cpp);
257
258    let blacklevel = self.get_blacklevel(camera, cpp)?;
259    let whitelevel = self.get_whitelevel(cpp)?;
260
261    let photometric = match cpp {
262      1 => RawPhotometricInterpretation::Cfa(CFAConfig::new_from_camera(&self.camera)),
263      3 => RawPhotometricInterpretation::LinearRaw,
264      _ => todo!(),
265    };
266
267    let mut img = RawImage::new(camera.clone(), image, cpp, wb, photometric, blacklevel, whitelevel, dummy);
268
269    if let Some(file_crop) = self.get_sensor_area()? {
270      assert!(
271        img.crop_area.is_none(),
272        "Camera {} has embedded crop params, remove crop from config file!",
273        self.camera.clean_make
274      );
275      img.crop_area = Some(file_crop);
276    } else {
277      //panic!("Camera {} has no embedded crops, but all CR2 should contain them?!", self.camera.clean_make);
278      // 1D and D2000C has no crops!
279    }
280
281    if cpp == 3 {
282      // We have a sRAW or mRAW: the active_area from camera config is invalid now!
283      // We just apply the crop_area that comes from metadata, which is correct.
284      img.active_area = img.crop_area;
285    }
286
287    debug!("Blacklevel: {:?}", img.blacklevel);
288    debug!("Whitelevel: {:?}", img.whitelevel);
289    debug!("Black areas: {:?}", img.blackareas);
290    debug!("Active area: {:?}", img.active_area);
291    debug!("Crop area: {:?}", img.crop_area);
292    Ok(img)
293  }

305  fn full_image(&self, file: &RawSource, params: &RawDecodeParams) -> Result<Option<DynamicImage>> {
306    if params.image_index != 0 {
307      return Ok(None);
308    }
309    // For CR2, there is a full resolution image in IFD0.
310    // This is compressed with old-JPEG compression (Compression = 6)
311    let root_ifd = &self.tiff.root_ifd();
312    let buf = root_ifd
313      .singlestrip_data_rawsource(file)
314      .map_err(|e| RawlerError::DecoderFailed(format!("Failed to get strip data: {}", e)))?;
315    let compression = root_ifd.get_entry(TiffCommonTag::Compression).ok_or("Missing tag")?.force_usize(0);
316    let width = fetch_tiff_tag!(root_ifd, TiffCommonTag::ImageWidth).force_usize(0);
317    let height = fetch_tiff_tag!(root_ifd, TiffCommonTag::ImageLength).force_usize(0);
318    if compression == 1 {
319      Ok(Some(DynamicImage::ImageRgb8(
320        ImageBuffer::<Rgb<u8>, Vec<u8>>::from_raw(width as u32, height as u32, buf.to_vec()).unwrap(),
321      )))
322    } else {
323      let img = image::load_from_memory_with_format(buf, image::ImageFormat::Jpeg).unwrap();
324      Ok(Some(img))
325    }
326  }

336  Sraw1,
337  Sraw2,

340impl<'a> Cr2Decoder<'a> {
341  fn get_mode(makernote: &Option<IFD>) -> Result<Cr2Mode> {
342    if let Some(settings) = makernote.as_ref().and_then(|mn| mn.get_entry(Cr2MakernoteTag::CameraSettings)) {
343      match settings.get_u16(46) {
344        Ok(Some(0)) => Ok(Cr2Mode::Raw),
345        Ok(Some(1)) => Ok(Cr2Mode::Sraw1),
346        Ok(Some(2)) => Ok(Cr2Mode::Sraw2),
347        Ok(None) => Ok(Cr2Mode::Raw),
348        Ok(Some(v)) => Err(RawlerError::DecoderFailed(format!("Unknown sraw quality value found: {}", v))),
349        Err(_) => Err(RawlerError::DecoderFailed("Unknown sraw quality value".to_string())),
350      }
351    } else {
352      Ok(Cr2Mode::Raw)
353    }
354  }

356  /// Construct new CR2 decoder
357  /// This parses the RawFile again to include specific sub IFDs.
358  pub fn new(file: &RawSource, _tiff: GenericTiffReader, rawloader: &'a RawLoader) -> Result<Cr2Decoder<'a>> {
359    debug!("CR2 decoder choosen");
360
361    // Parse the TIFF again, with custom settings
362    let tiff = GenericTiffReader::new(&mut file.reader(), 0, 0, None, &[33424])?;
363
364    let exif = Self::new_exif_ifd(file, &tiff, rawloader)?;
365    let makernote = Self::new_makernote(file, &tiff, &exif, rawloader)?;
366
367    let mode = Self::get_mode(&makernote)?;
368
369    debug!("sRaw quality: {:?}", mode);
370    let mode_str = match mode {
371      Cr2Mode::Raw => "",
372      Cr2Mode::Sraw1 => "sRaw1",
373      Cr2Mode::Sraw2 => "sRaw2",
374    };
375
376    let camera = rawloader.check_supported_with_mode(tiff.root_ifd(), mode_str)?;
377
378    let xpacket = Self::read_xpacket(file, &tiff, rawloader)?;
379
380    let model_id = makernote
381      .as_ref()
382      .and_then(|mn| mn.get_entry(Cr2MakernoteTag::ModelId).and_then(|v| v.get_u32(0).transpose()))
383      .transpose()
384      .map_err(|_| RawlerError::DecoderFailed("CR2: invalid model id".to_string()))?;
385    Ok(Cr2Decoder {
386      tiff,
387      rawloader,
388      exif,
389      makernote,
390      mode,
391      xpacket,
392      camera,
393      model_id,
394    })
395  }

430  /// Get lens description by analyzing TIFF tags and makernotes
431  fn get_lens_description(&self) -> Result<Option<&'static LensDescription>> {
432    let exif_lens_name = if let Some(Entry {
433      value: Value::Ascii(lens_id), ..
434    }) = self.exif.get_entry(ExifTag::LensModel)
435    {
436      lens_id.strings().get(0)
437    } else {
438      None
439    };
440    match self.makernote.as_ref().and_then(|mn| mn.get_entry(Cr2MakernoteTag::CameraSettings)) {
441      Some(Entry {
442        value: Value::Short(settings), ..
443      }) => {
444        let lens_info = settings[22];
445        debug!("Lens Info tag: {}", lens_info);
446        let resolver = LensResolver::new()
447          .with_lens_keyname(exif_lens_name)
448          .with_camera(&self.camera) // must follow with_lens_keyname() as it my override key
449          .with_lens_id((lens_info as u32, 0))
450          .with_focal_len(self.get_focal_len()?)
451          .with_mounts(&[CANON_CN_MOUNT.into(), CANON_EF_MOUNT.into()]);
452        return Ok(resolver.resolve());
453      }
454      _ => {
455        log::warn!("Camera settings in makernote not found, no lens data available");
456      }
457    }
458    Ok(None)
459  }

461  /// Parse the Canon makernote IFD
462  fn new_makernote(file: &RawSource, tiff: &GenericTiffReader, exif_ifd: &IFD, _rawloader: &RawLoader) -> Result<Option<IFD>> {
463    if let Some(entry) = exif_ifd.get_entry(TiffCommonTag::Makernote) {
464      let offset = entry.offset().expect("Makernote internal offset is not present but should be");
465      let makernote = tiff.parse_ifd(&mut file.reader(), offset as u32, 0, 0, exif_ifd.endian, &[])?;
466      return Ok(Some(makernote));
467    }
468    info!("No makernote tag found");
469    Ok(None)
470  }

505  /// Build firmware value from string
506  fn get_firmware(&self) -> Result<Option<u32>> {
507    Ok(
508      match self
509        .makernote
510        .as_ref()
511        .and_then(|mn| mn.get_entry(Cr2MakernoteTag::FirmareVer).and_then(|v| v.as_string()))
512      {
513        Some(fw) => {
514          let str: String = fw.chars().filter(|c| c.is_ascii_digit() || c == &'.').collect();
515          let v: Vec<u8> = str.split('.').map(|v| v.parse().expect("Only digits here")).collect();
516          Some(v.iter().rev().enumerate().map(|(i, v)| 10_u32.pow(i as u32 * 3) * *v as u32).sum())
517        }
518        None => None,
519      },
520    )
521  }

523  /// Get the SRAW white balance coefficents from COLORDATA tag
524  /// The offsets are always at offset 78.
525  /// These coefficents are used for SRAW YUV2RGB conversion.
526  fn get_sraw_wb(&self, rawfile: &RawSource, _cam: &Camera) -> Result<[f32; 4]> {
527    if let Some(levels) = self
528      .makernote
529      .as_ref()
530      .and_then(|mn| mn.get_entry_raw(Cr2MakernoteTag::ColorData, &mut rawfile.reader()).transpose())
531      .transpose()?
532    {
533      let offset = 78;
534      return Ok([
535        levels.get_force_u16(offset) as f32,
536        (levels.get_force_u16(offset + 1) as f32 + levels.get_force_u16(offset + 2) as f32) / 2.0,
537        levels.get_force_u16(offset + 3) as f32,
538        f32::NAN,
539      ]);
540    }
541    Ok([f32::NAN, f32::NAN, f32::NAN, f32::NAN])
542  }

544  /// Get the white balance coefficents from COLORDATA tag
545  /// The offsets are different, so we take the offset from camera params.
546  fn get_wb(&self, rawfile: &RawSource, _cam: &Camera) -> Result<[f32; 4]> {
547    if let Some(colordata) = self.makernote.as_ref().and_then(|mn| mn.get_entry(Cr2MakernoteTag::ColorData)) {
548      let raw_wb = colordata::parse_colordata(colordata)?.wb;
549      return Ok(normalize_wb(raw_wb));
550    }
551
552    // TODO: check if these tags belongs to RootIFD or makernote
553    if let Some(levels) = self.tiff.get_entry_raw(TiffCommonTag::Cr2PowerShotWB, &mut rawfile.reader())? {
554      Ok([
555        levels.get_force_u32(3) as f32,
556        levels.get_force_u32(2) as f32,
557        levels.get_force_u32(4) as f32,
558        f32::NAN,
559      ])
560    } else if let Some(levels) = self.tiff.get_entry(TiffCommonTag::Cr2OldWB) {
561      Ok([levels.force_f32(0), levels.force_f32(1), levels.force_f32(2), f32::NAN])
562    } else {
563      // At least the D2000 has no WB

568  /// Get the black level from COLORDATA tag
569  /// The offsets are different, so we take the offset from camera params.
570  fn get_blacklevel(&self, cam: &Camera, cpp: usize) -> Result<Option<BlackLevel>> {
571    if let Some(colordata) = self.makernote.as_ref().and_then(|mn| mn.get_entry(Cr2MakernoteTag::ColorData)) {
572      if let Some(blacklevel) = colordata::parse_colordata(colordata)?.blacklevel {
573        match cpp {
574          1 => return Ok(Some(BlackLevel::new(&blacklevel, cam.cfa.width, cam.cfa.height, cpp))),
575          3 => {
576            let avg = blacklevel.into_iter().sum::<u16>() / 4;
577            let levels: [u16; 3] = [avg, avg, avg];
578            return Ok(Some(BlackLevel::new(&levels, 1, 1, cpp)));
579          }
580          _ => unreachable!(),
581        }
582      }
583    }
584    Ok(None)
585  }

620  /// Interpolate YCbCr (YUV) data
621  fn interpolate_yuv(&self, ljpeg: &LjpegDecompressor, width: usize, _height: usize, image: &mut [u16]) {
622    if ljpeg.super_h() == 1 && ljpeg.super_v() == 1 {
623      return; // No interpolation needed
624    }
625    // Iterate over a block of 3 rows, smaller chunks are okay
626    // but must always a multiple of row width.
627    image.par_chunks_mut(width * 3).for_each(|slice| {
628      // Do horizontal interpolation.
629      // [y1 Cb Cr ] [ y2 . . ] [y1 Cb Cr ] [ y2 . . ] ...
630      if ljpeg.super_h() == 2 {
631        debug_assert_eq!(slice.len() % width, 0);
632        for row in 0..(slice.len() / width) {
633          for col in (6..width).step_by(6) {
634            let pix1 = row * width + col - 6;
635            let pix2 = pix1 + 3;
636            let pix3 = row * width + col;
637            slice[pix2 + 1] = ((slice[pix1 + 1] as i32 + slice[pix3 + 1] as i32 + 1) / 2) as u16;
638            slice[pix2 + 2] = ((slice[pix1 + 2] as i32 + slice[pix3 + 2] as i32 + 1) / 2) as u16;
639          }
640        }
641      }
642      // Do vertical interpolation
643      //          pixel n      pixel n+1       pixel n+2    pixel n+3       ...
644      // row i  : [y1 Cb  Cr ] [ y2 Cb*  Cr* ] [y1 Cb  Cr ] [ y2 Cb*  Cr* ] ...
645      // row i+1: [y3 Cb* Cr*] [ y4 Cb** Cr**] [y3 Cb* Cr*] [ y4 Cb** Cr**] ...
646      // row i+2: [y1 Cb  Cr ] [ y2 Cb*  Cr* ] [y1 Cb  Cr ] [ y2 Cb*  Cr* ] ...
647      // row i+3: [y3 Cb* Cr*] [ y4 Cb** Cr**] [y3 Cb* Cr*] [ y4 Cb** Cr**] ...
648      if ljpeg.super_v() == 2 && slice.len() == width * 3 {
649        for col in (0..width).step_by(3) {
650          let pix1 = col;
651          let pix2 = width + col;
652          let pix3 = 2 * width + col;
653          slice[pix2 + 1] = ((slice[pix1 + 1] as i32 + slice[pix3 + 1] as i32 + 1) / 2) as u16;
654          slice[pix2 + 2] = ((slice[pix1 + 2] as i32 + slice[pix3 + 2] as i32 + 1) / 2) as u16;
655        }
656      }
657    });
658
659    /* Old non-parallel code

687  /// Convert YCbCr (YUV) data to linear RGB
688  fn convert_to_rgb(
689    &self,
690    rawfile: &RawSource,
691    cam: &Camera,
692    ljpeg: &LjpegDecompressor,
693    width: usize,
694    height: usize,
695    image: &mut [u16],
696    dummy: bool,
697  ) -> Result<()> {
698    debug!("YUV2RGB: Regular WB: {:?}", self.get_wb(rawfile, cam));
699    debug!("YUV2RGB: SRAW WB: {:?}", self.get_sraw_wb(rawfile, cam));
700    debug!("Model ID: 0x{:X}", self.model_id.unwrap_or(0));
701    if dummy {
702      return Ok(());
703    }
704
705    let do_interpolate = std::env::var("RAWLER_CR2_YUV_INTERPOLATE")
706      .ok()
707      .map(|id| id.parse::<bool>().expect("RAWLER_CR2_YUV_INTERPOLATE must by of type bool"))
708      .unwrap_or(true);
709    if do_interpolate {
710      self.interpolate_yuv(ljpeg, width, height, image);
711    }
712
713    let coeffs = self.get_sraw_wb(rawfile, cam)?;
714    let (c1, c2, c3) = if cam.find_hint("invert_sraw_wb") {
715      let c1 = (1024.0 * 1024.0 / coeffs[0]) as i32;
716      let c2 = coeffs[1] as i32;
717      let c3 = (1024.0 * 1024.0 / coeffs[2]) as i32;
718      (c1, c2, c3)
719    } else {
720      (coeffs[0] as i32, coeffs[1] as i32, coeffs[2] as i32)
721    };
722
723    // Starting with 40D, sRaw format was introduced. This uses
724    // version 0. With 5D Mark II, version 1 gets used.
725    // And with 5D Mark III, back to version 0 method
726    // but without an offset of 512 for y.
727    let version = if cam.find_hint("sraw_40d") {
728      0
729    } else if cam.find_hint("sraw_new") {
730      2
731    } else {
732      1
733    };
734
735    let fw = self.get_firmware()?;
736    debug!("Firmware: {:?}", fw);
737
738    // This magic comes from dcraw.
739    // Seems to because of rounding during interpolation, we need to
740    // adjust the hue a little bit (only guessing)
741    let hue = match self.model_id {
742      None => 0,
743      Some(model_id) => {
744        if model_id >= 0x80000281 || (model_id == 0x80000218 && fw.unwrap_or(0) > 1000006) {
745          (((ljpeg.super_h() * ljpeg.super_v()) - 1) >> 1) as i32
746        } else {
747          (ljpeg.super_h() * ljpeg.super_v()) as i32
748        }
749      }
750    };
751    debug!("SRAW hue correction: {:?}", hue);
752
753    // Now calculate RGB for each YUV tuple.
754    image.par_chunks_exact_mut(3).for_each(|pix| {
755      let y = pix[0] as i32;
756      let cb = pix[1] as i32 - 16383;
757      let cr = pix[2] as i32 - 16383;
758      match version {
759        0 => {
760          let y = y - 512; // correction for 40D and others
761          let r = c1 * (y + cr);
762          let g = c2 * (y + ((-778 * cb - (cr << 11)) >> 12));
763          let b = c3 * (y + cb);
764          pix[0] = clampbits(r >> 8, 16);
765          pix[1] = clampbits(g >> 8, 16);
766          pix[2] = clampbits(b >> 8, 16);
767        }
768        1 => {
769          // found in EOS 5D Mark II
770          let cb = (cb << 2) + hue;
771          let cr = (cr << 2) + hue;
772          let r = c1 * (y + ((50 * cb + 22929 * cr) >> 14));
773          let g = c2 * (y + ((-5640 * cb - 11751 * cr) >> 14));
774          let b = c3 * (y + ((29040 * cb - 101 * cr) >> 14));
775          pix[0] = clampbits(r >> 8, 16);
776          pix[1] = clampbits(g >> 8, 16);
777          pix[2] = clampbits(b >> 8, 16);
778        }
779        2 => {
780          // found in EOS 5D Mark III and others
781          let r = c1 * (y + cr);
782          let g = c2 * (y + ((-778 * cb - (cr << 11)) >> 12));
783          let b = c3 * (y + cb);
784          pix[0] = clampbits(r >> 8, 16);
785          pix[1] = clampbits(g >> 8, 16);
786          pix[2] = clampbits(b >> 8, 16);
787        }
788        _ => {
789          unreachable!()
790        }
791      }
792    });
793    Ok(())
794  }
795}

797fn normalize_wb(raw_wb: [f32; 4]) -> [f32; 4] {
798  debug!("CR2 raw wb: {:?}", raw_wb);
799  // We never have more then RGB colors so far (no RGBE etc.)
800  // So we combine G1 and G2 to get RGB wb.
801  let div = raw_wb[1]; // G1 should be 1024 and we use this as divisor
802  let mut norm = raw_wb;
803  norm.iter_mut().for_each(|v| {
804    if v.is_normal() {
805      *v /= div
806    }
807  });
808  [norm[0], (norm[1] + norm[2]) / 2.0, norm[3], f32::NAN]
809}
