59impl<'a> Rw2Decoder<'a> {
60  pub fn new(_file: &RawSource, tiff: GenericTiffReader, rawloader: &'a RawLoader) -> Result<Rw2Decoder<'a>> {
61    let raw = {
62      let data = tiff.find_ifds_with_tag(TiffCommonTag::PanaOffsets);
63      if !data.is_empty() {
64        data[0]
65      } else {
66        tiff.find_first_ifd_with_tag(TiffCommonTag::StripOffsets).unwrap()
67      }
68    };
69
70    let width = fetch_tiff_tag!(raw, TiffCommonTag::PanaWidth).force_usize(0);
71    let height = fetch_tiff_tag!(raw, TiffCommonTag::PanaLength).force_usize(0);
72
73    let mode = {
74      let ratio = width * 100 / height;
75      if ratio < 125 {
76        "1:1"
77      } else if ratio < 145 {
78        "4:3"
79      } else if ratio < 165 {
80        "3:2"
81      } else {
82        "16:9"
83      }
84    };
85    let camera = rawloader.check_supported_with_mode(tiff.root_ifd(), mode)?;
86
87    let camera_ifd = if let Some(ifd) = tiff.get_entry(PanasonicTag::CameraIFD) {
88      let buf = ifd.get_data();
89      match IFD::new_root(&mut std::io::Cursor::new(buf), 0) {
90        Ok(ifd) => Some(ifd),
91        Err(_) => None,
92      }
93    } else {
94      None
95    };
96
97    Ok(Rw2Decoder {
98      rawloader,
99      tiff,
100      camera_ifd,
101      camera,
102    })
103  }
104}

107  fn raw_image(&self, file: &RawSource, _params: &RawDecodeParams, dummy: bool) -> Result<RawImage> {
108    let width;
109    let height;
110
111    let (raw, split) = {
112      let data = self.tiff.find_ifds_with_tag(TiffCommonTag::PanaOffsets);
113      if !data.is_empty() {
114        (data[0], true)
115      } else {
116        (self.tiff.find_first_ifd_with_tag(TiffCommonTag::StripOffsets).unwrap(), false)
117      }
118    };
119
120    let compression = raw.get_entry(PanasonicTag::Compression).map(|entry| entry.force_u16(0)).unwrap_or_default(); // TODO BUG
121    //let compression = fetch_tiff_tag!(raw, PanasonicTag::Compression).force_u16(0);
122
123    let raw_format = raw.get_entry(PanasonicTag::RawFormat).map(|entry| entry.force_u16(0)).unwrap_or_default(); // TODO BUG
124
125    let bps = fetch_tiff_tag!(raw, PanasonicTag::BitsPerSample).force_u32(0);
126    let multishot = raw.get_entry(PanasonicTag::Multishot).map(|entry| entry.force_u32(0) == 65536).unwrap_or(false);
127
128    let image = {
129      let data = self.tiff.find_ifds_with_tag(TiffCommonTag::PanaOffsets);
130      if !data.is_empty() {
131        let raw = data[0];
132        width = fetch_tiff_tag!(raw, TiffCommonTag::PanaWidth).force_usize(0);
133        height = fetch_tiff_tag!(raw, TiffCommonTag::PanaLength).force_usize(0);
134        let offset = fetch_tiff_tag!(raw, TiffCommonTag::PanaOffsets).force_usize(0);
135        //let size = fetch_tiff_tag!(raw, TiffCommonTag::StripByteCounts).force_usize(0);
136        log::debug!("PanaOffset: {}", offset);
137        let src = file.subview_until_eof_padded(offset as u64)?; // TODO add size and check all samples
138        Rw2Decoder::decode_panasonic(file, &src, width, height, split, raw_format, bps, self.tiff.root_ifd(), dummy)
139      } else {
140        let raw = self.tiff.find_first_ifd_with_tag(TiffCommonTag::StripOffsets).unwrap();
141        width = fetch_tiff_tag!(raw, TiffCommonTag::PanaWidth).force_usize(0);
142        height = fetch_tiff_tag!(raw, TiffCommonTag::PanaLength).force_usize(0);
143        let offset = fetch_tiff_tag!(raw, TiffCommonTag::StripOffsets).force_usize(0);
144        //let size = fetch_tiff_tag!(raw, TiffCommonTag::StripByteCounts).force_usize(0);
145        log::debug!("StripOffset: {}", offset);
146        let src = file.subview_until_eof_padded(offset as u64)?; // TODO add size and check all samples
147
148        if src.len() >= width * height * 2 {
149          decode_12le_unpacked_left_aligned(&src, width, height, dummy)
150        } else if src.len() >= width * height * 3 / 2 {
151          decode_12le_wcontrol(&src, width, height, dummy)
152        } else {
153          Rw2Decoder::decode_panasonic(file, &src, width, height, split, raw_format, bps, self.tiff.root_ifd(), dummy)
154        }
155      }
156    };
157
158    log::debug!(
159      "RW2 raw: {}, compression: {}, bps: {}, width: {}, height: {}, multishot: {}",
160      raw_format,
161      compression,
162      bps,
163      width,
164      height,
165      multishot
166    );
167
168    let cpp = 1;
169    let blacklevel = self.get_blacklevel()?;
170    let mut camera = self.camera.clone();
171    if let Some(cfa) = self.get_cfa()? {
172      camera.cfa = cfa;
173    }
174    let photometric = RawPhotometricInterpretation::Cfa(CFAConfig::new_from_camera(&camera));
175    let mut img = RawImage::new(camera, image, cpp, normalize_wb(self.get_wb()?), photometric, blacklevel, None, dummy);
176
177    if let Some(area) = self.get_active_area()? {
178      img.active_area = Some(area);
179      img.crop_area = Some(area);
180    } else if let Some(area) = self.get_crop()? {
181      img.crop_area = Some(area);
182    }
183
184    Ok(img)
185  }

187  fn full_image(&self, _file: &RawSource, params: &RawDecodeParams) -> Result<Option<DynamicImage>> {
188    if params.image_index != 0 {
189      return Ok(None);
190    }
191    if let Some(data) = self.tiff.get_entry(PanasonicTag::JpegData) {
192      let buf = data.get_data();
193      let img = image::load_from_memory_with_format(buf, image::ImageFormat::Jpeg)
194        .map_err(|e| RawlerError::DecoderFailed(format!("Unable to load jpeg preview: {:?}", e)))?;
195      return Ok(Some(img));
196    }
197    Ok(None)
198  }

204  fn raw_metadata(&self, _file: &RawSource, _params: &RawDecodeParams) -> Result<RawMetadata> {
205    let mut exif = Exif::new(self.tiff.root_ifd())?;
206    if exif.iso_speed.unwrap_or(0) == 0 && exif.iso_speed_ratings.unwrap_or(0) == 0 && exif.recommended_exposure_index.unwrap_or(0) == 0 {
207      // Use ISO from PanasonicRaw IFD
208      if let Some(iso) = self.tiff.get_entry(PanasonicTag::ISO) {
209        exif.iso_speed_ratings = Some(iso.force_u16(0));
210      }
211    }
212    let mdata = RawMetadata::new_with_lens(&self.camera, exif, self.get_lens_description()?.cloned());
213    Ok(mdata)
214  }

221impl<'a> Rw2Decoder<'a> {
222  fn get_wb(&self) -> Result<[f32; 4]> {
223    if self.tiff.has_entry(PanasonicTag::PanaWBsR) && self.tiff.has_entry(PanasonicTag::PanaWBsB) {
224      let r = fetch_tiff_tag!(self.tiff, PanasonicTag::PanaWBsR).force_u32(0) as f32;
225      let b = fetch_tiff_tag!(self.tiff, PanasonicTag::PanaWBsB).force_u32(0) as f32;
226      Ok([r, 256.0, 256.0, b])
227    } else if self.tiff.has_entry(PanasonicTag::PanaWBs2R) && self.tiff.has_entry(PanasonicTag::PanaWBs2G) && self.tiff.has_entry(PanasonicTag::PanaWBs2B) {
228      let r = fetch_tiff_tag!(self.tiff, PanasonicTag::PanaWBs2R).force_u32(0) as f32;
229      let g = fetch_tiff_tag!(self.tiff, PanasonicTag::PanaWBs2G).force_u32(0) as f32;
230      let b = fetch_tiff_tag!(self.tiff, PanasonicTag::PanaWBs2B).force_u32(0) as f32;
231      Ok([r, g, g, b])
232    } else {
233      Err(RawlerError::DecoderFailed("RW2: Couldn't find WB".to_string()))
234    }
235  }
236
237  fn get_cfa(&self) -> Result<Option<CFA>> {
238    if self.tiff.has_entry(PanasonicTag::CFAPattern) {
239      let pattern = fetch_tiff_tag!(self.tiff, PanasonicTag::CFAPattern).force_u16(0);
240      Ok(Some(match pattern {
241        1 => CFA::new("RGGB"),
242        2 => CFA::new("GRBG"),
243        3 => CFA::new("GBRG"),
244        4 => CFA::new("BGGR"),
245        _ => return Err(format!("RW2: Unknown CFA pattern: {}", pattern).into()),
246      }))
247    } else {
248      Ok(None)
249    }
250  }
251
252  fn get_blacklevel(&self) -> Result<Option<BlackLevel>> {
253    if self.tiff.has_entry(PanasonicTag::BlackLevelRed) {
254      let r = fetch_tiff_tag!(self.tiff, PanasonicTag::BlackLevelRed).force_u16(0);
255      let g = fetch_tiff_tag!(self.tiff, PanasonicTag::BlackLevelGreen).force_u16(0);
256      let b = fetch_tiff_tag!(self.tiff, PanasonicTag::BlackLevelBlue).force_u16(0);
257      Ok(Some(BlackLevel::new(&[r, g, g, b], self.camera.cfa.width, self.camera.cfa.height, 1)))
258    } else {
259      Ok(None)
260    }
261  }

263  /// Get lens description by analyzing TIFF tags and makernotes
264  fn get_lens_description(&self) -> Result<Option<&'static LensDescription>> {
265    const MFT_MOUNT: &str = "MFT-mount";
266    if let Some(ifd) = &self.camera_ifd {
267      if ifd.has_entry(CameraIfdTag::LensTypeMake) && ifd.has_entry(CameraIfdTag::LensTypeModel) {
268        let make_id = fetch_tiff_tag!(ifd, CameraIfdTag::LensTypeMake);
269        let model_id = fetch_tiff_tag!(ifd, CameraIfdTag::LensTypeModel);
270
271        if make_id.value_type() == 3 && model_id.value_type() == 3 {
272          let composite_id = format!(
273            "{:02X} {:02X} {:02X}",
274            make_id.force_u16(0) & 0xFF,
275            model_id.force_u16(0) & 0xFF,
276            model_id.force_u16(0) >> 8
277          );
278          log::debug!("RW2 lens composite ID: {}", composite_id);
279          let resolver = LensResolver::new()
280            .with_camera(&self.camera)
281            .with_olympus_id(Some(composite_id))
282            .with_focal_len(self.get_focal_len()?)
283            .with_mounts(&[MFT_MOUNT.into()]);
284          return Ok(resolver.resolve());
285        } else {
286          log::info!("Unknown value types for lens tags: {}, {}", make_id.value_type(), model_id.value_type());
287        }
288      }
289    }
290    log::warn!("No lens data available");
291    Ok(None)
292  }

294  fn get_focal_len(&self) -> Result<Option<Rational>> {
295    if let Some(exif) = self.tiff.find_first_ifd_with_tag(ExifTag::MakerNotes) {
296      if let Some(Entry {
297        value: Value::Short(focal), ..
298      }) = exif.get_entry(ExifTag::FocalLength)
299      {
300        return Ok(focal.get(1).map(|v| Rational::new(*v as u32, 1)));
301      }
302    }
303    Ok(None)
304  }
305
306  fn get_crop(&self) -> Result<Option<Rect>> {
307    if self.tiff.has_entry(PanasonicTag::CropLeft) {
308      let crop_left = fetch_tiff_tag!(self.tiff, PanasonicTag::CropLeft).force_usize(0);
309      let crop_top = fetch_tiff_tag!(self.tiff, PanasonicTag::CropTop).force_usize(0);
310      let crop_right = fetch_tiff_tag!(self.tiff, PanasonicTag::CropRight).force_usize(0);
311      let crop_bottom = fetch_tiff_tag!(self.tiff, PanasonicTag::CropBottom).force_usize(0);
312      Ok(Some(Rect::new(
313        Point::new(crop_left, crop_top),
314        Dim2::new(crop_right - crop_left, crop_bottom - crop_top),
315      )))
316    } else {
317      Ok(None)
318    }
319  }
320
321  fn get_active_area(&self) -> Result<Option<Rect>> {
322    if self.tiff.has_entry(PanasonicTag::SensorLeftBorder) {
323      let sensor_left = fetch_tiff_tag!(self.tiff, PanasonicTag::SensorLeftBorder).force_usize(0);
324      let sensor_top = fetch_tiff_tag!(self.tiff, PanasonicTag::SensorTopBorder).force_usize(0);
325      let sensor_right = fetch_tiff_tag!(self.tiff, PanasonicTag::SensorRightBorder).force_usize(0);
326      let sensor_bottom = fetch_tiff_tag!(self.tiff, PanasonicTag::SensorBottomBorder).force_usize(0);
327      Ok(Some(Rect::new(
328        Point::new(sensor_left, sensor_top),
329        Dim2::new(sensor_right - sensor_left, sensor_bottom - sensor_top),
330      )))
331    } else {
332      Ok(None)
333    }
334  }

336  pub(crate) fn decode_panasonic(
337    file: &RawSource,
338    buf: &[u8],
339    width: usize,
340    height: usize,
341    split: bool,
342    raw_format: u16,
343    bps: u32,
344    ifd: &IFD,
345    dummy: bool,
346  ) -> PixU16 {
347    log::debug!("width: {}, height: {}, bps: {}", width, height, bps);
348    match raw_format {
349      3 => decode_panasonic_v4(buf, width, height, split, dummy),
350      4 => decode_panasonic_v4(buf, width, height, split, dummy),
351      5 => decode_panasonic_v5(buf, width, height, bps, dummy),
352      6 => decode_panasonic_v6(buf, width, height, bps, dummy),
353      7 => decode_panasonic_v7(buf, width, height, bps, dummy),
354      8 => decode_panasonic_v8(file, width, height, bps, ifd, dummy).unwrap(), // TODO
355      _ => todo!("Format {} is not implemented", raw_format),                  // TODO: return error
356    }
357  }

360fn normalize_wb(raw_wb: [f32; 4]) -> [f32; 4] {
361  log::debug!("RW2 raw wb: {:?}", raw_wb);
362  let div = raw_wb[1];
363  let mut norm = raw_wb;
364  norm.iter_mut().for_each(|v| {
365    if v.is_normal() {
366      *v /= div
367    }
368  });
369  [norm[0], (norm[1] + norm[2]) / 2.0, norm[3], f32::NAN]
370}
