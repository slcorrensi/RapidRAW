13pub fn dump_image_u16(data: &[u16], width: usize, height: usize, path: impl AsRef<str>) {
14  let img = ImageBuffer::<Luma<u16>, Vec<u16>>::from_vec(width as u32, height as u32, data.to_vec()).unwrap();
15  img.save_with_format(path.as_ref(), ImageFormat::Tiff).unwrap();
16}
17
18pub fn dump_image_u16_rgb(data: &[u16], width: usize, height: usize, path: impl AsRef<str>) {
19  let img = ImageBuffer::<Rgb<u16>, Vec<u16>>::from_vec(width as u32, height as u32, data.to_vec()).unwrap();
20  img.save_with_format(path.as_ref(), ImageFormat::Tiff).unwrap();
21}

23pub fn dump_buf<T>(path: &str, buf: T)
24where
25  T: AsRef<[u8]>,
26{
27  let mut f = BufWriter::new(File::create(path).expect("Unable to create file"));
28  f.write_all(buf.as_ref()).expect("Failed to dump buffer to file");
29  f.flush().expect("Failed to flush file");
30}
31
32pub fn dump_buf_u16<T>(path: &str, buf: T)
33where
34  T: AsRef<[u16]>,
35{
36  let mut f = BufWriter::new(File::create(path).expect("Unable to create file"));
37  for v in buf.as_ref().iter() {
38    f.write_u16::<LittleEndian>(*v).expect("Failed to dump buffer to file");
39  }
40  f.flush().expect("Failed to flush file");
41}
