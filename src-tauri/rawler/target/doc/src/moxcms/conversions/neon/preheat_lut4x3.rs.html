29use crate::conversions::neon::hypercube::HypercubeNeon;
30use crate::conversions::neon::interpolator::NeonVector;
31use crate::trc::{lut_interp_linear_float, lut_interp_linear_float_clamped};
32use crate::{CmsError, DataColorSpace, InterpolationMethod, Stage};
33use std::arch::aarch64::{vdupq_n_f32, vgetq_lane_f32, vmaxq_f32, vminq_f32};
34
35#[derive(Default)]
36pub(crate) struct Lut4x3Neon {
37    pub(crate) linearization: [Vec<f32>; 4],
38    pub(crate) clut: Vec<f32>,
39    pub(crate) grid_size: u8,
40    pub(crate) output: [Vec<f32>; 3],
41    pub(crate) interpolation_method: InterpolationMethod,
42    pub(crate) pcs: DataColorSpace,
43}
44
45impl Lut4x3Neon {
46    fn transform_impl<Fetch: Fn(f32, f32, f32, f32) -> NeonVector>(
47        &self,
48        src: &[f32],
49        dst: &mut [f32],
50        fetch: Fetch,
51    ) -> Result<(), CmsError> {
52        let linearization_0 = &self.linearization[0];
53        let linearization_1 = &self.linearization[1];
54        let linearization_2 = &self.linearization[2];
55        let linearization_3 = &self.linearization[3];
56        for (dest, src) in dst.chunks_exact_mut(3).zip(src.chunks_exact(4)) {
57            debug_assert!(self.grid_size as i32 >= 1);
58            let linear_x = lut_interp_linear_float(src[0], linearization_0);
59            let linear_y = lut_interp_linear_float(src[1], linearization_1);
60            let linear_z = lut_interp_linear_float(src[2], linearization_2);
61            let linear_w = lut_interp_linear_float(src[3], linearization_3);
62
63            unsafe {
64                let mut v = fetch(linear_x, linear_y, linear_z, linear_w).v;
65                v = vmaxq_f32(v, vdupq_n_f32(0.));
66                v = vminq_f32(v, vdupq_n_f32(1.));
67
68                let pcs_x =
69                    lut_interp_linear_float_clamped(vgetq_lane_f32::<0>(v), &self.output[0]);
70                let pcs_y =
71                    lut_interp_linear_float_clamped(vgetq_lane_f32::<1>(v), &self.output[1]);
72                let pcs_z =
73                    lut_interp_linear_float_clamped(vgetq_lane_f32::<2>(v), &self.output[2]);
74                dest[0] = pcs_x;
75                dest[1] = pcs_y;
76                dest[2] = pcs_z;
77            }
78        }
79        Ok(())
80    }
81}
82
83macro_rules! dispatch_preheat {
84    ($heater: ident) => {
85        impl Stage for $heater {
86            fn transform(&self, src: &[f32], dst: &mut [f32]) -> Result<(), CmsError> {
87                let l_tbl = HypercubeNeon::new(
88                    &self.clut,
89                    [
90                        self.grid_size,
91                        self.grid_size,
92                        self.grid_size,
93                        self.grid_size,
94                    ],
95                    3,
96                );
97
98                // If Source PCS is LAB trilinear should be used
99                if self.pcs == DataColorSpace::Lab {
100                    return self
101                        .transform_impl(src, dst, |x, y, z, w| l_tbl.quadlinear_vec3(x, y, z, w));
102                }
103
104                match self.interpolation_method {
105                    #[cfg(feature = "options")]
106                    InterpolationMethod::Tetrahedral => {
107                        self.transform_impl(src, dst, |x, y, z, w| l_tbl.tetra_vec3(x, y, z, w))?;
108                    }
109                    #[cfg(feature = "options")]
110                    InterpolationMethod::Pyramid => {
111                        self.transform_impl(src, dst, |x, y, z, w| l_tbl.pyramid_vec3(x, y, z, w))?;
112                    }
113                    #[cfg(feature = "options")]
114                    InterpolationMethod::Prism => {
115                        self.transform_impl(src, dst, |x, y, z, w| l_tbl.prism_vec3(x, y, z, w))?
116                    }
117                    InterpolationMethod::Linear => {
118                        self.transform_impl(src, dst, |x, y, z, w| {
119                            l_tbl.quadlinear_vec3(x, y, z, w)
120                        })?
121                    }
122                }
123                Ok(())
124            }
125        }
126    };
127}
128
129dispatch_preheat!(Lut4x3Neon);
