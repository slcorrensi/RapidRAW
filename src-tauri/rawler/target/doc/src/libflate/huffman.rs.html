1//! Length-limited Huffman Codes.

7const MAX_BITWIDTH: u8 = 15;
8
9#[derive(Debug, Clone, PartialEq, Eq)]
10pub struct Code {
11    pub width: u8,
12    pub bits: u16,
13}
14impl Code {
15    pub fn new(width: u8, bits: u16) -> Self {
16        debug_assert!(width <= MAX_BITWIDTH);
17        Code { width, bits }
18    }
19    fn inverse_endian(&self) -> Self {
20        let mut f = self.bits;
21        let mut t = 0;
22        for _ in 0..self.width {
23            t <<= 1;
24            t |= f & 1;
25            f >>= 1;
26        }
27        Code::new(self.width, t)
28    }
29}

31pub trait Builder: Sized {
32    type Instance;
33    fn set_mapping(&mut self, symbol: u16, code: Code) -> io::Result<()>;
34    fn finish(self) -> Self::Instance;
35    fn restore_canonical_huffman_codes(mut self, bitwidthes: &[u8]) -> io::Result<Self::Instance> {
36        debug_assert!(!bitwidthes.is_empty());
37
38        let mut symbols = bitwidthes
39            .iter()
40            .enumerate()
41            .filter(|&(_, &code_bitwidth)| code_bitwidth > 0)
42            .map(|(symbol, &code_bitwidth)| (symbol as u16, code_bitwidth))
43            .collect::<Vec<_>>();
44        symbols.sort_by_key(|x| x.1);
45
46        let mut code = 0;
47        let mut prev_width = 0;
48        for (symbol, bitwidth) in symbols {
49            code <<= bitwidth - prev_width;
50            self.set_mapping(symbol, Code::new(bitwidth, code))?;
51            code += 1;
52            prev_width = bitwidth;
53        }
54        Ok(self.finish())
55    }
56}

58pub struct DecoderBuilder {
59    table: Vec<u16>,
60    eob_symbol: Option<u16>,
61    safely_peek_bitwidth: Option<u8>,
62    max_bitwidth: u8,
63}
64impl DecoderBuilder {
65    pub fn new(
66        max_bitwidth: u8,
67        safely_peek_bitwidth: Option<u8>,
68        eob_symbol: Option<u16>,
69    ) -> Self {
70        debug_assert!(max_bitwidth <= MAX_BITWIDTH);
71        DecoderBuilder {
72            table: vec![u16::from(MAX_BITWIDTH) + 1; 1 << max_bitwidth],
73            eob_symbol,
74            safely_peek_bitwidth,
75            max_bitwidth,
76        }
77    }
78    pub fn from_bitwidthes(
79        bitwidthes: &[u8],
80        safely_peek_bitwidth: Option<u8>,
81        eob_symbol: Option<u16>,
82    ) -> io::Result<Decoder> {
83        let builder = Self::new(
84            bitwidthes.iter().cloned().max().unwrap_or(0),
85            safely_peek_bitwidth,
86            eob_symbol,
87        );
88        builder.restore_canonical_huffman_codes(bitwidthes)
89    }
90    pub fn safely_peek_bitwidth(&self) -> Option<u8> {
91        self.safely_peek_bitwidth
92    }
93}
94impl Builder for DecoderBuilder {
95    type Instance = Decoder;
96    fn set_mapping(&mut self, symbol: u16, code: Code) -> io::Result<()> {
97        debug_assert!(code.width <= self.max_bitwidth);
98        if Some(symbol) == self.eob_symbol {
99            self.safely_peek_bitwidth = Some(code.width);
100        }
101
102        // `bitwidth` encoded `to` value
103        let value = (symbol << 5) | u16::from(code.width);
104
105        // Sets the mapping to all possible indices
106        let code_be = code.inverse_endian();
107        for padding in 0..(1 << (self.max_bitwidth - code.width)) {
108            let i = ((padding << code.width) | code_be.bits) as usize;
109            if self.table[i] != u16::from(MAX_BITWIDTH) + 1 {
110                #[cfg(feature = "std")]
111                let message = format!(
112                    "Bit region conflict: i={}, old_value={}, new_value={}, symbol={}, code={:?}",
113                    i, self.table[i], value, symbol, code
114                );
115                #[cfg(not(feature = "std"))]
116                let message = "Bit region conflict";
117                return Err(io::Error::new(io::ErrorKind::InvalidData, message));
118            }
119            self.table[i] = value;
120        }
121        Ok(())
122    }
123    fn finish(self) -> Self::Instance {
124        Decoder {
125            table: self.table,
126            safely_peek_bitwidth: cmp::min(
127                self.max_bitwidth,
128                self.safely_peek_bitwidth.unwrap_or(1),
129            ),
130            max_bitwidth: self.max_bitwidth,
131        }
132    }
133}

135#[derive(Debug)]
136pub struct Decoder {
137    table: Vec<u16>,
138    safely_peek_bitwidth: u8,
139    max_bitwidth: u8,
140}
141impl Decoder {
142    pub fn safely_peek_bitwidth(&self) -> u8 {
143        self.safely_peek_bitwidth
144    }
145
146    #[inline(always)]
147    pub fn decode<R>(&self, reader: &mut bit::BitReader<R>) -> io::Result<u16>
148    where
149        R: io::Read,
150    {
151        let v = self.decode_unchecked(reader);
152        reader.check_last_error()?;
153        Ok(v)
154    }
155
156    #[inline(always)]
157    pub fn decode_unchecked<R>(&self, reader: &mut bit::BitReader<R>) -> u16
158    where
159        R: io::Read,
160    {
161        let mut value;
162        let mut bitwidth;
163        let mut peek_bitwidth = self.safely_peek_bitwidth;
164        loop {
165            let code = reader.peek_bits_unchecked(peek_bitwidth);
166            value = self.table[code as usize];
167            bitwidth = (value & 0b1_1111) as u8;
168            if bitwidth <= peek_bitwidth {
169                break;
170            }
171            if bitwidth > self.max_bitwidth {
172                reader.set_last_error(invalid_data_error!("Invalid huffman coded stream"));
173                break;
174            }
175            peek_bitwidth = bitwidth;
176        }
177        reader.skip_bits(bitwidth);
178        value >> 5
179    }
180}

182#[derive(Debug)]
183pub struct EncoderBuilder {
184    table: Vec<Code>,
185}
186impl EncoderBuilder {
187    pub fn new(symbol_count: usize) -> Self {
188        EncoderBuilder {
189            table: vec![Code::new(0, 0); symbol_count],
190        }
191    }
192    pub fn from_bitwidthes(bitwidthes: &[u8]) -> io::Result<Encoder> {
193        let symbol_count = bitwidthes
194            .iter()
195            .enumerate()
196            .filter(|e| *e.1 > 0)
197            .last()
198            .map_or(0, |e| e.0)
199            + 1;
200        let builder = Self::new(symbol_count);
201        builder.restore_canonical_huffman_codes(bitwidthes)
202    }
203    pub fn from_frequencies(symbol_frequencies: &[usize], max_bitwidth: u8) -> io::Result<Encoder> {
204        let max_bitwidth = cmp::min(
205            max_bitwidth,
206            ordinary_huffman_codes::calc_optimal_max_bitwidth(symbol_frequencies),
207        );
208        let code_bitwidthes = length_limited_huffman_codes::calc(max_bitwidth, symbol_frequencies);
209        Self::from_bitwidthes(&code_bitwidthes)
210    }
211}
212impl Builder for EncoderBuilder {
213    type Instance = Encoder;
214    fn set_mapping(&mut self, symbol: u16, code: Code) -> io::Result<()> {
215        debug_assert_eq!(self.table[symbol as usize], Code::new(0, 0));
216        self.table[symbol as usize] = code.inverse_endian();
217        Ok(())
218    }
219    fn finish(self) -> Self::Instance {
220        Encoder { table: self.table }
221    }
222}

224#[derive(Debug, Clone)]
225pub struct Encoder {
226    table: Vec<Code>,
227}
228impl Encoder {
229    #[inline(always)]
230    pub fn encode<W>(&self, writer: &mut bit::BitWriter<W>, symbol: u16) -> io::Result<()>
231    where
232        W: io::Write,
233    {
234        let code = self.lookup(symbol);
235        debug_assert_ne!(code, Code::new(0, 0));
236        writer.write_bits(code.width, code.bits)
237    }
238    #[inline(always)]
239    pub fn lookup(&self, symbol: u16) -> Code {
240        debug_assert!(
241            symbol < self.table.len() as u16,
242            "symbol:{}, table:{}",
243            symbol,
244            self.table.len()
245        );
246        self.table[symbol as usize].clone()
247    }
248    pub fn used_max_symbol(&self) -> Option<u16> {
249        self.table
250            .iter()
251            .rev()
252            .position(|x| x.width > 0)
253            .map(|trailing_zeros| (self.table.len() - 1 - trailing_zeros) as u16)
254    }

257#[allow(dead_code)]
258mod ordinary_huffman_codes {
259    use core::cmp;
260    use dary_heap::BinaryHeap;
261
262    pub fn calc_optimal_max_bitwidth(frequencies: &[usize]) -> u8 {
263        let mut heap = BinaryHeap::new();
264        for &freq in frequencies.iter().filter(|&&f| f > 0) {
265            let weight = -(freq as isize);
266            heap.push((weight, 0_u8));
267        }
268        while heap.len() > 1 {
269            let (weight1, width1) = heap.pop().unwrap();
270            let (weight2, width2) = heap.pop().unwrap();
271            heap.push((weight1 + weight2, 1 + cmp::max(width1, width2)));
272        }
273        let max_bitwidth = heap.pop().map_or(0, |x| x.1);
274        cmp::max(1, max_bitwidth)
275    }

277mod length_limited_huffman_codes {
278    use alloc::{vec, vec::Vec};
279    use core::mem;
280
281    #[derive(Debug, Clone)]
282    struct Node {
283        symbols: Vec<u16>,
284        weight: usize,
285    }
286    impl Node {
287        pub fn empty() -> Self {
288            Node {
289                symbols: vec![],
290                weight: 0,
291            }
292        }
293        pub fn single(symbol: u16, weight: usize) -> Self {
294            Node {
295                symbols: vec![symbol],
296                weight,
297            }
298        }
299        pub fn merge(&mut self, other: Self) {
300            self.weight += other.weight;
301            self.symbols.extend(other.symbols);
302        }
303    }
304
305    /// Reference: [A Fast Algorithm for Optimal Length-Limited Huffman Codes][LenLimHuff.pdf]
306    ///
307    /// [LenLimHuff.pdf]: https://www.ics.uci.edu/~dan/pubs/LenLimHuff.pdf
308    pub fn calc(max_bitwidth: u8, frequencies: &[usize]) -> Vec<u8> {
309        // NOTE: unoptimized implementation
310        let mut source = frequencies
311            .iter()
312            .enumerate()
313            .filter(|&(_, &f)| f > 0)
314            .map(|(symbol, &weight)| Node::single(symbol as u16, weight))
315            .collect::<Vec<_>>();
316        source.sort_by_key(|o| o.weight);
317
318        let weighted =
319            (0..max_bitwidth - 1).fold(source.clone(), |w, _| merge(package(w), source.clone()));
320
321        let mut code_bitwidthes = vec![0; frequencies.len()];
322        for symbol in package(weighted)
323            .into_iter()
324            .flat_map(|n| n.symbols.into_iter())
325        {
326            code_bitwidthes[symbol as usize] += 1;
327        }
328        code_bitwidthes
329    }
330    fn merge(x: Vec<Node>, y: Vec<Node>) -> Vec<Node> {
331        let mut z = Vec::with_capacity(x.len() + y.len());
332        let mut x = x.into_iter().peekable();
333        let mut y = y.into_iter().peekable();
334        loop {
335            let x_weight = x.peek().map(|s| s.weight);
336            let y_weight = y.peek().map(|s| s.weight);
337            if x_weight.is_none() {
338                z.extend(y);
339                break;
340            } else if y_weight.is_none() {
341                z.extend(x);
342                break;
343            } else if x_weight < y_weight {
344                z.push(x.next().unwrap());
345            } else {
346                z.push(y.next().unwrap());
347            }
348        }
349        z
350    }
351    fn package(mut nodes: Vec<Node>) -> Vec<Node> {
352        if nodes.len() >= 2 {
353            let new_len = nodes.len() / 2;
354
355            for i in 0..new_len {
356                nodes[i] = mem::replace(&mut nodes[i * 2], Node::empty());
357                let other = mem::replace(&mut nodes[i * 2 + 1], Node::empty());
358                nodes[i].merge(other);
359            }
360            nodes.truncate(new_len);
361        }
362        nodes
363    }
364}
